public class Solution {
    public int SecondHighest(string s) {
        int first = -1, second = -1;
        foreach (char c in s) {
            if (char.IsDigit(c)) {
                int num = c - '0';
                if (num > first) {
                    second = first;
                    first = num;
                } else if (num < first && num > second) {
                    second = num;
                }
            }
        }
        return second;
    }
}
