var findChampion = function(n, edges) {
    let degree = new Array(n).fill(0);
    edges.forEach(e => {
        degree[e[1]]++;
    });
    let champion = -1;
    for (let i = 0; i < n; i++) {
        if (degree[i] === 0) {
            if (champion === -1) {
                champion = i;
            } else {
                return -1;
            }
        }
    }
    return champion;
};
