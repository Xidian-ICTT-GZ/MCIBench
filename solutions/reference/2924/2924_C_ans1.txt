int findChampion(int n, int** edges, int edgesSize, int* edgesColSize) {
    int degree[n];
    memset(degree, 0, sizeof(degree));
    for (int i = 0; i < edgesSize; i++) {
        degree[edges[i][1]]++;
    }
    int champion = -1;
    for (int i = 0; i < n; i++) {
        if (degree[i] == 0) {
            if (champion == -1) {
                champion = i;
            } else {
                return -1;
            }
        }
    }
    return champion;
}
