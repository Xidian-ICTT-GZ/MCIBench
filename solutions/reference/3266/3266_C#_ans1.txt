public class Solution {
    private long QuickMul(long x, long y, long m) {
        long res = 1;
        while (y > 0) {
            if ((y & 1) == 1) {
                res = (res * x) % m;
            }
            y >>= 1;
            x = (x * x) % m;
        }
        return res;
    }

    public int[] GetFinalState(int[] nums, int k, int multiplier) {
        if (multiplier == 1) {
            return nums;
        }
        int n = nums.Length, mx = 0;
        long m = 1000000007L;
        var pq = new PriorityQueue<long[], long[]>(Comparer<long[]>.Create((x, y) => {
            if (x[0] != y[0]) {
                return x[0].CompareTo(y[0]);
            } else {
                return x[1].CompareTo(y[1]);
            }
        }));
        for (int i = 0; i < n; i++) {
            mx = Math.Max(mx, nums[i]);
            pq.Enqueue(new long[]{nums[i], i}, new long[]{nums[i], i});
        }
        for (; pq.Peek()[0] < mx && k > 0; k--) {
            var x = pq.Dequeue();
            x[0] *= multiplier;
            pq.Enqueue(x, x);
        }
        for (int i = 0; i < n; i++) {
            var x = pq.Dequeue();
            int t = k / n + (i < k % n ? 1 : 0);
            nums[(int)x[1]] = (int)((x[0] % m) * QuickMul(multiplier, t, m) % m);
        }
        return nums;
    }
}
