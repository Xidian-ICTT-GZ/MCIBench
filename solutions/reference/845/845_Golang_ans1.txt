func longestMountain(arr []int) int {
    n := len(arr)
    left := make([]int, n)
    for i := 1; i < n; i++ {
        if arr[i-1] < arr[i] {
            left[i] = left[i-1] + 1
        }
    }
    right := make([]int, n)
    for i := n - 2; i >= 0; i-- {
        if arr[i+1] < arr[i] {
            right[i] = right[i+1] + 1
        }
    }
    ans := 0
    for i, l := range left {
        r := right[i]
        if l > 0 && r > 0 && l+r+1 > ans {
            ans = l + r + 1
        }
    }
    return ans
}
