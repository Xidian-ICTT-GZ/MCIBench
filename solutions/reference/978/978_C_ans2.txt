int maxTurbulenceSize(int* arr, int arrSize) {
    int dp[arrSize][2];
    for (int i = 0; i < arrSize; i++) {
        dp[i][0] = dp[i][1] = 1;
    }
    dp[0][0] = dp[0][1] = 1;
    for (int i = 1; i < arrSize; i++) {
        if (arr[i - 1] > arr[i]) {
            dp[i][0] = dp[i - 1][1] + 1;
        } else if (arr[i - 1] < arr[i]) {
            dp[i][1] = dp[i - 1][0] + 1;
        }
    }

    int ret = 1;
    for (int i = 0; i < arrSize; i++) {
        ret = fmax(ret, dp[i][0]);
        ret = fmax(ret, dp[i][1]);
    }
    return ret;
}
