int maxTurbulenceSize(int* arr, int arrSize) {
    int ret = 1;
    int dp0 = 1, dp1 = 1;
    for (int i = 1; i < arrSize; i++) {
        if (arr[i - 1] > arr[i]) {
            dp0 = dp1 + 1;
            dp1 = 1;
        } else if (arr[i - 1] < arr[i]) {
            dp1 = dp0 + 1;
            dp0 = 1;
        } else {
            dp0 = 1;
            dp1 = 1;
        }
        ret = fmax(ret, dp0);
        ret = fmax(ret, dp1);
    }
    return ret;
}
