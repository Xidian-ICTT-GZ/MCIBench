int longestSubarray(int* nums, int numsSize) {
    int maxValue = nums[0];
    int ans = 1, cnt = 1;
    for (int i = 1; i < numsSize; i++) {
        if (nums[i] > maxValue) {
            ans = cnt = 1;
            maxValue = nums[i];
        } else if (nums[i] < maxValue) {
            cnt = 0;
        } else if (nums[i] == maxValue) {
            cnt++;
        }
        ans = fmax(ans, cnt);
    }
    return ans;
}
