impl Solution {
    pub fn longest_subarray(nums: Vec<i32>) -> i32 {
        let mut max_value = nums[0];
        let mut ans = 1;
        let mut cnt = 1;
        for i in 1..nums.len() {
            if nums[i] > max_value {
                ans = 1;
                cnt = 1;
                max_value = nums[i];
            } else if nums[i] < max_value {
                cnt = 0;
            } else if nums[i] == max_value {
                cnt += 1;
            }
            ans = ans.max(cnt);
        }
        ans
    }
}
