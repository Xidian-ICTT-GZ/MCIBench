var longestSubarray = function(nums) {
    let maxValue = nums[0];
    let ans = 1, cnt = 1;
    for (let i = 1; i < nums.length; i++) {
        if (nums[i] > maxValue) {
            ans = cnt = 1;
            maxValue = nums[i];
        } else if (nums[i] < maxValue) {
            cnt = 0;
        } else if (nums[i] === maxValue) {
            cnt++;
        }
        ans = Math.max(cnt, ans);
    }
    return ans;
};
