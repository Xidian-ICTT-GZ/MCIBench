int longestAlternatingSubarray(int* nums, int numsSize, int threshold){
    int res = 0, dp = 0;
    for (int l = numsSize - 1; l >= 0; l--) {
        if (nums[l] > threshold) {
            dp = 0;
        } else if (l == numsSize - 1 || nums[l] % 2 != nums[l + 1] % 2) {
            dp++;
        } else {
            dp = 1;
        }
        if (nums[l] % 2 == 0 && dp > res) {
            res = dp;
        }
    }
    return res;
}
