var diagonalSort = function(mat) {
    const n = mat.length;
    const m = mat[0].length;
    const diag = new Array(m + n).fill().map(() => []);
    for (let i = 0; i < n; i++) {
        for (let j = 0; j < m; j++) {
            diag[i - j + m].push(mat[i][j]);
        }
    }
    diag.forEach(d => d.sort((a, b) => b - a));
    for (let i = 0; i < n; i++) {
        for (let j = 0; j < m; j++) {
            mat[i][j] = diag[i - j + m].pop();
        }
    }
    return mat;
};
