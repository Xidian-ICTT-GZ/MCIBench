public class Solution {
    public int CountVowelPermutation(int n) {
        long mod = 1_000_000_007;
        long[][] factor =
        {
            new long[]{0, 1, 0, 0, 0}, 
            new long[]{1, 0, 1, 0, 0}, 
            new long[]{1, 1, 0, 1, 1}, 
            new long[]{0, 0, 1, 0, 1}, 
            new long[]{1, 0, 0, 0, 0}
        };

        long[][] res = FastPow(factor, n - 1, mod);
        long ans = 0;
        for (int i = 0; i < 5; ++i) {
            for (int j = 0; j < 5; ++j) {
                ans = (ans + res[i][j]) % mod;
            }
        }
        return (int)ans;
    }

    public long[][] FastPow(long[][] matrix, int n, long mod) {
        int m = matrix.Length;
        long[][] res = new long[m][];
        for (int i = 0; i < m; ++i) {
            res[i] = new long[m];
        }
        long[][] curr = matrix;

        for(int i = 0; i < m; ++i) {
            res[i][i] = 1;
        }
        for (int i = n; i != 0; i >>= 1) {
            if ((i % 2) == 1) {
                res = Multiply(curr, res, mod);
            }
            curr = Multiply(curr, curr, mod);
        }
        return res;
    }

    public long[][] Multiply(long[][] matrixA, long[][] matrixB, long mod) {
        int m = matrixA.Length;
        int n = matrixB[0].Length;
        long[][] res = new long[m][];
        for (int i = 0; i < m; ++i) {
            res[i] = new long[n];
        }

        for (int i = 0; i < m; ++i) {
            for (int j = 0; j < n; ++j) {
                res[i][j] = 0;
                for (int k = 0; k < matrixA[i].Length; ++k) {
                    res[i][j] = (res[i][j] + matrixA[i][k] * matrixB[k][j]) % mod;
                }
            }
        }
        return res;
    }
}
