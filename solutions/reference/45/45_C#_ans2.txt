public class Solution {
    public int Jump(int[] nums) {
        int maxPos = 0, end = 0, steps = 0;
        for (int i = 0; i < nums.Length - 1; ++i) {
            if (maxPos >= i) {
                maxPos = Math.Max(maxPos, i + nums[i]);
                if (i == end) {
                    end = maxPos;
                    ++steps;
                }
            }
        }
        return steps;
    }
}
