class Solution:
    def numDifferentIntegers(self, word: str) -> int:
        s = set()
        n = len(word)
        p1 = 0
        while True:
            while p1 < n and not word[p1].isdigit():
                p1 += 1
            if p1 == n:
                break
            p2 = p1
            while p2 < n and word[p2].isdigit():
                p2 += 1
            while p2 - p1 > 1 and word[p1] == '0':  
                p1 += 1
            s.add(word[p1:p2])
            p1 = p2
        return len(s)
