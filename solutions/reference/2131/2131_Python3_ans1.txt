class Solution:
    def longestPalindrome(self, words: List[str]) -> int:
        freq = Counter(words)   
        res = 0   
        mid = False   
        for word, cnt in freq.items():
            
            rev = word[1] + word[0]   
            if word == rev:
                if cnt % 2 == 1:
                    mid = True
                res += 2 * (cnt // 2 * 2)
            elif word > rev:   
                res += 4 * min(freq[word], freq[rev])
        if mid:
            
            res += 2
        return res
