int base, count, maxCount;
int *answer, answerSize;

void update(int x) {
    if (x == base) {
        ++count;
    } else {
        count = 1;
        base = x;
    }
    if (count == maxCount) {
        answer[answerSize++] = base;
    }
    if (count > maxCount) {
        maxCount = count;
        answerSize = 0;
        answer[answerSize++] = base;
    }
}

int* findMode(struct TreeNode* root, int* returnSize) {
    struct TreeNode *cur = root, *pre = NULL;
    answer = malloc(sizeof(int) * 5001);
    answerSize = count = maxCount = 0;
    while (cur) {
        if (!cur->left) {
            update(cur->val);
            cur = cur->right;
            continue;
        }
        pre = cur->left;
        while (pre->right && pre->right != cur) {
            pre = pre->right;
        }
        if (!pre->right) {
            pre->right = cur;
            cur = cur->left;
        } else {
            pre->right = NULL;
            update(cur->val);
            cur = cur->right;
        }
    }
    *returnSize = answerSize;
    return answer;
}
