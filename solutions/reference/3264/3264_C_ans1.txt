int* getFinalState(int* nums, int numsSize, int k, int multiplier, int* returnSize) {
    int *ret = (int *)malloc(sizeof(int) * numsSize);
    memcpy(ret, nums, sizeof(int) * numsSize);
    while (k--) {
        int m = 0;
        for (int j = 0; j < numsSize; j++) {
            if (ret[j] < ret[m]) {
                m = j;
            }
        }
        ret[m] *= multiplier;
    }
    *returnSize = numsSize;
    return ret;
}
