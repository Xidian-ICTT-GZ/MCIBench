class Solution {
public:
    int countElements(vector<int>& nums) {
        int smallest = *min_element(nums.begin(), nums.end());
        int largest = *max_element(nums.begin(), nums.end());
        int res = 0;
        for (int num: nums) {
            if (smallest < num && num < largest) {
                ++res;
            }
        }
        return res;
    }
};
