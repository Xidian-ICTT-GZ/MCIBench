impl Solution {
    pub fn special_perm(nums: Vec<i32>) -> i32 {
        const MOD: i64 = 1_000_000_007;
        let n = nums.len();
        let mut f = vec![vec![0 as i64; n]; 1 << n];

        for i in 0..n {
            f[1 << i][i] = 1;
        }

        for state in 1..(1 << n) {
            for i in 0..n {
                if state >> i & 1 == 0 {
                    continue;
                }
                for j in 0..n {
                    if i == j || state >> j & 1 == 0 {
                        continue;
                    }
                    let x = nums[i];
                    let y = nums[j];
                    if x % y != 0 && y % x != 0 {
                        continue;
                    }
                    f[state][i] = (f[state][i] + f[state ^ (1 << i)][j]) % MOD;
                }
            }
        }

        let mut sum = 0 as i64;
        for i in 0..n {
            sum = (sum + f[(1 << n) - 1][i]) % MOD;
        }
        sum as i32
    }
}
