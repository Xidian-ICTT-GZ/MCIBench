class Solution:
    def maxHeightOfTriangle(self, red: int, blue: int) -> int:
        def maxHeight(x: int, y: int) -> int:
            i = 1
            while True:
                if i % 2 == 1:
                    x -= i
                    if x < 0:
                        return i - 1
                else:
                    y -= i
                    if y < 0:
                        return i - 1
                i += 1
        
        return max(maxHeight(red, blue), maxHeight(blue, red))
