use std::cmp::max;

impl Solution {
    pub fn max_height_of_triangle(red: i32, blue: i32) -> i32 {
        fn max_height(mut x: i32, mut y: i32) -> i32 {
            let mut i = 1;
            loop {
                if i % 2 == 1 {
                    x -= i;
                    if x < 0 {
                        return i - 1;
                    }
                } else {
                    y -= i;
                    if y < 0 {
                        return i - 1;
                    }
                }
                i += 1;
            }
        }
        max(max_height(red, blue), max_height(blue, red))
    }
}
