int leastInterval(char* tasks, int tasksSize, int n) {
    int freq[26];
    memset(freq, 0, sizeof(freq));
    for (int i = 0; i < tasksSize; ++i) {
        ++freq[tasks[i] - 'A'];
    }

    
    int maxExec = 0;
    for (int i = 0; i < 26; i++) {
        maxExec = fmax(maxExec, freq[i]);
    }
    
    int maxCount = 0;
    for (int i = 0; i < 26; i++) {
        if (maxExec == freq[i]) {
            maxCount++;
        }
    }

    return fmax((maxExec - 1) * (n + 1) + maxCount, tasksSize);
}
