int equalSubstring(char* s, char* t, int maxCost) {
    int n = strlen(s);
    int diff[n];
    memset(diff, 0, sizeof(diff));
    for (int i = 0; i < n; i++) {
        diff[i] = fabs(s[i] - t[i]);
    }
    int maxLength = 0;
    int start = 0, end = 0;
    int sum = 0;
    while (end < n) {
        sum += diff[end];
        while (sum > maxCost) {
            sum -= diff[start];
            start++;
        }
        maxLength = fmax(maxLength, end - start + 1);
        end++;
    }
    return maxLength;
}
