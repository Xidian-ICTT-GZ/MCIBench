public class Solution {
    public int MinimumBoxes(int n) {
        int i = (int) Math.Pow(6.0 * n, 1.0 / 3);
        if (G(i) < n) {
            i++;
        }
        n = (int) (n - G(i - 1));
        int j = (int) Math.Ceiling(1.0 * (Math.Sqrt((long) 8 * n + 1) - 1) / 2);
        return (i - 1) * i / 2 + j;
    }

    public long G(int x) {
        return (long) x * (x + 1) * (x + 2) / 6;
    }
}
