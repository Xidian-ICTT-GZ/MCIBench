public class Solution {
    public int MatrixSum(int[][] nums) {
        int res = 0;
        int m = nums.Length;
        int n = nums[0].Length;
        for (int i = 0; i < m; i++) {
            Array.Sort(nums[i]);
        }
        for (int j = 0; j < n; j++) {
            int maxVal = 0;
            for (int i = 0; i < m; i++) {
                maxVal = Math.Max(maxVal, nums[i][j]);
            }
            res += maxVal;
        }
        return res;
    }
}
