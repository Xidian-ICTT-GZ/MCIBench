long long sumDigitDifferences(int* nums, int numsSize) {
    long long res = 0;
    while (nums[0] > 0) {
        int cnt[10] = {0};
        for (int i = 0; i < numsSize; i++) {
            cnt[nums[i] % 10]++;
            nums[i] /= 10;
        }
        for (int i = 0; i < 10; i++) {
            res += (long long) (numsSize - cnt[i]) * cnt[i];
        }
    }
    return res / 2;
}
