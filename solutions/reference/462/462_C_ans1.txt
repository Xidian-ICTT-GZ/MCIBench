int cmp(const void *pa, const void *pb) {
    return *(int *)pa - *(int *)pb;
}

int minMoves2(int* nums, int numsSize){
    qsort(nums, numsSize, sizeof(int), cmp);
    int ret = 0, x = nums[numsSize / 2];
    for (int i = 0; i < numsSize; i++) {
        ret += abs(nums[i] - x);
    }
    return ret;
}
