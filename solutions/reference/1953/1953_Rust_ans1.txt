impl Solution {
    pub fn number_of_weeks(milestones: Vec<i32>) -> i64 {
        
        let mut longest = 0;
        let mut rest: i64 = 0;
        for &count in milestones.iter() {
            longest = std::cmp::max(longest, count);
            rest += count as i64;
        }
        
        rest -= longest as i64;
        if longest as i64 > rest + 1 {
            
            rest * 2 + 1
        } else {
            
            longest as i64 + rest
        }
    }
}
