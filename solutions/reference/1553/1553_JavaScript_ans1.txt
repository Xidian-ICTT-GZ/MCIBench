var minDays = function(n) {
    const memo = new Map([[0, 0], [1, 1]]);

    function helper(n) {
        if (memo.has(n)) {
            return memo.get(n);
        }
        memo.set(n, Math.min(n % 2 + 1 + helper(Math.floor(n / 2)), n % 3 + 1 + helper(Math.floor(n / 3))));
        return memo.get(n);
    }
    return helper(n);
};
