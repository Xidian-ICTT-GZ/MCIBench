impl Solution {
    pub fn count_hill_valley(nums: Vec<i32>) -> i32 {
        let mut res = 0;   
        let n = nums.len();
        for i in 1..n-1 {
            if nums[i] == nums[i - 1] {
                
                continue;
            }
            let mut left = 0;   
            for j in (0..i).rev() {
                if nums[j] > nums[i] {
                    left = 1;
                    break;
                } else if nums[j] < nums[i] {
                    left = -1;
                    break;
                }
            }
            let mut right = 0;   
            for j in i+1..n {
                if nums[j] > nums[i] {
                    right = 1;
                    break;
                } else if nums[j] < nums[i] {
                    right = -1;
                    break;
                }
            }
            if left == right && left != 0 {
                
                res += 1;
            }
        }
        res
    }
}
