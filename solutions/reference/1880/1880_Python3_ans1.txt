class Solution:
    def isSumEqual(self, firstWord: str, secondWord: str, targetWord: str) -> bool:
        def decode(word: str) -> int:
            res = 0
            for ch in word:
                res *= 10
                res += ord(ch) - ord('a')
            return res
        
        return decode(firstWord) + decode(secondWord) == decode(targetWord)
