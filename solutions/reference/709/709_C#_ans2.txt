public class Solution {
    public string ToLowerCase(string s) {
        StringBuilder sb = new StringBuilder();
        foreach (char ch in s) {
            char chNew = (int) ch >= 65 && (int) ch <= 90 ? (char) (ch | 32) : (char) ch;
            sb.Append(chNew);
        }
        return sb.ToString();
    }
}
