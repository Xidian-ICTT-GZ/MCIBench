public class Solution {
    public int MinFallingPathSum(int[][] grid) {
        int n = grid.Length;
        int[][] d = new int[n][];
        for (int i = 0; i < n; i++) {
            d[i] = new int[n];
            for (int j = 0; j < n; j++) {
                d[i][j] = int.MaxValue;
            }
        }
        for (int i = 0; i < n; i++) {
            d[0][i] = grid[0][i];
        }
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < n; j++) {
                for (int k = 0; k < n; k++) {
                    if (j == k) {
                        continue;
                    }
                    d[i][j] = Math.Min(d[i][j], d[i - 1][k] + grid[i][j]);
                }
            }
        }
        int res = int.MaxValue;
        for (int j = 0; j < n; j++) {
            res = Math.Min(res, d[n - 1][j]);
        }
        return res;
    }
}
