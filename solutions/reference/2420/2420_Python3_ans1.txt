class Solution:
    def goodIndices(self, nums: List[int], k: int) -> List[int]:
        n = len(nums)
        left = [1] * n
        right = [1] * n
        for i in range(1, n):
            if nums[i] <= nums[i - 1]:
                left[i] = left[i - 1] + 1
            if nums[n - i - 1] <= nums[n - i]:
                right[n - i - 1] = right[n - i] + 1
        return [i for i in range(k, n - k) if left[i - 1] >= k and right[i + 1] >= k]
