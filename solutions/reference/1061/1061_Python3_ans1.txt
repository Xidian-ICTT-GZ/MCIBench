class UnionFind:
    def __init__(self, n):
        self.f = list(range(n))

    def find(self, x):
        if self.f[x] != x:
            self.f[x] = self.find(self.f[x])
        return self.f[x]

    def unite(self, x, y):
        x = self.find(x)
        y = self.find(y)
        if x == y:
            return
        
        if x > y:
            x, y = y, x
        self.f[y] = x

class Solution:
    def smallestEquivalentString(self, s1: str, s2: str, baseStr: str) -> str:
        uf = UnionFind(26)
        for a, b in zip(s1, s2):
            uf.unite(ord(a) - ord('a'), ord(b) - ord('a'))
        return ''.join(chr(ord('a') + uf.find(ord(c) - ord('a'))) for c in baseStr)
