var minimumTime = function(time, totalTrips) {
    
    const check = (t) => {
        let cnt = 0;
        for (const period of time) {
            cnt += Math.floor(t / period);
        }
        return cnt >= totalTrips;
    };
    
    
    let l = 1;
    let r = totalTrips * Math.max(...time);
    
    while (l < r) {
        const mid = Math.floor((l + r) / 2);
        if (check(mid)) {
            r = mid;
        } else {
            l = mid + 1;
        }
    }
    return l;
};
