bool check(long long t, int* time, int timeSize, int totalTrips) {
    long long cnt = 0;
    for (int i = 0; i < timeSize; i++) {
        cnt += t / time[i];
    }
    return cnt >= totalTrips;
}

long long minimumTime(int* time, int timeSize, int totalTrips) {
    
    long long l = 1;
    long long r = (long long) totalTrips * time[0];
    for (int i = 1; i < timeSize; i++) {
        if (time[i] > r / totalTrips) {
            r = (long long) totalTrips * time[i];
        }
    }
    
    while (l < r) {
        long long mid = l + (r - l) / 2;
        if (check(mid, time, timeSize, totalTrips)) {
            r = mid;
        } else {
            l = mid + 1;
        }
    }
    return l;
}
