class Solution:
    def minimumTime(self, time: List[int], totalTrips: int) -> int:
        
        def check(t: int) -> bool:
            cnt = 0
            for period in time:
                cnt += t // period
            return cnt >= totalTrips
        
        
        l = 1
        r = totalTrips * max(time)
        
        while l < r:
            mid = l + (r - l) // 2
            if check(mid):
                r = mid
            else:
                l = mid + 1
        return l
