struct ListNode *reverse(struct ListNode *head) {
    struct ListNode dummy = {
        next:NULL
    };
    while (head != NULL) {
        struct ListNode *p = head;
        head = head->next;
        p->next = dummy.next;
        dummy.next = p;
    }
    return dummy.next;
}

struct ListNode *removeNodes(struct ListNode *head) {
    head = reverse(head);
    for (struct ListNode *p = head; p->next != NULL; ) {
        if (p->val > p->next->val) {
            p->next = p->next->next;
        } else {
            p = p->next;
        }
    }
    return reverse(head);
}
