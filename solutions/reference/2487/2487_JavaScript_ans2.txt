var removeNodes = function(head) {
    let st = new Array();
    for (; head != null; head = head.next) {
        st.push(head);
    }
    for (; st.length > 0; st.pop()) {
        if (head == null || st[st.length - 1].val >= head.val) {
            st[st.length - 1].next = head;
            head = st[st.length - 1];
        }
    }
    return head;
};
