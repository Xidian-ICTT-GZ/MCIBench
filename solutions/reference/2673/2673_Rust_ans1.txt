impl Solution {
    pub fn min_increments(n: i32, cost: Vec<i32>) -> i32 {
        let mut ans = 0;
        let mut cost = cost.clone();
        for i in (0..(n as usize) - 1).rev().step_by(2) {
            ans += (cost[i] - cost[i + 1]).abs();
            
            cost[i / 2] += cost[i].max(cost[i + 1]);
        }
        ans
    }
}
