int* rowAndMaximumOnes(int** mat, int matSize, int* matColSize, int* returnSize) {
    int maxOnes = 0;
    int rowIndex = 0;
    for (int i = 0; i < matSize; i++) {
        int tot = 0;
        for (int j = 0; j < matColSize[i]; j++) {
            tot += mat[i][j];
        }
        if (tot > maxOnes) {
            maxOnes = tot;
            rowIndex = i;
        }
    }
    int* result = (int*)malloc(2 * sizeof(int));
    *returnSize = 2;
    result[0] = rowIndex;
    result[1] = maxOnes;
    return result;
}
