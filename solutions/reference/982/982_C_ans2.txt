int countTriplets(int* nums, int numsSize) {
    int *cnt = (int *)calloc(sizeof(int), 1 << 16);
    for (int i = 0; i < numsSize; i++) {
        int x = nums[i];
        for (int j = 0; j < numsSize; j++) {
            int y = nums[j];
            ++cnt[x & y];
        }
    }
    int ans = 0;
    for (int i = 0; i < numsSize; i++) {
        int x = nums[i] ^ 0xffff;
        for (int sub = x; sub; sub = (sub - 1) & x) {
            ans += cnt[sub];
        }
        ans += cnt[0];
    }
    free(cnt);
    return ans;
}
