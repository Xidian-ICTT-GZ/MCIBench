var isArraySpecial = function(nums, queries) {
    const n = nums.length;
    const dp = new Array(n).fill(1);
    for (let i = 1; i < n; i++) {
        if ((nums[i] ^ nums[i - 1]) & 1) {
            dp[i] = dp[i - 1] + 1;
        }
    }

    const res = [];
    for (const [x, y] of queries) {
        res.push(dp[y] >= y - x + 1);
    }
    return res;
};
