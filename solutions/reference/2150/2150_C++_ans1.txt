class Solution {
public:
    vector<int> findLonely(vector<int>& nums) {
        vector<int> res;
        unordered_map<int, int> freq;   
        for (int num: nums) {
            ++freq[num];
        }
        for (int num: nums) {
            if (freq[num-1] == 0 && freq[num+1] == 0 && freq[num] == 1) {
                res.push_back(num);
            }
        }
        return res;
    }
};
