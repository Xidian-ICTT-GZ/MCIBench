class Solution {
public:
    bool checkZeroOnes(string s) {
        int mx0 = 0;
        int mx1 = 0;
        char prev = '#';   
        int cnt = 0;
        for (char ch: s){
            
            if (ch == prev){
                ++cnt;
            }
            
            else{
                if (prev == '0'){
                    mx0 = max(mx0, cnt);
                }
                else if (prev == '1'){
                    mx1 = max(mx1, cnt);
                }
                cnt = 1;
            }
            prev = ch;
        }
        
        if (prev == '0'){
            mx0 = max(mx0, cnt);
        }
        else if (prev == '1'){
            mx1 = max(mx1, cnt);
        }
        return mx1 > mx0;
    }
};
