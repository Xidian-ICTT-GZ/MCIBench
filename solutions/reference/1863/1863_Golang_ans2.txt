func subsetXORSum(nums []int) int {
    res := 0
    n := len(nums)
    for i := 0; i < (1 << n); i++ {
        tmp := 0
        for j := 0; j < n; j++ {
            if (i & (1 << j)) != 0 {
                tmp ^= nums[j]
            }
        }
        res += tmp
    }
    return res
}
