impl Solution {
    pub fn subset_xor_sum(nums: Vec<i32>) -> i32 {
        let mut res = 0;
        let n = nums.len();
        for i in 0..(1 << n) {
            let mut tmp = 0;
            for j in 0..n {
                if (i & (1 << j)) != 0 {
                    tmp ^= nums[j];
                }
            }
            res += tmp;
        }
        res
    }
}
