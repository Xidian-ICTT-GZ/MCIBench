class Solution {
    int res;
    int n;

    
    void dfs(int val, int idx, int[] nums) {
        if (idx == n) {
            
            res += val;
            return;
        }
        
        dfs(val ^ nums[idx], idx + 1, nums);
        
        dfs(val, idx + 1, nums);
    }

    public int subsetXORSum(int[] nums) {
        res = 0;
        n = nums.length;
        dfs(0, 0, nums);
        return res;
    }
}
