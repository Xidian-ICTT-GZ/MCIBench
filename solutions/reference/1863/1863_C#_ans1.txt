public class Solution {
    int res;
    int n;

    
    void Dfs(int val, int idx, int[] nums) {
        if (idx == n) {
            
            res += val;
            return;
        }
        
        Dfs(val ^ nums[idx], idx + 1, nums);
        
        Dfs(val, idx + 1, nums);
    }

    public int SubsetXORSum(int[] nums) {
        res = 0;
        n = nums.Length;
        Dfs(0, 0, nums);
        return res;
    }
}
