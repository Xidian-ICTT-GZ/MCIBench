impl Solution {
    pub fn subset_xor_sum(nums: Vec<i32>) -> i32 {
        fn dfs(val: i32, idx: usize, nums: &[i32]) -> i32 {
            if idx == nums.len() {
                
                return val;
            }
            
            dfs(val ^ nums[idx], idx + 1, nums) + dfs(val, idx + 1, nums)
        }
        
        dfs(0, 0, &nums)
    }
}
