public class Solution {
    int[] nums;
    Random random;

    public Solution(int[] nums) {
        this.nums = nums;
        random = new Random();
    }

    public int Pick(int target) {
        int ans = 0;
        for (int i = 0, cnt = 0; i < nums.Length; ++i) {
            if (nums[i] == target) {
                ++cnt; 
                if (random.Next(cnt) == 0) {
                    ans = i;
                }
            }
        }
        return ans;
    }
}
