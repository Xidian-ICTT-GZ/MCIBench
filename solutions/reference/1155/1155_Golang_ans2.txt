func numRollsToTarget(n int, k int, target int) int {
    mod := int(1e9 + 7)
    f := make([]int, target + 1)
    f[0] = 1;
    for i := 1; i <= n; i++ {
        g := make([]int, target + 1)
        for j := 0; j <= target; j++ {
            for x := 1; x <= k; x++ {
                if j - x >= 0 {
                    g[j] = (g[j] + f[j - x]) % mod
                }
            }
        }
        f = g
    }
    return f[target]
}
