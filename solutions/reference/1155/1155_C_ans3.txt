const int MOD = 1e9 + 7;

int numRollsToTarget(int n, int k, int target) {
    int f[target + 1];
    memset(f, 0, sizeof(f));
    f[0] = 1;
    for (int i = 1; i <= n; ++i) {
        for (int j = target; j >= 0; --j) {
            f[j] = 0;
            for (int x = 1; x <= k; ++x) {
                if (j - x >= 0) {
                    f[j] = (f[j] + f[j - x]) % MOD;
                }
            }
        }
    }
    return f[target];
}
