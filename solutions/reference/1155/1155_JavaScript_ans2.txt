var numRollsToTarget = function(n, k, target) {
    const mod = 1e9 + 7;
    f = new Array(target + 1).fill(0);
    f[0] = 1;
    for (let i = 1; i <= n; i++) {
        g = new Array(target + 1).fill(0);
        for (let j = 0; j <= target; j++) {
            for (let x = 1; x <= k; x++) {
                if (j - x >= 0) {
                    g[j] = (g[j] + f[j - x]) % mod;
                }
            }
        }
        f = g;
    }
    return f[target];
};
