func countWords(words1 []string, words2 []string) int {
    
    freq1 := make(map[string]int)
    freq2 := make(map[string]int)
    for _, w := range words1 {
        freq1[w]++
    }
    for _, w := range words2 {
        freq2[w]++
    }

    
    res := 0
    for w, cnt1 := range freq1 {
        if cnt1 == 1 && freq2[w] == 1 {
            res++
        }
    }
    return res
}
