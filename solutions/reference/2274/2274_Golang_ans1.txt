func maxConsecutive(bottom int, top int, special []int) int {
    special = append(special, bottom - 1)
    special = append(special, top + 1)
    sort.Ints(special)
    ans := 0
    for i := 0; i < len(special) - 1; i++ {
        ans = max(ans, special[i + 1] - special[i] - 1)
    }
    return ans
}
