func maximumNumberOfStringPairs(words []string) int {
    n := len(words)
    ans := 0
    for i := 0; i < n; i++ {
        for j := i + 1; j < n; j++ {
            if words[i][0] == words[j][1] && words[i][1] == words[j][0] {
                ans++
            }
        }
    }
    return ans
}
