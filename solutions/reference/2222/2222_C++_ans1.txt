class Solution {
public:
    long long numberOfWays(string s) {
        int n = s.size();
        int n1 = count(s.begin(), s.end(), '1');   
        long long res = 0;   
        int cnt = 0;   
        for (int i = 0; i < n; ++i) {
            if (s[i] == '1') {
                res += (long long) (i - cnt) * (n - n1 - i + cnt);
                ++cnt;
            } else {
                res += (long long) cnt * (n1 - cnt);
            }
        }
        return res;
    }
};
