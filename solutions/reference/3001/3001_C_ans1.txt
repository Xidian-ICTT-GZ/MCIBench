int minMovesToCaptureTheQueen(int a, int b, int c, int d, int e, int f) {
    
    if (a == e && (c != a || d <= fmin(b, f) || d >= fmax(b, f))) {
        return 1;
    }
    
    if (b == f && (d != b || c <= fmin(a, e) || c >= fmax(a, e))) {
        return 1;
    }
    
    if (abs(c - e) == abs(d - f) && ((c - e) * (b - f) != (a - e) * (d - f) 
        || a < fmin(c, e) || a > fmax(c, e))) {
        return 1;
    }
    return 2;
}
