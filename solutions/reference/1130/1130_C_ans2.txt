int mctFromLeafValues(int* arr, int arrSize) {
    int res = 0;
    int stack[arrSize];
    int top = 0;
    for (int i = 0; i < arrSize; i++) {
        int x = arr[i];
        while (top > 0 && stack[top - 1] <= x) {
            int y = stack[top - 1];
            top--;
            if (top == 0 || stack[top - 1] > x) {
                res += y * x;
            } else {
                res += stack[top - 1] * y;
            }
        }
        stack[top++] = x;
    }
    while (top >= 2) {
        int x = stack[top - 1];
        top--;
        res += stack[top - 1] * x;
    }
    return res;
}
