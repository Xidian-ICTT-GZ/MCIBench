public class Solution {
    public int MaximumRequests(int n, int[][] requests) {
        int[] delta = new int[n];
        int ans = 0, m = requests.Length;
        for (int mask = 0; mask < (1 << m); ++mask) {
            int cnt = BitCount(mask);
            if (cnt <= ans) {
                continue;
            }
            Array.Fill(delta, 0);
            for (int i = 0; i < m; ++i) {
                if ((mask & (1 << i)) != 0) {
                    ++delta[requests[i][0]];
                    --delta[requests[i][1]];
                }
            }
            bool flag = true;
            foreach (int x in delta) {
                if (x != 0) {
                    flag = false;
                    break;
                }
            }
            if (flag) {
                ans = cnt;
            }
        }
        return ans;
    }

    private static int BitCount(int i) {
        i = i - ((i >> 1) & 0x55555555);
        i = (i & 0x33333333) + ((i >> 2) & 0x33333333);
        i = (i + (i >> 4)) & 0x0f0f0f0f;
        i = i + (i >> 8);
        i = i + (i >> 16);
        return i & 0x3f;
    }
}
