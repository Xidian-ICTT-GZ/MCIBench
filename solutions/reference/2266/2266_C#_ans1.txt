using System;
using System.Collections.Generic;

public class Solution {
    public int CountTexts(string pressedKeys) {
        int m = 1000000007;
        int n = pressedKeys.Length;
        List<long> dp3 = new List<long> {1, 1, 2, 4};   
        List<long> dp4 = new List<long> {1, 1, 2, 4};   
        for (int i = 4; i <= n; ++i) {
            dp3.Add((dp3[i-1] + dp3[i-2] + dp3[i-3]) % m);
            dp4.Add((dp4[i-1] + dp4[i-2] + dp4[i-3] + dp4[i-4]) % m);
        }
        long res = 1;   
        int cnt = 1;   
        for (int i = 1; i < n; ++i) {
            if (pressedKeys[i] == pressedKeys[i-1]) {
                ++cnt;
            } else {
                
                if (pressedKeys[i-1] == '7' || pressedKeys[i-1] == '9') {
                    res = (res * dp4[cnt]) % m;
                } else {
                    res = (res * dp3[cnt]) % m;
                }
                cnt = 1;
            }
        }
        
        if (pressedKeys[n-1] == '7' || pressedKeys[n-1] == '9') {
            res = (res * dp4[cnt]) % m;
        } else {
            res = (res * dp3[cnt]) % m;
        }
        return (int) res;
    }
}
