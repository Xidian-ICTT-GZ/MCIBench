public class Solution {
    public long CountSubarrays(int[] nums, int k) {
        int mx = nums.Max();
        long ans = 0;
        int cnt = 0, left = 0;
        foreach (int x in nums) {
            if (x == mx) {
                cnt++;
            }
            while (cnt == k) {
                if (nums[left] == mx) {
                    cnt--;
                }
                left++;
            }
            ans += left;
        }
        return ans;
    }
}
