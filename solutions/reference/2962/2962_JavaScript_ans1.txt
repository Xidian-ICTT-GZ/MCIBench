var countSubarrays = function(nums, k) {
    const n = nums.length;
    const mx = Math.max(...nums);
    const pos = [-1];
    for (let i = 0; i < n; i++) {
        if (nums[i] === mx) {
            pos.push(i);
        }
    }
    let left = 1, right = k;
    let ans = 0;
    while (right < pos.length) {
        ans += (pos[left] - pos[left - 1]) * (n - pos[right]);
        left++;
        right++;
    }
    return ans;
};
