var countSubarrays = function(nums, k) {
    const mx = Math.max(...nums);
    let ans = 0, cnt = 0, left = 0;
    for (const x of nums) {
        if (x === mx) {
            cnt++;
        }
        while (cnt === k) {
            if (nums[left] === mx) {
                cnt--;
            }
            left++;
        }
        ans += left;
    }
    return ans;
};
