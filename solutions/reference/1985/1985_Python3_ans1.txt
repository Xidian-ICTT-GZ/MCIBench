from functools import cmp_to_key
class Solution:
    def kthLargestNumber(self, nums: List[str], k: int) -> str:
        
        def cmp(s1: str, s2: str) -> int:
            
            if len(s1) > len(s2):
                return -1
            elif len(s1) < len(s2):
                return 1
            else:
                
                if s1 > s2:
                    return -1
                else:
                    return 1
            
        nums.sort(key=cmp_to_key(cmp))
        return nums[k-1]
