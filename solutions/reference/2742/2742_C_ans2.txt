int min(int a, int b) {
    return a < b ? a : b;
}

int paintWalls(int *cost, int costSize, int *time, int timeSize){
    int n = costSize;
    int *f = (int *)malloc((n + 2) * sizeof(int));
    for (int i = 0; i < n + 2; i++) {
        f[i] = INT_MAX / 2;
    }
    f[0] = 0;
    for (int i = 0; i < n; ++i) {
        for (int j = n + 1; j >= 0; --j) {
            f[min(j + time[i], n) + 1] = min(f[min(j + time[i], n) + 1], f[j] + cost[i]);
        }
    }
    return min(f[n], f[n + 1]);
}
