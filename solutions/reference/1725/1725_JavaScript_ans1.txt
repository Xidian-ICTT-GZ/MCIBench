var countGoodRectangles = function(rectangles) {
    let res = 0, maxLen = 0;
    for (const rectangle of rectangles) {
        const l = rectangle[0], w = rectangle[1];
        const k = Math.min(l, w);
        if (k === maxLen) {
            ++res;
        } else if (k > maxLen) {
            res = 1;
            maxLen = k;
        }
    }
    return res;
};
