public class Solution {
    public int HeightChecker(int[] heights) {
        int m = heights.Max();
        int[] cnt = new int[m + 1];
        foreach (int h in heights) {
            ++cnt[h];
        }

        int idx = 0, ans = 0;
        for (int i = 1; i <= m; ++i) {
            for (int j = 1; j <= cnt[i]; ++j) {
                if (heights[idx] != i) {
                    ++ans;
                }
                ++idx;
            }
        }
        return ans;
    }
}
