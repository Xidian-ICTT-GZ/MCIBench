int maxDivScore(int* nums, int numsSize, int* divisors, int divisorsSize){
    int cnt = -1, ans = 0;

    for (int i = 0; i < divisorsSize; i++) {
        int tmp = 0;
        for (int j = 0; j < numsSize; j++) {
            if (nums[j] % divisors[i] == 0) {
                tmp++;
            }
        }

        if (tmp > cnt || (tmp == cnt && divisors[i] < ans)) {
            cnt = tmp;
            ans = divisors[i];
        } 
    }
    return ans;
}
