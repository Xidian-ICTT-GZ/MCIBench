impl Solution {
    pub fn max_div_score(nums: Vec<i32>, divisors: Vec<i32>) -> i32 {
        let mut cnt = -1;
        let mut ans = 0;

        for i in 0..divisors.len() {
            let mut tmp = 0;
            for j in 0..nums.len() {
                if nums[j] % divisors[i] == 0 {
                    tmp += 1;
                }
            }

            if tmp > cnt || tmp == cnt && divisors[i] < ans {
                cnt = tmp;
                ans = divisors[i];
            }
        }
        return ans;
    }
}
