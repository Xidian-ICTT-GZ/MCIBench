public class Solution {
    public int MaxDivScore(int[] nums, int[] divisors) {
        int cnt = -1, ans = 0;

        for (int i = 0; i < divisors.Length; i++) {
            int tmp = 0;
            for (int j = 0; j < nums.Length; j++) {
                if (nums[j] % divisors[i] == 0) {
                    tmp++;
                }
            }

            if (tmp > cnt || (tmp == cnt && divisors[i] < ans)) {
                cnt = tmp;
                ans = divisors[i];
            }
        }
        return ans;
    }
}
