long long numberOfRightTriangles(int** grid, int gridSize, int* gridColSize) {
    int n = gridSize, m = gridColSize[0];
    int col[m];
    memset(col, 0, sizeof(col));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            col[j] += grid[i][j];
        }
    }

    long long res = 0;
    for (int i = 0; i < n; i++) {
        int row = 0;
        for (int j = 0; j < m; j++) {
            row += grid[i][j];
        }
        for (int j = 0; j < m; j++) {
            if (grid[i][j] == 1) {
                res += (long long)(row - 1) * (col[j] - 1);
            }
        }
    }
    return res;
}
