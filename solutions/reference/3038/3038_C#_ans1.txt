public class Solution {
    public int MaxOperations(int[] nums) {
        int n = nums.Length, t = 0;
        for (int i = 1; i < n; i += 2) {
            if (nums[i] + nums[i - 1] != nums[1] + nums[0]) {
                break;
            }
            t++;
        }
        return t;
    }
}
