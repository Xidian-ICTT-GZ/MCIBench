public class Solution {
    public int MaximumValue(string[] strs) {
        int res = 0;
        foreach (string s in strs) {
            bool isDigits = true;
            foreach (char c in s) {
                isDigits &= char.IsDigit(c);
            }
            res = Math.Max(res, isDigits? int.Parse(s) : s.Length);
        }
        return res;
    }
}
