func longestSubarray(nums []int) int {
    ans := 0
    p0, p1 := 0, 0
    for _, num := range nums {
        if num == 0 {
            p1 = p0
            p0 = 0
        } else {
            p0++
            p1++
        }
        if p1 > ans {
            ans = p1
        }
    }
    if ans == len(nums) {
        ans--
    }
    return ans
}
