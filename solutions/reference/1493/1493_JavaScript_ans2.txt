var longestSubarray = function(nums) {
    let ans = 0;
    let p0 = 0, p1 = 0;
    for (const num of nums) {
        if (num === 0) {
            p1 = p0;
            p0 = 0;
        } else {
            p0++;
            p1++;
        }
        ans = Math.max(ans, p1);
    }
    if (ans === nums.length) {
        ans--;
    }
    return ans;
};
