public class Solution {
    public int LongestSubarray(int[] nums) {
        int ans = 0;
        int p0 = 0, p1 = 0;
        foreach (int num in nums) {
            if (num == 0) {
                p1 = p0;
                p0 = 0;
            }
            else {
                ++p0;
                ++p1;
            }
            ans = Math.Max(ans, p1);
        }
        if (ans == nums.Length) {
            --ans;
        }
        return ans;
    }
}
