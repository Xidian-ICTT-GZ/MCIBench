char* getSmallestString(char* s) {
    int len = strlen(s);
    for (int i = 0; i + 1 < len; i++) {
        
        if (s[i] > s[i + 1] && (s[i] % 2 == s[i + 1] % 2)) {
            char tmp = s[i];
            s[i] = s[i + 1];
            s[i + 1] = tmp;
            break;
        }
    }
    return s;
}
