class Solution:
    def countUnguarded(self, m: int, n: int, guards: List[List[int]], walls: List[List[int]]) -> int:
        grid = [[0] * n for  _ in range(m)]   
        q = deque([])   
        
        dx = [1, 0, -1, 0]
        dy = [0, 1, 0, -1]
        for i, j in guards:
            grid[i][j] = -1
            for k in range(4):
                
                q.append((i, j, k))
        for i, j in walls:
            grid[i][j] = -2
        while q:
            x, y, k = q.popleft()
            nx, ny = x + dx[k], y + dy[k]
            if 0 <= nx < m and 0 <= ny < n and grid[nx][ny] >= 0:
                
                if grid[nx][ny] & (1 << k) == 0:
                    
                    grid[nx][ny] |= (1 << k)
                    q.append((nx, ny, k))
        res = 0   
        for i in range(m):
            for j in range(n):
                if grid[i][j] == 0:
                    res += 1
        return res
