impl Solution {
    pub fn even_odd_bit(n: i32) -> Vec<i32> {
        let mut res = vec![0, 0];
        let mut i = 0;
        let mut n_mut = n;
        while n_mut > 0 {
            res[i as usize] += n_mut & 1;
            n_mut >>= 1;
            i ^= 1;
        }
        res
    }
}
