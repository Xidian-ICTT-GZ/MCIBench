class Solution {
private:
    
    
    vector<int> dfs(int n, vector<int>& sums) {
        
        
        if (n == 1) {
            if (sums[0] == 0) {
                return {sums[1]};
            }
            if (sums[1] == 0) {
                return {sums[0]};
            }
            return {};
        }

        int d = sums[1] - sums[0];
        
        int left = 0, right = 0;
        vector<int> s, t;
        
        vector<int> used(1 << n);
        while (true) {
            
            while (left < (1 << n) && used[left]) {
                ++left;
            }
            if (left == (1 << n)) {
                break;
            }
            s.push_back(sums[left]);
            used[left] = true;
            
            while (used[right] || sums[right] != sums[left] + d) {
                ++right;
            }
            t.push_back(sums[right]);
            used[right] = true;
        }

        
        vector<int> ans = dfs(n - 1, s);
        if (!ans.empty()) {
            ans.push_back(d);
            return ans;
        }
        
        ans = dfs(n - 1, t);
        if (!ans.empty()) {
            ans.push_back(-d);
            return ans;
        }
        
        return {};
    }

public:
    vector<int> recoverArray(int n, vector<int>& sums) {
        
        sort(sums.begin(), sums.end());
        return dfs(n, sums);
    }
};
