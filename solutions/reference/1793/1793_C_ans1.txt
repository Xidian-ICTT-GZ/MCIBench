int maximumScore(int* nums, int numsSize, int k) {
    int left = k - 1, right = k + 1;
    int ans = 0;
    for (int i = nums[k];; --i) {
        while (left >= 0 && nums[left] >= i) {
            --left;
        }
        while (right < numsSize && nums[right] >= i) {
            ++right;
        }
        ans = fmax(ans, (right - left - 1) * i);
        if (left == -1 && right == numsSize) {
            break;
        }
    }
    return ans;
}
