func maximumScore(nums []int, k int) int {
    n := len(nums)
    left, right := k - 1, k + 1
    ans := 0
    for i := nums[k]; ; i-- {
        for left >= 0 && nums[left] >= i {
            left--
        }
        for right < n && nums[right] >= i {
            right++
        }
        ans = max(ans, (right - left - 1) * i)
        if left == -1 && right == n {
            break
        }
    }
    return ans
}
