class Solution {
    public int longestCycle(int[] edges) {
        int n = edges.length;
        int[] label = new int[n];
        int current_label = 0, ans = -1;
        for (int i = 0; i < n; ++i) {
            if (label[i] != 0) {
                continue;
            }
            int pos = i, start_label = current_label;
            while (pos != -1) {
                ++current_label;
                
                if (label[pos] != 0) {
                    
                    if (label[pos] > start_label) {
                        ans = Math.max(ans, current_label - label[pos]);
                    }
                    break;
                }
                label[pos] = current_label;
                pos = edges[pos];
            }
        }
        return ans;
    }
}
