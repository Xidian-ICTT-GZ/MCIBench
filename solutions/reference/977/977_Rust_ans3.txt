impl Solution {
    pub fn sorted_squares(nums: Vec<i32>) -> Vec<i32> {
        let n = nums.len();
        let mut ans = vec![0; n];
        let (mut i, mut j) = (0, n - 1);
        let mut pos = n - 1;
        while i <= j {
            if nums[i] * nums[i] > nums[j] * nums[j] {
                ans[pos] = nums[i] * nums[i];
                i += 1;
            } else {
                ans[pos] = nums[j] * nums[j];
                j -= 1;
            }
            if pos == 0 { break; }
            pos -= 1;
        }
        ans
    }
}
