var smallestRangeII = function(nums, k) {
    nums.sort((a, b) => a - b);
    const mi = nums[0], ma = nums[nums.length - 1];
    let res = ma - mi;
    for (let i = 0; i < nums.length - 1; i++) {
        const a = nums[i], b = nums[i + 1];
        res = Math.min(res, Math.max(ma - k, a + k) - Math.min(mi + k, b - k));
    }
    return res;
};
