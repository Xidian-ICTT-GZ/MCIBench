func smallestRangeII(nums []int, k int) int {
    sort.Ints(nums)
    mi, ma := nums[0], nums[len(nums) - 1]
    res := ma - mi
    for i := 0; i < len(nums) - 1; i++ {
        a, b := nums[i], nums[i + 1]
        res = min(res, max(ma - k, a + k) - min(mi + k, b - k))
    }
    return res
}

func min(a, b int) int {
    if a < b {
        return a
    }
    return b
}

func max(a, b int) int {
    if a > b {
        return a
    }
    return b
}
