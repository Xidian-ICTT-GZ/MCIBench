func findEvenNumbers(digits []int) []int {
    nums := make(map[int]bool)
    n := len(digits)
    
    for i := 0; i < n; i++ {
        for j := 0; j < n; j++ {
            for k := 0; k < n; k++ {
                
                if i == j || j == k || i == k {
                    continue
                }
                num := digits[i] * 100 + digits[j] * 10 + digits[k]
                if num >= 100 && num % 2 == 0 {
                    nums[num] = true
                }
            }
        }
    }
    
    res := make([]int, 0, len(nums))
    for num := range nums {
        res = append(res, num)
    }
    sort.Ints(res)
    return res
}
