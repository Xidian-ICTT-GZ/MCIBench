func findEvenNumbers(digits []int) []int {
    res := []int{}
    freq := make(map[int]int)
    
    for _, d := range digits {
        freq[d]++
    }
    
    for i := 100; i < 1000; i += 2 {
        freq1 := make(map[int]int)
        num := i
        
        for num > 0 {
            d := num % 10
            freq1[d]++
            num /= 10
        }
        
        isValid := true
        for d, count := range freq1 {
            if freq[d] < count {
                isValid = false
                break
            }
        }
        if isValid {
            res = append(res, i)
        }
    }
    return res
}
