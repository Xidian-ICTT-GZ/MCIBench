func canMakeSquare(grid [][]byte) bool {
    for i := 0; i <= 1; i++ {
        for j := 0; j <= 1; j++ {
            if check(grid, i, j) {
                return true
            }
        }
    }
    return false
}
func check(grid[][]byte, x, y int) bool {
    count := 0
    for i := 0; i <= 1; i++ {
        for j := 0; j <= 1; j++ {
            if grid[x + i][y + j] == 'B' {
                count++
            }
        }
    }
    return count != 2
}
