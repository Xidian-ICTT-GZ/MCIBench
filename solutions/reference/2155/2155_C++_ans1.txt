class Solution {
public:
    vector<int> maxScoreIndices(vector<int>& nums) {
        int n = nums.size();
        
        int presum = 0, best = 0;
        
        vector<int> ans = {0};
        for (int i = 0; i < n; ++i) {
            if (nums[i] == 0) {
                ++presum;
                if (presum > best) {
                    best = presum;
                    ans = {i + 1};
                }
                else if (presum == best) {
                    ans.push_back(i + 1);
                }
            }
            else {
                --presum;
            }
        }
        return ans;
    }
};
