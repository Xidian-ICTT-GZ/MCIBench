class Solution:
    def rearrangeArray(self, nums: List[int]) -> List[int]:
        n = len(nums)
        pos = neg = 0
        ans = list()

        for i in range(n // 2):
            while nums[pos] < 0:
                pos += 1
            ans.append(nums[pos])
            pos += 1
            while nums[neg] > 0:
                neg += 1
            ans.append(nums[neg])
            neg += 1
        
        return ans
