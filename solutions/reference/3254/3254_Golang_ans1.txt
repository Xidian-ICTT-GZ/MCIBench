func resultsArray(nums []int, k int) []int {
    n := len(nums)
    ans := make([]int, n - k + 1)
    for i := range ans {
        ans[i] = -1
    }
    for i := 0; i <= n-k; i++ {
        valid := true
        for j := i + 1; j < i + k; j++ {
            if nums[j] - nums[j - 1] != 1 {
                valid = false
                break
            }
        }
        if valid {
            ans[i] = nums[i + k - 1]
        }
    }
    return ans
}
