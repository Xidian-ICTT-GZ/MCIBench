bool check(int* nums, int numsSize) {
    int x = 0;
    for (int i = 1; i < numsSize; ++i) {
        if (nums[i] < nums[i - 1]) {
            x = i;
            break;
        }
    }
    if (x == 0) {
        return true;
    }
    for (int i = x + 1; i < numsSize; ++i) {
        if (nums[i] < nums[i - 1]) {
            return false;
        }
    }
    return nums[0] >= nums[numsSize - 1];
}
