func mergeNodes(head *ListNode) *ListNode {
    dummy := &ListNode{}
    tail := dummy
    total := 0
    for cur := head.Next; cur != nil; cur = cur.Next {
        if cur.Val == 0 {
            node := &ListNode{Val: total}
            tail.Next = node
            tail = tail.Next
            total = 0
        } else {
            total += cur.Val
        }
    }
    return dummy.Next
}
