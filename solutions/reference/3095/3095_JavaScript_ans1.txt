var minimumSubarrayLength = function(nums, k) {
    let n = nums.length;
    let res = Number.MAX_VALUE;
    for (let i = 0; i < n; i++) {
        let value = 0;
        for (let j = i; j < n; j++) {
            value |= nums[j];
            if (value >= k) {
                res = Math.min(res, j - i + 1);
                break;
            }
        }
    }
    return res === Number.MAX_VALUE ? -1 : res;
};
