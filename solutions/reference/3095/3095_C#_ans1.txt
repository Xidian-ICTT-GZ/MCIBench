public class Solution {
    public int MinimumSubarrayLength(int[] nums, int k) {
        int n = nums.Length;
        int res = int.MaxValue;
        for (int i = 0; i < n; i++) {
            int value = 0;
            for (int j = i; j < n; j++) {
                value |= nums[j];
                if (value >= k) {
                    res = Math.Min(res, j - i + 1);
                    break;
                }
            }
        }
        return res == int.MaxValue ? -1 : res;
    }
}
