char * bestHand(int* ranks, int ranksSize, char* suits, int suitsSize) {
    int suitsSet = 0;
    for (int i = 0; i < suitsSize; i++) {
        suitsSet |= 1 << (suits[i] - 'a');
    }
    if (__builtin_popcount(suitsSet) == 1) {
        return "Flush";
    }
    int h[16];
    memset(h, 0, sizeof(h));
    for (int i = 0; i < ranksSize; i++) {
        h[ranks[i]]++;
    }
    int cnt = 0;
    for (int i = 0; i < 16; i++) {
        if (h[i] > 2) {
            return "Three of a Kind";
        } else if (h[i] == 1) {
            cnt++;
        }
    }
    if (cnt == 5) {
        return "High Card";
    }
    return "Pair";
}
