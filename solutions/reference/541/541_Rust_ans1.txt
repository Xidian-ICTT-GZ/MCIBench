impl Solution {
    pub fn reverse_str(s: String, k: i32) -> String {
        let mut chars: Vec<char> = s.chars().collect();
        let n = chars.len();
        for i in (0..n).step_by((2 * k) as usize) {
            let end = std::cmp::min(i + k as usize, n);
            chars[i..end].reverse();
        }
        chars.into_iter().collect()
    }
}
