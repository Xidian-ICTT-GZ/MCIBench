public class Solution {
    public bool IncreasingTriplet(int[] nums) {
        int n = nums.Length;
        if (n < 3) {
            return false;
        }
        int first = nums[0], second = int.MaxValue;
        for (int i = 1; i < n; i++) {
            int num = nums[i];
            if (num > second) {
                return true;
            } else if (num > first) {
                second = num;
            } else {
                first = num;
            }
        }
        return false;
    }
}
