var smallerNumbersThanCurrent = function(nums) {
    const n = nums.length;
    const ret = [];
    for (let i = 0; i < n; ++i) {
        let cnt = 0;
        for (let j = 0; j < n; ++j) {
            if (nums[j] < nums[i]) {
                cnt++;
            }
        }
        ret[i] = cnt;
    }
    return ret;
};
