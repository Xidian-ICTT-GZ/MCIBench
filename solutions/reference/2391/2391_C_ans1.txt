int garbageCollection(char** garbage, int garbageSize, int* travel, int travelSize) {
    int distance[26] = {0}; 
    int res = 0, curDis = 0;
    for (int i = 0; i < garbageSize; i++) {
        res += strlen(garbage[i]);
        if (i > 0) {
            curDis += travel[i - 1];
        }
        for (int j = 0; garbage[i][j] != '\0'; j++) {
            distance[garbage[i][j] - 'A'] = curDis;
        }
    }
    for (int i = 0; i < 26; i++) {
        res += distance[i];
    }
    return res;
}
