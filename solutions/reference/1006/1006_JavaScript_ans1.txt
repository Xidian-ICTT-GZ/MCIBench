var clumsy = function(n) {
    const stack = [n--];

    let index = 0; 
    while (n > 0) {
        if (index % 4 == 0) {
            stack.push(stack.pop() * n);
        } else if (index % 4 == 1) {
            const cur = stack.pop();
            stack.push(cur > 0 ? Math.floor(cur / n) : Math.ceil(cur / n));
        } else if (index % 4 == 2) {
            stack.push(n);
        } else {
            stack.push(-n);
        }
        index++;
        n--;
    }

    
    let sum = 0;
    stack.forEach((element) => {
        sum += element;
    })
    return sum;
};
