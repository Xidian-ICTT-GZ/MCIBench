func longestEqualSubarray(nums []int, k int) int {
    n := len(nums)
    ans := 0
    cnt := make(map[int]int)
    for i, j := 0, 0; j < n; j++ {
        cnt[nums[j]]++
        
        for j - i + 1 - cnt[nums[i]] > k {
            cnt[nums[i]]--
            i++
        }
        if cnt[nums[j]] > ans {
            ans = cnt[nums[j]]
        }
    }
    return ans
}
