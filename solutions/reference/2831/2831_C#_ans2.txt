public class Solution {
    public int LongestEqualSubarray(IList<int> nums, int k) {
        int n = nums.Count;
        int ans = 0;
        IDictionary<int, int> cnt = new Dictionary<int, int>();
        for (int i = 0, j = 0; j < n; j++) {
            cnt.TryAdd(nums[j], 0);
            cnt[nums[j]]++;
            
            while (j - i + 1 - cnt[nums[i]] > k) {
                cnt[nums[i]]--;
                i++;
            }
            ans = Math.Max(ans, cnt[nums[j]]);
        }
        return ans;
    }
}
