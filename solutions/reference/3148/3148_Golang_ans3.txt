func maxScore(grid [][]int) int {
    m := len(grid)
	n := len(grid[0])
	premin := make([][]int, m)
	for i := range premin {
		premin[i] = make([]int, n)
		for j := range premin[i] {
			premin[i][j] = math.MaxInt32
		}
	}
	ans := math.MinInt32

	for i := 0; i < m; i++ {
		for j := 0; j < n; j++ {
			pre := math.MaxInt32
			if i > 0 {
				pre = min(pre, premin[i - 1][j])
			}
			if j > 0 {
				pre = min(pre, premin[i][j - 1])
			}
			
			if i + j > 0 {
				ans = max(ans, grid[i][j] - pre)
			}
			premin[i][j] = min(pre, grid[i][j])
		}
	}
	return ans
}
