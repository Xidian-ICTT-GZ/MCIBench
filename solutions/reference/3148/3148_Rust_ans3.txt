impl Solution {
    pub fn max_score(grid: Vec<Vec<i32>>) -> i32 {
        let m = grid.len();
        let n = grid[0].len();
        let mut premin = vec![vec![i32::MAX; n]; m];
        let mut ans = i32::MIN;

        for i in 0..m {
            for j in 0..n {
                let mut pre = i32::MAX;
                if i > 0 {
                    pre = pre.min(premin[i - 1][j]);
                }
                if j > 0 {
                    pre = pre.min(premin[i][j - 1]);
                }
                
                if i + j > 0 {
                    ans = ans.max(grid[i][j] - pre);
                }
                premin[i][j] = pre.min(grid[i][j]);
            }
        }

        ans
    }
}
