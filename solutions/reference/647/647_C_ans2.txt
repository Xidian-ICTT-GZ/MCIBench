int countSubstrings(char* s) {
    int n = strlen(s);
    char* t = malloc(sizeof(char) * (n * 2 + 4));
    t[0] = '$', t[1] = '#';
    for (int i = 0; i < n; i++) {
        t[2 * i + 2] = s[i];
        t[2 * i + 3] = '#';
    }
    t[2 * n + 2] = '!';
    t[2 * n + 3] = '\0';
    n = 2 * n + 3;

    int f[n];
    memset(f, 0, sizeof(f));
    int iMax = 0, rMax = 0, ans = 0;
    for (int i = 1; i < n; ++i) {
        
        f[i] = (i <= rMax) ? fmin(rMax - i + 1, f[2 * iMax - i]) : 1;
        
        while (t[i + f[i]] == t[i - f[i]]) ++f[i];
        
        if (i + f[i] - 1 > rMax) {
            iMax = i;
            rMax = i + f[i] - 1;
        }
        
        ans += (f[i] / 2);
    }
    free(t);

    return ans;
}
