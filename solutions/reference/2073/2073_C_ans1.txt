int timeRequiredToBuy(int* tickets, int ticketsSize, int k) {
        int n = ticketsSize;
        int res = 0;
        for (int i = 0; i < n; ++i){
            
            if (i <= k){
                res += fmin(tickets[i], tickets[k]);
            }
            else{
                res += fmin(tickets[i], tickets[k] - 1);
            }
        }
        return res;
}
