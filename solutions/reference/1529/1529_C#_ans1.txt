public class Solution {
    public int MinFlips(string target) {
        int flips = 0;
        char prev = '0';
        int n = target.Length;
        for (int i = 0; i < n; i++) {
            char curr = target[i];
            if (curr != prev) {
                flips++;
                prev = curr;
            }
        }
        return flips;
    }
}
