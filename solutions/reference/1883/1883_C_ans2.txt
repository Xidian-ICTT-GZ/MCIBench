typedef long long LL;

int minSkips(int* dist, int distSize, int speed, int hoursBefore) {
    int n = distSize;
    LL f[n + 1][n + 1];
    for (int i = 0; i <= n; ++i) {
        for (int j = 0; j <= n; ++j) {
            f[i][j] = LLONG_MAX / 2;
        }
    }
    f[0][0] = 0;
    for (int i = 1; i <= n; ++i) {
        for (int j = 0; j <= i; ++j) {
            if (j != i) {
                f[i][j] = fmin(f[i][j], ((f[i - 1][j] + dist[i - 1] - 1) / speed + 1) * speed);
            }
            if (j != 0) {
                f[i][j] = fmin(f[i][j], f[i - 1][j - 1] + dist[i - 1]);
            }
        }
    }
    for (int j = 0; j <= n; ++j) {
        if (f[n][j] <= (LL)hoursBefore * speed) {
            return j;
        }
    }
    return -1;
}
