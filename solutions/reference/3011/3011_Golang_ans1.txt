func canSortArray(nums []int) bool {
    lastCnt := 0
    lastGroupMax := 0
    curGroupMax := 0

    for _, num := range nums {
        curCnt := bits.OnesCount(uint(num))
        if curCnt == lastCnt {
            curGroupMax = max(curGroupMax, num)
        } else {
            lastCnt = curCnt
            lastGroupMax = curGroupMax
            curGroupMax = num
        }
        if num < lastGroupMax {
            return false
        }
    }
    return true
}
