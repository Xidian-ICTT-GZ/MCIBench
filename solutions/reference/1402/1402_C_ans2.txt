int cmp(const void *a, const void *b) {
    return *(int *)b - *(int *)a;
}

int maxSatisfaction(int* satisfaction, int satisfactionSize) {
    qsort(satisfaction, satisfactionSize, sizeof(int), cmp);
    int presum = 0, ans = 0;
    for (int i = 0; i < satisfactionSize; i++) {
        int si = satisfaction[i];
        if (presum + si > 0) {
            presum += si;
            ans += presum;
        } else {
            break;
        }
    }
    return ans;
}
