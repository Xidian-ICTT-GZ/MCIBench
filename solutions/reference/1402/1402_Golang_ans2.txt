func maxSatisfaction(satisfaction []int) int {
    sort.Slice(satisfaction, func(i, j int) bool {
        return satisfaction[i] > satisfaction[j]
    })
    presum, ans := 0, 0
    for _, si := range satisfaction {
        if presum + si > 0 {
            presum += si
            ans += presum
        } else {
            break
        }
    }
    return ans
}
