class Solution:
    def abbreviateProduct(self, left: int, right: int) -> str:
        log_ans = 0.0
        cnt2 = cnt5 = 0
        for i in range(left, right + 1):
            x = i
            while x % 2 == 0:
                cnt2 += 1
                x //= 2
            while x % 5 == 0:
                cnt5 += 1
                x //= 5
            log_ans += log10(i)
        
        cnt0 = min(cnt2, cnt5)
        k = int(log_ans)
        if k + 1 - cnt0 > 10:
            ans_d = log_ans - k
            
            first = int((10 ** ans_d) * 10000)
            
            need2 = need5 = cnt0
            last = 1
            for i in range(left, right + 1):
                x = i
                while need2 and x % 2 == 0:
                    need2 -= 1
                    x //= 2
                while need5 and x % 5 == 0:
                    need5 -= 1
                    x //= 5
                last = last * x % 100000
            
            return f"{first}...{last:05d}e{cnt0}"
        else:
            need2 = need5 = cnt0
            ans = 1
            for i in range(left, right + 1):
                x = i
                while need2 and x % 2 == 0:
                    need2 -= 1
                    x //= 2
                while need5 and x % 5 == 0:
                    need5 -= 1
                    x //= 5
                ans *= x
            return f"{ans}e{cnt0}"
