var occurrencesOfElement = function(nums, queries, x) {
    const indices = [];
        for (let i = 0; i < nums.length; i++) {
            if (nums[i] === x) {
                indices.push(i);
            }
        }
        const res = [];
        for (const q of queries) {
            if (indices.length < q) {
                res.push(-1);
            } else {
                res.push(indices[q - 1]);
            }
        }
        return res;
};
