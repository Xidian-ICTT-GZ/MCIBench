#define MAX(a, b) ((a) > (b) ? (a) : (b))

int maxRepeating(char * sequence, char * word){
    int n = strlen(sequence), m = strlen(word);
    if (n < m) {
        return 0;
    }

    int f[n];
    memset(f, 0, sizeof(f));
    for (int i = m - 1; i < n; ++i) {
        bool valid = true;
        for (int j = 0; j < m; ++j) {
            if (sequence[i - m + j + 1] != word[j]) {
                valid = false;
                break;
            }
        }
        if (valid) {
            f[i] = (i == m - 1 ? 0 : f[i - m]) + 1;
        }
    }
    int res = 0;
    for (int i = 0; i < n; i++) {
        res = MAX(res, f[i]);
    }
    return res;
}
