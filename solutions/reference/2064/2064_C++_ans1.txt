class Solution {
public:
    int minimizedMaximum(int n, vector<int>& quantities) {
        
        auto check = [&](int x) -> bool{
            
            int cnt = 0;
            for (int q: quantities){
                cnt += (q - 1) / x + 1; 
            }
            return cnt <= n;
        };
        
        int l = 1, r = *max_element(quantities.begin(), quantities.end()) + 1;
        
        while (l < r){
            int mid = l + (r - l) / 2;
            if (check(mid)){
                r = mid;
            }
            else{
                l = mid + 1;
            }
        }
        return l;
    }
};
