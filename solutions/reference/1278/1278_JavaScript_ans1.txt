var palindromePartition = function(s, k) {
    const n = s.length;
    const cost = (l, r) => {
        let ret = 0;
        let i = l, j = r;
        while (i < j) {
            ret += (s[i] === s[j] ? 0 : 1);
            i++;
            j--;
        }
        return ret;
    }
    const f = Array.from({ length: n + 1 }, () => Array(k + 1).fill(1000000000));
    f[0][0] = 0;
    for (let i = 1; i <= n; i++) {
        for (let j = 1; j <= Math.min(k, i); j++) {
            if (j === 1) {
                f[i][j] = cost(0, i - 1);
            } else {
                for (let i0 = j - 1; i0 < i; i0++) {
                    f[i][j] = Math.min(f[i][j], f[i0][j - 1] + cost(i0, i - 1));
                }
            }
        }
    }
    return f[n][k];
};
