int cost(int l, int r, const char *s) {
    int ret = 0;
    int i = l, j = r;
    while (i < j) {
        ret += (s[i] == s[j] ? 0 : 1);
        i++;
        j--;
    }
    return ret;
}

int palindromePartition(char* s, int k) {
    int n = strlen(s);
    int f[n + 1][k + 1];
    for (int i = 0; i <= n; i++) {
        for (int j = 0; j <= k; j++) {
            f[i][j] = INT_MAX;
        }
    }
    f[0][0] = 0;
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= (i < k ? i : k); j++) {
            if (j == 1) {
                f[i][j] = cost(0, i - 1, s);
            } else {
                for (int i0 = j - 1; i0 < i; i0++) {
                    f[i][j] = fmin(f[i][j], f[i0][j - 1] + cost(i0, i - 1, s));
                }
            }
        }
    }

    return f[n][k];
}
