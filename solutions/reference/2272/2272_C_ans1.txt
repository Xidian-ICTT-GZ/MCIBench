int max(int a, int b) {
    return a > b ? a : b;
}

int largestVariance(char* s) {
    int n = strlen(s);
    int pos[26][n];
    int posSize[26] = {0};

    for (int i = 0; i < n; i++) {
        int j = s[i] - 'a';
        pos[j][posSize[j]] = i;
        posSize[j]++;
    }
    
    int ans = 0;
    for (int c0 = 0; c0 < 26; c0++) {
        for (int c1 = 0; c1 < 26; c1++) {
            if (c0 != c1 && posSize[c0] > 0 && posSize[c1] > 0) {
                int i = 0, j = 0;
                int f = 0, g = INT_MIN;
                while (i < posSize[c0] || j < posSize[c1]) {
                    if (j == posSize[c1] || (i < posSize[c0] && pos[c0][i] < pos[c1][j])) {
                        f = max(f, 0) + 1;
                        g = g + 1;
                        i++;
                    } else {
                        g = max(f, max(g, 0)) - 1;
                        f = max(f, 0) - 1;
                        j++;
                    }
                    ans = max(ans, g);
                }
            }
        }
    }

    return ans;
}
