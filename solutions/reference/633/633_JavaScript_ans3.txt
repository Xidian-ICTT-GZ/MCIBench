var judgeSquareSum = function(c) {
    for (let base = 2; base * base <= c; base++) {
        
        if (c % base !== 0) {
            continue;
        }

        
        let exp = 0;
        while (c % base == 0) {
            c /= base;
            exp++;
        }

        
        if (base % 4 === 3 && exp % 2 !== 0) {
            return false;
        }
    }

    
    
    return c % 4 !== 3;
};
