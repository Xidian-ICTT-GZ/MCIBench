public class Solution {
    public int MinStartValue(int[] nums) {
        int m = nums.Min();
        if (m >= 0) {
            return 1;
        }
        int left = 1, right = -m * nums.Length + 1;
        while (left < right) {
            int medium = (left + right) / 2;
            if (Valid(medium, nums)) {
                right = medium;
            } else {
                left = medium + 1;
            }
        }
        return left;
    }

    public bool Valid(int startValue, int[] nums) {
        foreach (int num in nums) {
            startValue += num;
            if (startValue <= 0) {
                return false;
            }
        }
        return true;
    }
}
