func max(a, b int) int {
    if a > b {
        return a
    }
    return b
}

func findMaxK(nums []int) int {
    k := -1
    for _, x := range nums {
        for _, y := range nums {
            if -x == y {
                k = max(x, k)
                break
            }
        }
    }
    return k;
}
