var findMaxK = function(nums) {
    nums.sort((a, b) => a - b);
    for (let i = 0, j = nums.length - 1; i < j; j--) {
        while (i < j && nums[i] < -nums[j]) {
            i++;
        }
        if (nums[i] === -nums[j]) {
            return nums[j];
        }
    }
    return -1;
};
