public class Solution {
    int res = 0;

    public int CountTime(string time) {
        char[] arr = time.ToCharArray();
        DFS(arr, 0);
        return res;
    }

    public void DFS(char[] arr, int pos) {
        if (pos == arr.Length) {
            if (Check(arr)) {
                res++;
            }
            return;
        }
        if (arr[pos] == '?') {
            for (int i = 0; i <= 9; i++) {
                arr[pos] = (char) ('0' + i);
                DFS(arr, pos + 1);
                arr[pos] = '?';
            }
        } else {
            DFS(arr, pos + 1);
        }
    }

    public bool Check(char[] arr) {
        int hh = (arr[0] - '0') * 10 + arr[1] - '0';
        int mm = (arr[3] - '0') * 10 + arr[4] - '0';
        return hh < 24 && mm < 60;
    }
}
