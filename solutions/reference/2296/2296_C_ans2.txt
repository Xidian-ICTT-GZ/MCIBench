#define MAX_LEN 1000000

typedef struct {
    char left[MAX_LEN];
    char right[MAX_LEN];
    int leftSize;
    int rightSize;
} TextEditor;

TextEditor* textEditorCreate() {
    TextEditor* editor = (TextEditor*)malloc(sizeof(TextEditor));
    editor->leftSize = 0;
    editor->rightSize = 0;
    return editor;
}

void textEditorAddText(TextEditor* obj, char* text) {
    while (*text) {
        obj->left[obj->leftSize++] = *text;
        text++;
    }
}

int textEditorDeleteText(TextEditor* obj, int k) {
    int count = (k > obj->leftSize) ? obj->leftSize : k;
    obj->leftSize -= count;
    return count;
}

char* textEditorCursorLeft(TextEditor* obj, int k) {
    while (k-- > 0 && obj->leftSize > 0) {
        obj->right[obj->rightSize++] = obj->left[--obj->leftSize];
    }
    int len = (obj->leftSize < 10) ? obj->leftSize : 10;
    char* result = (char*)malloc(len + 1);
    strncpy(result, obj->left + obj->leftSize - len, len);
    result[len] = '\0';
    return result;
}

char* textEditorCursorRight(TextEditor* obj, int k) {
    while (k-- > 0 && obj->rightSize > 0) {
        obj->left[obj->leftSize++] = obj->right[--obj->rightSize];
    }
    int len = (obj->leftSize < 10) ? obj->leftSize : 10;
    char* result = (char*)malloc(len + 1);
    strncpy(result, obj->left + obj->leftSize - len, len);
    result[len] = '\0';
    return result;
}

void textEditorFree(TextEditor* obj) {
    free(obj);
}
