class Node:
    def __init__(self, val: str = ''):
        self.val = val
        self.prev = None
        self.next = None

    def insert(self, val: str):
        node = Node(val)
        node.next = self
        node.prev = self.prev
        if self.prev:
            self.prev.next = node
        self.prev = node

    def remove(self):
        node = self.prev
        self.prev = node.prev
        if node.prev:
            node.prev.next = self

    def range(self, end: 'Node') -> str:
        node = self
        res = ''
        while node != end:
            res += node.val
            node = node.next
        return res

class TextEditor:
    def __init__(self):
        self.cursor = Node()

    def addText(self, text: str) -> None:
        for c in text:
            self.cursor.insert(c)

    def deleteText(self, k: int) -> int:
        count = 0
        while k > 0 and self.cursor.prev is not None:
            self.cursor.remove()
            k, count = k - 1, count + 1
        return count

    def cursorLeft(self, k: int) -> str:
        while k > 0 and self.cursor.prev is not None:
            self.cursor = self.cursor.prev
            k -= 1
        head = self.cursor
        for i in range(10):
            if head.prev is None:
                break
            head = head.prev
        return head.range(self.cursor)

    def cursorRight(self, k: int) -> str:
        while k > 0 and self.cursor.val != '':
            self.cursor = self.cursor.next
            k -= 1
        head = self.cursor
        for i in range(10):
            if head.prev is None:
                break
            head = head.prev
        return head.range(self.cursor)
