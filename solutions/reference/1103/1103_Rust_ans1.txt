impl Solution {
    pub fn distribute_candies(candies: i32, num_people: i32) -> Vec<i32> {
        let mut ans = vec![0; num_people as usize];
        let mut i = 0;
        let mut candies = candies;
        while candies != 0 {
            ans[(i % num_people) as usize] += candies.min(i + 1);
            candies -= candies.min(i + 1);
            i += 1;
        }
        ans
    }
}
