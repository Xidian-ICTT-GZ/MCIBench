public class Solution {
    public IList<int> Postorder(Node root) {
        IList<int> res = new List<int>();
        Helper(root, res);
        return res;
    }

    public void Helper(Node root, IList<int> res) {
        if (root == null) {
            return;
        }
        foreach (Node ch in root.children) {
            Helper(ch, res);
        }
        res.Add(root.val);
    }
}
