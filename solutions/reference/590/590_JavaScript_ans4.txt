var postorder = function(root) {
    const res = [];
    if (root == null) {
        return res;
    }

    const stack = [];
    stack.push(root);
    while (stack.length) {
        const node = stack.pop();
        res.push(node.val);
        for (const item of node.children) {
            stack.push(item);
        }
    }
    res.reverse();
    return res;
};
