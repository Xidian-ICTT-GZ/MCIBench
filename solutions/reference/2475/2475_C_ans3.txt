int unequalTriplets(int* nums, int numsSize) {
    int count[1001] = {0};
    for (int i = 0; i < numsSize; i++) {
        count[nums[i]]++;
    }
    int res = 0, n = numsSize, t = 0;
    for (int i = 0; i <= 1000; i++) {
        int v = count[i];
        res += t * v * (n - t - v);
        t += v;
    }
    return res;
}
