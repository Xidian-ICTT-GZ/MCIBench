public class Solution {
    public int[][] ModifiedMatrix(int[][] matrix) {
        int n = matrix.Length;
        int m = matrix[0].Length;
        for (int j = 0; j < m; j++) {
            int zd = -1;
            for (int i = 0; i < n; i++) {
                zd = Math.Max(zd, matrix[i][j]);
            }
            for (int i = 0; i < n; i++) {
                if (matrix[i][j] == -1) {
                    matrix[i][j] = zd;
                }
            }
        }
        return matrix;
    }
}
