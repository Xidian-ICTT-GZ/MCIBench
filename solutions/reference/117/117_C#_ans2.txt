public class Solution {
    Node last = null, nextStart = null;

    public Node Connect(Node root) {
        if (root == null) {
            return null;
        }
        Node start = root;
        while (start != null) {
            last = null;
            nextStart = null;
            for (Node p = start; p != null; p = p.next) {
                if (p.left != null) {
                    Handle(p.left);
                }
                if (p.right != null) {
                    Handle(p.right);
                }
            }
            start = nextStart;
        }
        return root;
    }

    public void Handle(Node p) {
        if (last != null) {
            last.next = p;
        } 
        if (nextStart == null) {
            nextStart = p;
        }
        last = p;
    }
}
