class Solution:
    def connect(self, root: 'Node') -> 'Node':
        if root is None:
            return None
        queue = deque([root])
        while queue:
            n = len(queue)
            last = None
            for _ in range(n):
                node = queue.popleft()
                if node.left:
                    queue.append(node.left)
                if node.right:
                    queue.append(node.right)
                if last:
                    last.next = node
                last = node
        return root
