class Solution:
    def maximizeWin(self, prizePositions: List[int], k: int) -> int:
        n = len(prizePositions)
        dp = [0] * (n + 1)
        ans = 0
        for i in range(n):
            x = bisect.bisect_left(prizePositions, prizePositions[i] - k)
            ans = max(ans, i - x + 1 + dp[x])
            dp[i + 1] = max(dp[i], i - x + 1)
        return ans
