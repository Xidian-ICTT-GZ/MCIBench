const inf = 0x3f3f3f3f

func minimumSubstringsInPartition(s string) int {
    n := len(s)
	d := make([]int, n + 1)
	for i := range d {
		d[i] = inf
	}
	d[0] = 0

	for i := 1; i <= n; i++ {
		maxCnt := 0
		occCnt := make(map[byte]int)
		for j := i; j >= 1; j-- {
			occCnt[s[j - 1]]++
			if occCnt[s[j-1]] > maxCnt {
				maxCnt = occCnt[s[j-1]]
			}
			if maxCnt * len(occCnt) == (i - j + 1) && d[j - 1] != inf {
				if d[i] > d[j - 1] + 1 {
					d[i] = d[j - 1] + 1
				}
			}
		}
	}
	return d[n]
}
