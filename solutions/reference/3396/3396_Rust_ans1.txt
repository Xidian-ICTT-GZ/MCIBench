use std::collections::HashSet;

impl Solution {
    pub fn minimum_operations(nums: Vec<i32>) -> i32 {
        let mut ans = 0;
        let check_unique = |start: usize| -> bool {
            let mut seen = HashSet::new();
            for &num in &nums[start..] {
                if seen.contains(&num) {
                    return false;
                }
                seen.insert(num);
            }
            true
        };

        for i in (0..nums.len()).step_by(3) {
            if check_unique(i) {
                return ans;
            }
            ans += 1;
        }
        ans
    }
}
