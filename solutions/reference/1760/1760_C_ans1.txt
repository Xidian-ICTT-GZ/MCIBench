#define MAX(a, b) ((a) > (b) ? (a) : (b))

int minimumSize(int* nums, int numsSize, int maxOperations) {
    int left = 1, right = nums[0];
    for (int i = 1; i < numsSize; i++) {
        right = MAX(right, nums[i]);
    }
    int ans = 0;
    while (left <= right) {
        int y = (left + right) / 2;
        long long ops = 0;
        for (int i = 0; i < numsSize; i++) {
            ops += (nums[i] - 1) / y;
        }
        if (ops <= maxOperations) {
            ans = y;
            right = y - 1;
        }
        else {
            left = y + 1;
        }
    }
    return ans;
}
