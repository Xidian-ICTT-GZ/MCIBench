char* breakPalindrome(char *palindrome) {
    int n = strlen(palindrome);
    if (n == 1) {
        return "";
    }
    for (int i = 0; i * 2 + 1 < n; i++) {
        if (palindrome[i] != 'a') {
            palindrome[i] = 'a';
            return palindrome;
        }
    }
    palindrome[n - 1]++;
    return palindrome;
}
