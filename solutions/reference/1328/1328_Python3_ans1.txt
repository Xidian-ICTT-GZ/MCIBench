class Solution:
    def breakPalindrome(self, palindrome: str) -> str:
        n = len(palindrome)
        if n == 1:
            return ""
        data = list(palindrome)
        for i in range(n // 2):
            if data[i] != 'a':
                data[i] = 'a'
                return "".join(data)
        data[-1] = 'b'
        return "".join(data)
