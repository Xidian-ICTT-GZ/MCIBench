public class Solution {
    public bool IsPossibleToSplit(int[] nums) {
        Dictionary<int, int> count = new Dictionary<int, int>();
        foreach (int num in nums) {
            if (count.ContainsKey(num)) {
                count[num]++;
            } else {
                count[num] = 1;
            }
            if (count[num] > 2) {
                return false;
            }
        }
        return true;
    }
}
