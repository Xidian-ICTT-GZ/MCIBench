public class Solution {
    public int LengthLongestPath(string input) {
        int n = input.Length;
        int pos = 0;
        int ans = 0;
        Stack<int> stack = new Stack<int>();

        while (pos < n) {
            
            int depth = 1;
            while (pos < n && input[pos] == '\t') {
                pos++;
                depth++;
            }
            
            bool isFile = false;  
            int len = 0;   
            while (pos < n && input[pos] != '\n') {
                if (input[pos] == '.') {
                    isFile = true;
                }
                len++;
                pos++;
            }
            
            pos++;

            while (stack.Count >= depth) {
                stack.Pop();
            }
            if (stack.Count > 0) {
                len += stack.Peek() + 1;
            }
            if (isFile) {
                ans = Math.Max(ans, len);
            } else {
                stack.Push(len);
            }
        }
        return ans;
    }
}
