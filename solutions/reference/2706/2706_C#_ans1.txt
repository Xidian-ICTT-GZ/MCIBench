public class Solution {
    public int BuyChoco(int[] prices, int money) {
        int fi = int.MaxValue, se = int.MaxValue;
        foreach (int price in prices) {
            if (price < fi) {
                se = fi;
                fi = price;
            } else if (price < se) {
                se = price;
            }
        }
        return money < fi + se ? money : money - fi - se;
    }
}
