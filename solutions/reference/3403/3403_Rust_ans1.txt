impl Solution {
    pub fn answer_string(word: String, num_friends: i32) -> String {
        if num_friends == 1 {
            return word;
        }
        let n = word.len();
        let mut res = String::new();
        for i in 0 ..= n - 1 {
            let s = &word[i .. std::cmp::min(i + n - (num_friends as usize) + 1, n)];
            if res < s.to_string() {
                res = s.to_string();
            }
        }
        res
    }
}
