impl Solution {
    pub fn average(salary: Vec<i32>) -> f64 {
        let mut max_value = i32::MIN;
        let mut min_value = i32::MAX;
        let mut sum = 0;
        for &s in salary.iter() {
            if s > max_value {
                max_value = s;
            }
            if s < min_value {
                min_value = s;
            }
            sum += s;
        }
        sum -= max_value + min_value;
        sum as f64 / (salary.len() as f64 - 2.0)
    }
}
