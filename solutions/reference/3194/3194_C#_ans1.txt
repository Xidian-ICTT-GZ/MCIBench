public class Solution {
    public double MinimumAverage(int[] nums) {
        Array.Sort(nums);
        int n = nums.Length;
        double res = double.MaxValue;
        for (int i = 0; i < n / 2; i++) {
            res = Math.Min(res, (nums[i] + nums[n - 1 - i]) / 2.0);
        }
        return res;
    }
}
