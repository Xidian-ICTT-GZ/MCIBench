var maximumUniqueSubarray = function(nums) {
    const n = nums.length;
    const seen = new Set();
    let ans = 0, psum = 0;
    for (let i = 0, j = 0; i < n; ++i) {
        psum += nums[i];
        while (seen.has(nums[i])) {
            seen.delete(nums[j]);
            psum -= nums[j];
            j++;
        }
        seen.add(nums[i]);
        ans = Math.max(ans, psum);
    }
    return ans;
};
