class Solution {
public:
    int maximumUniqueSubarray(vector<int>& nums) {
        int n = nums.size();
        vector<int> psum(n + 1);
        unordered_map<int, int> cnt;
        int ans = 0, pre = 0;
        for(int i = 0; i < n; ++i) {
            psum[i + 1] = psum[i] + nums[i];
            pre = max(pre, cnt[nums[i]]);
            ans = max(ans, psum[i + 1] - psum[pre]);
            cnt[nums[i]] = i + 1;
        }
        
        return ans;
    }
};
