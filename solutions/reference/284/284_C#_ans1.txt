class PeekingIterator {
    private IEnumerator<int> iterator;
    private bool flag;

    public PeekingIterator(IEnumerator<int> iterator) {
        this.iterator = iterator;
        flag = true;
    }
    
    public int Peek() {
        return iterator.Current;
    }
    
    public int Next() {
        int ret = iterator.Current;
        flag = iterator.MoveNext();
        return ret;
    }
    
    public bool HasNext() {
        return flag;
    }
}
