public class Solution {
    public long MaximumOr(int[] nums, int k) {
        long orSum = 0, multiBits = 0;
        foreach (int x in nums) {
            multiBits |= x & orSum;
            orSum |= x;
        }
        long res = 0;
        foreach (int x in nums) {
            res = Math.Max(res, (orSum ^ x) | ((long)x << k) | multiBits);
        }
        return res;
    }
}
