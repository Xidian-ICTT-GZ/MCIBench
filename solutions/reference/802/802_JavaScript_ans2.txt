var eventualSafeNodes = function(graph) {
    const n = graph.length;
    const rg = new Array(n).fill(0).map(() => new Array());
    const inDeg = new Array(n).fill(0);
    for (let x = 0; x < n; ++x) {
        for (let y of graph[x]) {
            rg[y].push(x);
        }
        inDeg[x] = graph[x].length;
    }

    const queue = [];
    for (let i = 0; i < n; ++i) {
        if (inDeg[i] === 0) {
            queue.push(i);
        }
    }
    while (queue.length) {
        const y = queue.shift();
        for (const x of rg[y]) {
            if (--inDeg[x] === 0) {
                queue.push(x);
            }
        }
    }

    const ans = [];
    for (let i = 0; i < n; ++i) {
        if (inDeg[i] === 0) {
            ans.push(i);
        }
    }
    return ans;
};
