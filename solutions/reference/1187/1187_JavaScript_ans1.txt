const INF = 0x3f3f3f3f;
var makeArrayIncreasing = function(arr1, arr2) {
    arr2.sort((a, b) => a - b);
    const list = [];
    let prev = -1;
    for (const num of arr2) {
        if (num !== prev) {
            list.push(num);
            prev = num;
        }
    }
    const n = arr1.length;
    const m = list.length;
    const dp = new Array(n + 1).fill(0).map(() => new Array(Math.min(m, n) + 1).fill(INF));
    dp[0][0] = -1;
    for (let i = 1; i <= n; i++) {
        for (let j = 0; j <= Math.min(i, m); j++) {
            
            if (arr1[i - 1] > dp[i - 1][j]) {
                dp[i][j] = arr1[i - 1];
            }
            if (j > 0 && dp[i - 1][j - 1] !== INF) {
                
                const idx = binarySearch(list, j - 1, dp[i - 1][j - 1]);
                if (idx !== list.length) {
                    dp[i][j] = Math.min(dp[i][j], list[idx]);
                }
            }
            if (i === n && dp[i][j] !== INF) {
                return j;
            }
        }
    }
    return -1;
}

const binarySearch = (list, low, target) => {
    let high = list.length;
    while (low < high) {
        const mid = low + Math.floor((high - low) / 2);
        if (list[mid] > target) {
            high = mid;
        } else {
            low = mid + 1;
        }
    }
    return low;
};
