var wiggleMaxLength = function(nums) {
    const n = nums.length;
    if (n < 2) { 
        return n;
    }

    let up = down = 1;
    for (let i = 1; i < n; i++) {
        if (nums[i] > nums[i - 1]) {
            up = down + 1;
        } else if (nums[i] < nums[i - 1]) {
            down = up + 1;
        }
    }
    return Math.max(up, down);
};
