public class Solution {
    public bool FindSubarrays(int[] nums) {
        int n = nums.Length;
        ISet<int> seen = new HashSet<int>();
        for (int i = 0; i < n - 1; ++i) {
            int sum = nums[i] + nums[i + 1];
            if (!seen.Add(sum)) {
                return true;
            }
        }
        return false;
    }
}
