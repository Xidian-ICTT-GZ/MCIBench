int stoneGameVII(int* stones, int stonesSize) {
    int n = stonesSize;
    int sum[n + 1];
    int dp[n][n];
    memset(dp, 0, sizeof(dp));
    memset(sum, 0, sizeof(sum));

    for (int i = 0; i < n; i++) {
        sum[i + 1] = sum[i] + stones[i];
    }
    for (int i = n - 2; i >= 0; i--) {
        for (int j = i + 1; j < n; j++) {
            dp[i][j] = fmax(sum[j + 1] - sum[i + 1] - dp[i + 1][j], sum[j] - sum[i] - dp[i][j - 1]);
        }
    }
    return dp[0][n - 1];
}
