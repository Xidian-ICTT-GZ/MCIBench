public class Solution {
    
    public int Maxlen(int[] candidates, int k) {
        int res = 0;
        foreach (int num in candidates) {
            if ((num & (1 << k)) != 0) {
                res++;
            }
        }
        return res;
    }

    public int LargestCombination(int[] candidates) {
        int res = 0;
        for (int i = 0; i < 24; i++) {
            
            res = Math.Max(res, Maxlen(candidates, i));
        }
        return res;
    }
}
