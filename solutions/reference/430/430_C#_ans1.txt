public class Solution {
    public Node Flatten(Node head) {
        DFS(head);
        return head;
    }

    public Node DFS(Node node) {
        Node cur = node;
        
        Node last = null;

        while (cur != null) {
            Node next = cur.next;
            
            if (cur.child != null) {
                Node childLast = DFS(cur.child);

                next = cur.next;
                
                cur.next = cur.child;
                cur.child.prev = cur;

                
                if (next != null) {
                    childLast.next = next;
                    next.prev = childLast;
                }

                
                cur.child = null;
                last = childLast;
            } else {
                last = cur;
            }
            cur = next;
        }
        return last;
    }
}
