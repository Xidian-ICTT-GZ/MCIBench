var reachableNodes = function(n, edges, restricted) {
    const isRestricted = new Array(n).fill(0);
    for (const x of restricted) {
        isRestricted[x] = 1;
    }

    const g = new Array(n).fill(null).map(() => []);
    for (const [u, v] of edges) {
        g[u].push(v);
        g[v].push(u);
    }

    let cnt = 0;
    const dfs = (x, f) => {
        cnt++;
        for (const y of g[x]) {
            if (y !== f && !isRestricted[y]) {
                dfs(y, x);
            }
        }
    }
    dfs(0, -1);
    return cnt;
};
