func getKth(lo int, hi int, k int) int {
    v := []int{}
	for i := lo; i <= hi; i++ {
		v = append(v, i)
	}
	sort.Slice(v, func(i, j int) bool {
		if getF(v[i]) != getF(v[j]) {
			return getF(v[i]) < getF(v[j])
		}
		return v[i] < v[j]
	})
	return v[k - 1]
}

func getF(x int) int {
	if x == 1 {
		return 0
	}
	if x & 1 == 1 {
		return getF(x * 3 + 1) + 1
	} else {
		return getF(x / 2) + 1
	}
}
