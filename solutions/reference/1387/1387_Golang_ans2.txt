var f = make(map[int]int)

func getF(x int) int {
	if val, exists := f[x]; exists {
		return val
	}
	if x == 1 {
		f[x] = 0
		return 0
	}
	if x&1 == 1 {
		f[x] = getF(x * 3 + 1) + 1
	} else {
		f[x] = getF(x / 2) + 1
	}
	return f[x]
}

func getKth(lo int, hi int, k int) int {
    v := make([]int, 0)
	for i := lo; i <= hi; i++ {
		v = append(v, i)
	}
	sort.Slice(v, func(i, j int) bool {
		if getF(v[i]) != getF(v[j]) {
			return getF(v[i]) < getF(v[j])
		}
		return v[i] < v[j]
	})
	return v[k - 1]
}
