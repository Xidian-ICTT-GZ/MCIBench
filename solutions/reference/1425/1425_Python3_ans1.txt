class Solution:
    def constrainedSubsetSum(self, nums: List[int], k: int) -> int:
        n = len(nums)
        
        
        f = [nums[0]] + [0] * (n - 1)
        
        
        q = collections.deque([0])
        
        ans = nums[0]
        for i in range(1, n):
            
            while q and i - q[0] > k:
                q.popleft()
            
            f[i] = max(f[q[0]], 0) + nums[i]
            ans = max(ans, f[i])
            
            while q and f[i] >= f[q[-1]]:
                q.pop()
            
            q.append(i)
        return ans
