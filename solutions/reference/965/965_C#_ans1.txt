public class Solution {
    public bool IsUnivalTree(TreeNode root) {
        if (root == null) {
            return true;
        }
        if (root.left != null) {
            if (root.val != root.left.val || !IsUnivalTree(root.left)) {
                return false;
            }
        }
        if (root.right != null) {
            if (root.val != root.right.val || !IsUnivalTree(root.right)) {
                return false;
            }
        }
        return true;
    }
}
