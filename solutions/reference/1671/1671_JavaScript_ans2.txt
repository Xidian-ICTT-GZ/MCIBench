var minimumMountainRemovals = function(nums) {
    const n = nums.length;
    const pre = getLISArray(nums)
    let suf = getLISArray(nums.reverse());
    suf.reverse();

    let ans = 0;
    for (let i = 0; i < n; ++i) {
        if (pre[i] > 1 && suf[i] > 1) {
            ans = Math.max(ans, pre[i] + suf[i] - 1);
        }
    }

    return n - ans;
};

var getLISArray = function(nums) {
    const n = nums.length;
    const dp = new Array(n).fill(1);
    const seq = new Array();
    for (let i = 0; i < n; ++i) {
        let index = binarySearch(seq, nums[i]);
        if (index == seq.length) {
            seq.push(nums[i]);
            dp[i] = seq.length;
        } else {
            seq[index] = nums[i];
            dp[i] = index + 1;
        }
    }
    console.log(dp);
    return dp;
}

var binarySearch = function(seq, target) {
    let low = 0, high = seq.length;
    while (low < high) {
        let mid = low + Math.floor((high - low) / 2);
        if (seq[mid] >= target) {
            high = mid;
        } else {
            low = mid + 1;
        }
    }
    return low;
}
