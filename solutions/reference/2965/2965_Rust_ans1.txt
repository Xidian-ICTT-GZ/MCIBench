impl Solution {
    pub fn find_missing_and_repeated_values(grid: Vec<Vec<i32>>) -> Vec<i32> {
        let n = grid.len();
        let mut count = vec![0; n * n + 1];
        for row in grid.iter() {
            for &val in row.iter() {
                count[val as usize] += 1;
            }
        }

        let mut res = vec![0; 2];
        for i in 1..=n * n {
            if count[i as usize] == 2 {
                res[0] = i as i32;
            }
            if count[i as usize] == 0 {
                res[1] = i as i32;
            }
        }
        res
    }
}
