class Solution:
    def countVowelSubstrings(self, word: str) -> int:
        n = len(word)
        res = 0
        vowelset = set("aeiou")   
        for i in range(n):
            
            charset = set()   
            for j in range(i, n):
                
                charset.add(word[j])
                if charset == vowelset:
                    res += 1
        return res
