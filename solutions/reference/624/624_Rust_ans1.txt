impl Solution {
    pub fn max_distance(arrays: Vec<Vec<i32>>) -> i32 {
        let mut res = 0;
        let n = arrays.len();
        for i in 0..n - 1 {
            for &x in &arrays[i] {
                for k in i + 1..n {
                    for &y in &arrays[k] {
                        res = res.max((x - y).abs());
                    }
                }
            }
        }
        res
    }
}
