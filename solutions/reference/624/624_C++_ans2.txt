class Solution {
public:
    int maxDistance(vector<vector<int>>& arrays) {
        vector<int> array1, array2;
        int res = 0;
        int n = arrays.size();
        for (int i = 0; i < n - 1; i++) {
            for (int j = i + 1; j < n; j++) {
                array1 = arrays[i];
                array2 = arrays[j];
                res = max(res, abs(array1[0] - array2[array2.size() - 1]));
                res = max(res, abs(array2[0] - array1[array1.size() - 1]));
            }
        }
        return res;
    }
};
