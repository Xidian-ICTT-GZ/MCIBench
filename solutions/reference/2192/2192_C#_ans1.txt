public class Solution {
    public IList<IList<int>> GetAncestors(int n, int[][] edges) {
        ISet<int>[] anc = new ISet<int>[n];   
        for (int i = 0; i < n; ++i) {
            anc[i] = new HashSet<int>();
        }
        IList<int>[] e = new IList<int>[n];   
        for (int i = 0; i < n; ++i) {
            e[i] = new List<int>();
        }
        int[] indeg = new int[n];   
        
        foreach (int[] edge in edges) {
            e[edge[0]].Add(edge[1]);
            ++indeg[edge[1]];
        }
        
        Queue<int> q = new Queue<int>();
        for (int i = 0; i < n; ++i) {
            if (indeg[i] == 0) {
                q.Enqueue(i);
            }
        }
        while (q.Count > 0) {
            int u = q.Dequeue();
            foreach (int v in e[u]) {
                
                anc[v].Add(u);
                foreach (int i in anc[u]) {
                    anc[v].Add(i);
                }
                --indeg[v];
                if (indeg[v] == 0) {
                    q.Enqueue(v);
                }
            }
        }
        
        IList<IList<int>> res = new List<IList<int>>();
        for (int i = 0; i < n; ++i) {
            res.Add(new List<int>());
            foreach (int j in anc[i]) {
                res[i].Add(j);
            }
            ((List<int>) res[i]).Sort();
        }
        return res;
    }
}
