func findMinimumTime(tasks [][]int) int {
    n := len(tasks)
    sort.Slice(tasks, func(i, j int) bool {
        return tasks[i][1] < tasks[j][1]
    })
    run := make([]int, tasks[n - 1][1] + 1)
    res := 0
    for i := 0; i < n; i++ {
        start, end, duration := tasks[i][0], tasks[i][1], tasks[i][2]
        for j := start; j <= end; j++ {
            duration -= run[j]
        }
        res += max(duration, 0)
        for j := end; j >= 0 && duration > 0; j-- {
            if run[j] == 0 {
                duration--
                run[j] = 1
            }
        }
    }
    return res
}
