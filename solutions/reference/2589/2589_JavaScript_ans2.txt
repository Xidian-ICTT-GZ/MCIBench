var findMinimumTime = function(tasks) {
    let res = 0;
    for (let i = 1; ; i++) {
        let finished = true, run = false;
        for (const task of tasks) {
            if (task[2] > 0 && task[1] - i + 1 === task[2]) {
                run = true;
            }
            if (i <= task[1]) {
                finished = false;
            }
        }
        if (finished) {
            break;
        }

        if (run) {
            for (const task of tasks) {
                if (i >= task[0] && i <= task[1] && task[2] > 0) {
                    task[2]--;
                }
            }
            res++;
        }
    }
    return res;
};
