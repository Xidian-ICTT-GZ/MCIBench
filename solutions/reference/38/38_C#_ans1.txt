public class Solution {
    public string CountAndSay(int n) {
        string str = "1";
        for (int i = 2; i <= n; ++i) {
            StringBuilder sb = new StringBuilder();
            int start = 0;
            int pos = 0;

            while (pos < str.Length) {
                while (pos < str.Length && str[pos] == str[start]) {
                    pos++;
                }
                sb.Append(pos - start).Append(str[start]);
                start = pos;
            }
            str = sb.ToString();
        }
        
        return str;
    }
}
