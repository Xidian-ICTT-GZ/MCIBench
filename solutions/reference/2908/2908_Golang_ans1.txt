func min(a, b int) int {
    if (a < b) {
        return a
    }
    return b
}

func minimumSum(nums []int) int {
    n := len(nums)
    res := 1000
    for i := 0; i < n; i++ {
        for j := i + 1; j < n; j++ {
            for k := j + 1; k < n; k++ {
                if nums[i] < nums[j] && nums[k] < nums[j] {
                    res = min(res, nums[i] + nums[j] + nums[k])
                }
            }
        }
    }
    if res < 1000 {
        return res
    }
    return -1
}
