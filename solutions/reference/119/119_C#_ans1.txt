public class Solution {
    public IList<int> GetRow(int rowIndex) {
        List<List<int>> C = new List<List<int>>();
        for (int i = 0; i <= rowIndex; ++i) {
            C.Add(new List<int>(new int[i + 1]));
            C[i][0] = C[i][i] = 1;
            for (int j = 1; j < i; ++j) {
                C[i][j] = C[i - 1][j - 1] + C[i - 1][j];
            }
        }
        return C[rowIndex];
    }
}
