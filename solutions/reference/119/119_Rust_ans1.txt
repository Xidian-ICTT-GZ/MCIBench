impl Solution {
    pub fn get_row(row_index: i32) -> Vec<i32> {
        let mut C: Vec<Vec<i32>> = Vec::with_capacity((row_index + 1) as usize);
        for i in 0..=row_index {
            let mut row: Vec<i32> = vec![1; (i + 1) as usize];
            for j in 1..i {
                row[j as usize] = C[(i - 1) as usize][(j - 1) as usize] + C[(i - 1) as usize][j as usize];
            }
            C.push(row);
        }
        C[row_index as usize].clone()
    }
}
