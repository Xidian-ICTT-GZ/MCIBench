impl Solution {
    pub fn final_position_of_snake(n: i32, commands: Vec<String>) -> i32 {
        let mut ans = 0;
        for c in commands {
            match c.chars().next().unwrap() {
                'U' => ans -= n,
                'D' => ans += n,
                'L' => ans -= 1,
                _ => ans += 1,
            }
        }
        ans
    }
}
