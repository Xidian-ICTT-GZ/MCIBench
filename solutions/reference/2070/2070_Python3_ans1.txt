class Solution:
    def maximumBeauty(self, items: List[List[int]], queries: List[int]) -> List[int]:
        
        items.sort(key=lambda x: x[0])
        n = len(items)
        
        for i in range(1, n):
            items[i][1] = max(items[i][1], items[i-1][1])
        
        def query(q: int) -> int:
            l, r = 0, n
            while l < r:
                mid = l + (r - l) // 2
                if items[mid][0] > q:
                    r = mid
                else:
                    l = mid + 1
            if l == 0:
                
                return 0
            else:
                
                return items[l-1][1]
        
        res = [query(q) for q in queries]
        return res
