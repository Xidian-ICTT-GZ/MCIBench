public class Solution {
    public int FindPairs(int[] nums, int k) {
        ISet<int> visited = new HashSet<int>();
        ISet<int> res = new HashSet<int>();
        foreach (int num in nums) {
            if (visited.Contains(num - k)) {
                res.Add(num - k);
            }
            if (visited.Contains(num + k)) {
                res.Add(num);
            }
            visited.Add(num);
        }
        return res.Count;
    }
}
