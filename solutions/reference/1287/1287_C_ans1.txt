int findSpecialInteger(int* arr, int arrSize) {
    int cur = arr[0], cnt = 0;
    for (int i = 0; i < arrSize; ++i) {
        if (arr[i] == cur) {
            ++cnt;
            if (cnt * 4 > arrSize) {
                return cur;
            }
        } else {
            cur = arr[i];
            cnt = 1;
        }
    }
    return -1;
}
