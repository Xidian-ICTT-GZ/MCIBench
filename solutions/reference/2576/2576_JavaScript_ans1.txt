var maxNumOfMarkedIndices = function(nums) {
    nums.sort((a, b) => a - b);
    const n = nums.length;
    let l = 0, r = Math.floor(n / 2);
    const check = (m) => {
        for (let i = 0; i < m; i++) {
            if (nums[i] * 2 > nums[n - m + i]) {
                return false;
            }
        }
        return true;
    };
    while (l < r) {
        const m = Math.floor((l + r + 1) / 2);
        if (check(m)) {
            l = m;
        } else {
            r = m - 1;
        }
    }
    return l * 2;
};
