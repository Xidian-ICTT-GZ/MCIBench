const dir = [[-1, 0], [1, 0], [0, -1], [0, 1]];
var closedIsland = function(grid) {
    const m = grid.length;
    const n = grid[0].length;
    let ans = 0;
            
    for (let i = 0; i < m; i++) {
        for (let j = 0; j < n; j++) {
            if (grid[i][j] === 0) {
                const qu = [];
                grid[i][j] = 1;
                let closed = true;

                qu.push([i, j]);
                while (qu.length) {
                    const arr = qu.shift();
                    let cx = arr[0], cy = arr[1];
                    if (cx === 0 || cy === 0 || cx === m - 1 || cy === n - 1) {
                        closed = false;
                    }
                    for (let d = 0; d < 4; d++) {
                        let nx = cx + dir[d][0];
                        let ny = cy + dir[d][1];
                        if (nx >= 0 && nx < m && ny >= 0 && ny < n && grid[nx][ny] === 0) {
                            grid[nx][ny] = 1;
                            qu.push([nx, ny]);
                        }
                    }
                }
                if (closed) {
                    ans++;
                }
            }
        }
    }
    return ans;
};
