class Solution:
    def minDifference(self, nums: List[int]) -> int:
        if len(nums) <= 4:
            return 0

        n = len(nums)
        maxn = [-10**9] * 4
        minn = [10**9] * 4

        for i in range(n):
            add = 0
            while add < 4 and maxn[add] > nums[i]:
                add += 1
            if add < 4:
                maxn[add:] = [nums[i]] + maxn[add:-1]
            
            add = 0
            while add < 4 and minn[add] < nums[i]:
                add += 1
            if add < 4:
                minn[add:] = [nums[i]] + minn[add:-1]
        
        ret = min(maxn[i] - minn[3 - i] for i in range(4))
        return ret
