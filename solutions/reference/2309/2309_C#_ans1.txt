public class Solution {
    public string GreatestLetter(string s) {
        ISet<char> ht = new HashSet<char>();
        foreach (char c in s) {
            ht.Add(c);
        }
        for (int i = 25; i >= 0; i--) {
            if (ht.Contains((char) ('a' + i)) && ht.Contains((char) ('A' + i))) {
                return ((char) ('A' + i)).ToString();
            }
        }
        return "";
    }
}
