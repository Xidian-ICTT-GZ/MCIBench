char * greatestLetter(char * s) {
    int lower = 0, upper = 0;
    for (int i = 0; s[i] != '\0'; i++) {
        char c = s[i];
        if (islower(c)) {
            lower |= 1 << (c - 'a');
        } else {
            upper |= 1 << (c - 'A');
        }
    }
    for (int i = 25; i >= 0; i--) {
        if (lower & upper & (1 << i)) {
            char *res = (char *)malloc(sizeof(char) * 2);
            res[0] = 'A' + i;
            res[1] = 0;
            return res;
        }
    }
    return "";
}
