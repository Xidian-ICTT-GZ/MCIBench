public class Solution {
    public IList<IList<int>> SubsetsWithDup(int[] nums) {
        Array.Sort(nums);
        IList<IList<int>> ans = new List<IList<int>>();
        int n = nums.Length;

        for (int mask = 0; mask < (1 << n); ++mask) {
            List<int> t = new List<int>();
            bool flag = true;
            for (int i = 0; i < n; ++i) {
                if ((mask & (1 << i)) != 0) {
                    if (i > 0 && (mask & (1 << (i - 1))) == 0 && nums[i] == nums[i - 1]) {
                        flag = false;
                        break;
                    }
                    t.Add(nums[i]);
                }
            }
            if (flag) {
                ans.Add(t);
            }
        }
        return ans;
    }
}
