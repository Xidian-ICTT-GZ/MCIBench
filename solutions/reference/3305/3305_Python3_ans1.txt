class Solution:
    def countOfSubstrings(self, word: str, k: int) -> int:
        vowels = {'a', 'e', 'i', 'o', 'u'}
        n = len(word)
        res = 0
        for i in range(n):
            occur = set()
            consonants = 0
            for j in range(i, n):
                if word[j] in vowels:
                    occur.add(word[j])
                else:
                    consonants += 1
                if len(occur) == 5 and consonants == k:
                    res += 1
        return res
