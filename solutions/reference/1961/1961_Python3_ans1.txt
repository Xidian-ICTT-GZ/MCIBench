class Solution:
    def isPrefixString(self, s: str, words: List[str]) -> bool:
        i = 0
        n = len(s)
        for word in words:
            for ch in word:
                if i < n and ch == s[i]:
                    i += 1
                else:
                    
                    return False
            if i == n:
                
                return True
        
        return False
