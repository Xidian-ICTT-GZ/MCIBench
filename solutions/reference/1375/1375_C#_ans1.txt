public class Solution {
    public int NumTimesAllBlue(int[] flips) {
        int n = flips.Length;
        int ans = 0, right = 0;
        for (int i = 0; i < n; ++i) {
            right = Math.Max(right, flips[i]);
            if (right == i + 1) {
                ++ans;
            }
        }
        return ans;
    }
}
