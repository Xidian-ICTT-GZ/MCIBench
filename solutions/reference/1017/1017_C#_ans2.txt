public class Solution {
    public string BaseNeg2(int n) {
        if (n == 0 || n == 1) {
            return n.ToString();
        }
        StringBuilder sb = new StringBuilder();
        while (n != 0) {
            int remainder = n & 1;
            sb.Append(remainder);
            n -= remainder;
            n /= -2;
        }
        StringBuilder res = new StringBuilder();
        for (int i = sb.Length - 1; i >= 0; i--) {
            res.Append(sb[i]);
        }
        return res.ToString();
    }
}
