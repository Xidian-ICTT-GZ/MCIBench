char * baseNeg2(int n) {
    int val = 0x55555555 ^ (0x55555555 - n);
    if (val == 0) {
        return "0";
    }
    char *res = (char *)calloc(sizeof(char), 32);
    int pos = 0;
    while (val) {
        res[pos++] = '0' + (val & 1);
        val >>= 1;
    }
    for (int l = 0, r = pos - 1; l < r; l++, r--) {
        char c = res[l];
        res[l] = res[r];
        res[r] = c;
    }
    return res;
}
