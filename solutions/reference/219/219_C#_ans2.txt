public class Solution {
    public bool ContainsNearbyDuplicate(int[] nums, int k) {
        ISet<int> set = new HashSet<int>();
        int length = nums.Length;
        for (int i = 0; i < length; i++) {
            if (i > k) {
                set.Remove(nums[i - k - 1]);
            }
            if (!set.Add(nums[i])) {
                return true;
            }
        }
        return false;
    }
}
