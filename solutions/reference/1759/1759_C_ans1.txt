int countHomogenous(char * s) {
    long long res = 0;
    long long mod = 1e9 + 7;
    int prev = s[0];
    int cnt = 0;
    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] == prev) {
            cnt++;
        } else {
            res += (long long)(cnt + 1) * cnt / 2;
            cnt = 1;
            prev = s[i];
        }
    }
    res += (long long)(cnt + 1) * cnt / 2;
    return res % mod;
}
