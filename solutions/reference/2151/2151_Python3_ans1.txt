class Solution:
    def maximumGood(self, statements: List[List[int]]) -> int:
        n = len(statements)
        ans = 0
        for mask in range(1, 1 << n):
            def check() -> bool:
                for i in range(n):
                    for j in range(n):
                        if i == j:
                            continue
                        if statements[i][j] == 0:
                            if mask & (1 << i) and mask & (1 << j):
                                return False
                        elif statements[i][j] == 1:
                            if mask & (1 << i) and not mask & (1 << j):
                                return False
                return True
            
            if check():
                ans = max(ans, bin(mask).count("1"))
        return ans
