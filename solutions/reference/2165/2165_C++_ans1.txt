class Solution {
public:
    long long smallestNumber(long long num) {
        if (num == 0) {
            return 0;
        }
        
        bool negative = (num < 0);
        num = abs(num);
        vector<int> digits;

        long long dummy = num;
        while (dummy) {
            digits.push_back(dummy % 10);
            dummy /= 10;
        }
        sort(digits.begin(), digits.end());
        if (negative) {
            reverse(digits.begin(), digits.end());
        }
        else {
            if (digits[0] == 0) {
                for (int i = 1;; ++i) {
                    if (digits[i] != 0) {
                        swap(digits[0], digits[i]);
                        break;
                    }
                }
            }
        }

        long long ans = 0;
        for (int digit: digits) {
            ans = ans * 10 + digit;
        }
        return negative ? -ans : ans;
    }
};
