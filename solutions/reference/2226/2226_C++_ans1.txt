class Solution {
public:
    int maximumCandies(vector<int>& candies, long long k) {
        
        auto check = [&](int i) -> bool {
            long long res = 0;
            for (int c: candies) {
                res += c / i;
            }
            return res >= k;
        };

        
        int l = 1;
        int r = 1 + *max_element(candies.begin(), candies.end());
        while (l < r) {
            int mid = l + (r - l) / 2;
            if (check(mid)) {
                l = mid + 1;
            } else {
                r = mid;
            }
        }
        return l - 1;
    }
};
