var removeStones = function(stones) {
    const n = stones.length;
    const edge = {};
    for (const [i, [x1, y1]] of stones.entries()) {
        for (const [j, [x2, y2]] of stones.entries()) {
            if (x1 === x2 || y1 === y2) {
                edge[i] ? edge[i].push(j) : edge[i] = [j];
            }
        }
    }

    vis = new Set();
    let num = 0;
    for (let i = 0; i < n; i++) {
        if (!vis.has(i)) {
            num++;
            dfs(i, vis, edge);
        }
    }
    return n - num;
};

const dfs = (x, vis, edge) => {
    vis.add(x);
    for (let y of edge[x]) {
        if (!vis.has(y)) {
            dfs(y, vis, edge);
        }
    }
}
