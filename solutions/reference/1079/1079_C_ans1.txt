int dfs(int *count, const char *tile, int i) {
    if (i == 0) {
        return 1;
    }
    int res = 1;
    for (int j = 0; tile[j] != '\0'; j++) {
        int t = tile[j];
        if (count[t - 'A'] > 0) {
            count[t - 'A']--;
            res += dfs(count, tile, i - 1);
            count[t - 'A']++;
        }
    }
    return res;
}

int numTilePossibilities(char * tiles) {
    int n = strlen(tiles);
    int count[26] = {0};
    char tile[27] = {0};
    for (int i = 0; i < n; i++) {
        char c = tiles[i];
        count[c - 'A']++;
    }
    int pos = 0;
    for (int i = 0; i < 26; i++) {
        if (count[i] > 0) {
            tile[pos++] = 'A' + i;
        }
    }
    tile[pos] = '\0';
    return dfs(count, tile, n) - 1;
}
