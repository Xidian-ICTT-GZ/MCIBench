func largestPathValue(colors string, edges [][]int) int {
    n := len(colors)
    
    g := make([][]int, n)
    
    indeg := make([]int, n)
    for _, edge := range edges {
        indeg[edge[1]]++
        g[edge[0]] = append(g[edge[0]], edge[1])
    }

    
    
    found := 0
    f := make([][26]int, n)
    q := []int{}
    for i := 0; i < n; i++ {
        if indeg[i] == 0 {
            q = append(q, i)
        }
    }

    for len(q) > 0 {
        found++
        u := q[0]
        q = q[1:]
        
        f[u][colors[u]-'a']++
        
        for _, v := range g[u] {
            indeg[v]--
            
            for c := 0; c < 26; c++ {
                f[v][c] = max(f[v][c], f[u][c])
            }
            if indeg[v] == 0 {
                q = append(q, v)
            }
        }
    }
    if found != n {
        return -1
    }
    ans := 0
    for i := 0; i < n; i++ {
        for c := 0; c < 26; c++ {
            ans = max(ans, f[i][c])
        }
    }
    return ans
}
