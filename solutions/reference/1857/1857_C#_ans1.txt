public class Solution {
    public int LargestPathValue(string colors, int[][] edges) {
        int n = colors.Length;
        
        List<List<int>> g = new List<List<int>>();
        for (int i = 0; i < n; ++i) {
            g.Add(new List<int>());
        }
        
        int[] indeg = new int[n];
        foreach (var edge in edges) {
            ++indeg[edge[1]];
            g[edge[0]].Add(edge[1]);
        }
        
        
        int found = 0;
        int[][] f = new int[n][];
        for (int i = 0; i < n; ++i) {
            f[i] = new int[26];
        }
        Queue<int> q = new Queue<int>();
        for (int i = 0; i < n; ++i) {
            if (indeg[i] == 0) {
                q.Enqueue(i);
            }
        }
        while (q.Count > 0) {
            ++found;
            int u = q.Dequeue();
            
            ++f[u][colors[u] - 'a'];
            
            foreach (int v in g[u]) {
                --indeg[v];
                
                for (int c = 0; c < 26; ++c) {
                    f[v][c] = Math.Max(f[v][c], f[u][c]);
                }
                if (indeg[v] == 0) {
                    q.Enqueue(v);
                }
            }
        }
        if (found != n) {
            return -1;
        }
        int ans = 0;
        for (int i = 0; i < n; ++i) {
            ans = Math.Max(ans, f[i].Max());
        }
        return ans;
    }
}
