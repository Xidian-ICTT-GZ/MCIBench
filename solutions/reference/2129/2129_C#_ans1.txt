public class Solution {
    public string CapitalizeTitle(string title) {
        StringBuilder sb = new StringBuilder(title);
        int n = title.Length;
        int l = 0, r = 0;   
        while (r < n) {
            while (r < n && sb[r] != ' ') {
                ++r;
            }
            
            if (r - l > 2) {
                sb[l] = char.ToUpper(sb[l]);
                ++l;
            }
            while (l < r) {
                sb[l] = char.ToLower(sb[l]);
                ++l;
            }
            l = r + 1;
            ++r;
        }
        return sb.ToString();
    }
}
