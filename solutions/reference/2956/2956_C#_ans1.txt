public class Solution {
    public int[] FindIntersectionValues(int[] nums1, int[] nums2) {
        int[] res = new int[2];
        foreach (int x1 in nums1) {
            foreach (int x2 in nums2) {
                if (x1 == x2) {
                    res[0]++;
                    break;
                }
            }
        }
        foreach (int x2 in nums2) {
            foreach (int x1 in nums1) {
                if (x2 == x1) {
                    res[1]++;
                    break;
                }
            }
        }
        return res;
    }
}
