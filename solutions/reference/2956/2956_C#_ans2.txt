public class Solution {
    public int[] FindIntersectionValues(int[] nums1, int[] nums2) {
        HashSet<int> s1 = new HashSet<int>();
        foreach (int x in nums1) {
            s1.Add(x);
        }
        HashSet<int> s2 = new HashSet<int>();
        foreach (int x in nums2) {
            s2.Add(x);
        }

        int[] res = new int[2];
        foreach (int x in nums1) {
            if (s2.Contains(x)) {
                res[0]++;
            }
        }
        foreach (int x in nums2) {
            if (s1.Contains(x)) {
                res[1]++;
            }
        }
        return res;
    }
}
