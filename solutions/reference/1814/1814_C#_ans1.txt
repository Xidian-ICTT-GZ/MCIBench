public class Solution {
    public int CountNicePairs(int[] nums) {
        const int MOD = 1000000007;
        int res = 0;
        IDictionary<int, int> h = new Dictionary<int, int>();
        foreach (int i in nums) {
            int temp = i, j = 0;
            while (temp > 0) {
                j = j * 10 + temp % 10;
                temp /= 10;
            }
            h.TryAdd(i - j, 0);
            res = (res + h[i - j]) % MOD;
            h[i - j]++;
        }
        return res;
    }
}
