func removeAnagrams(words []string) []string {
    res := []string{words[0]}   
    n := len(words)
    
    
    compare := func(word1, word2 string) bool {
        freq := make([]int, 26)
        for _, ch := range word1 {
            freq[ch-'a']++
        }
        for _, ch := range word2 {
            freq[ch-'a']--
        }
        for _, x := range freq {
            if x != 0 {
                return false
            }
        }
        return true
    }

    for i := 1; i < n; i++ {
        if !compare(words[i], words[i - 1]) {
            res = append(res, words[i])
        }
    }
    return res
}
