var findKDistantIndices = function(nums, key, k) {
    const res = [];
    const n = nums.length;
    
    for (let i = 0; i < n; ++i) {
        for (let j = 0; j < n; ++j) {
            if (nums[j] === key && Math.abs(i - j) <= k) {
                res.push(i);
                break; 
            }
        }
    }
    return res;
};
