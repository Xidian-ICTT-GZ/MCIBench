public class Solution {
    public int LastStoneWeightII(int[] stones) {
        int sum = 0;
        foreach (int weight in stones) {
            sum += weight;
        }
        int m = sum / 2;
        bool[] dp = new bool[m + 1];
        dp[0] = true;
        foreach (int weight in stones) {
            for (int j = m; j >= weight; --j) {
                dp[j] = dp[j] || dp[j - weight];
            }
        }
        for (int j = m;; --j) {
            if (dp[j]) {
                return sum - 2 * j;
            }
        }
    }
}
