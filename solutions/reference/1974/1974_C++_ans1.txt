class Solution {
public:
    int minTimeToType(string word) {
        int res = 0;
        int prev = 0;   
        for (char ch : word){
            
            int curr = ch - 'a';
            res += 1 + min(abs(curr - prev), 26 - abs(curr - prev));
            prev = curr;
        }
        return res;
    }
};
