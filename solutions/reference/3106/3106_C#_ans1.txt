public class Solution {
    public string GetSmallestString(string s, int k) {
        char[] ans = s.ToCharArray();
        for (int i = 0; i < s.Length; ++i) {
            int dis = Math.Min(s[i] - 'a', 'z' - s[i] + 1);
            if (dis <= k) {
                ans[i] = 'a';
                k -= dis;
            } else {
                ans[i] = (char) (ans[i] - k);
                break;
            }
        }
        return new string(ans);
    }
}
