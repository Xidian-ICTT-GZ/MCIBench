var knightDialer = function(n) {
    const mod = 1_000_000_007;
    const moves = [
        [4, 6],
        [6, 8],
        [7, 9],
        [4, 8],
        [3, 9, 0],
        [],
        [1, 7, 0],
        [2, 6],
        [1, 3],
        [2, 4],
    ];
    const d = Array.from({ length: 2 }, () => Array(10).fill(0));
    d[1].fill(1);
    for (let i = 2; i <= n; i++) {
        const x = i & 1;
        for (let j = 0; j < 10; j++) {
            d[x][j] = 0;
            for (const k of moves[j]) {
                d[x][j] = (d[x][j] + d[x ^ 1][k]) % mod;
            }
        }
    }
    return d[n % 2].reduce((res, x) => (res + x) % mod, 0);
};
