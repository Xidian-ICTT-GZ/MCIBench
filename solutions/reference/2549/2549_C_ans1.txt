int distinctIntegers(int n) {
    int *nums = malloc(sizeof(int) * (n + 1));
    memset(nums, 0, sizeof(int) * (n + 1));
    nums[n] = 1;
    for (int k = 0; k < n; k++) {
        for (int x = 1; x <= n; x++) {
            if (nums[x] == 0) {
                continue;
            }
            for (int i = 1; i <= n; i++) {
                if (x % i == 1) {
                    nums[i] = 1;
                }
            }
        }
    }
    int res = 0;
    for (int i = 1; i <= n; i++) {
        res += nums[i];
    }
    free(nums);
    return res;
}
