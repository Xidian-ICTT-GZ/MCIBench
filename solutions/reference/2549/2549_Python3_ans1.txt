class Solution:
    def distinctIntegers(self, n: int) -> int:
        nums = [0] * (n + 1)
        nums[n] = 1
        for _ in range(0, n):
            for x in range(1, n + 1):
                if nums[x] == 0:
                    continue
                for i in range(1, n + 1):
                    if x % i == 1:
                        nums[i] = 1
        return sum(nums)
