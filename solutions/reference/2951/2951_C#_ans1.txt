public class Solution {
    public IList<int> FindPeaks(int[] mountain) {
        var res = new List<int>();
        for (int i = 1; i < mountain.Count() - 1; i++) {
            if (mountain[i - 1] < mountain[i] && mountain[i] > mountain[i + 1]) {
                res.Add(i);
            }
        }
        return res;
    }
}
