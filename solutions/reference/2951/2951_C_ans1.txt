int* findPeaks(int* mountain, int mountainSize, int* returnSize) {
    *returnSize = 0;
    int* res = (int*)malloc(mountainSize * sizeof(int));
    for (int i = 1; i < mountainSize - 1; i++) {
        if (mountain[i - 1] < mountain[i] && mountain[i] > mountain[i + 1]) {
            res[*returnSize] = i;
            (*returnSize)++;
        }
    }
    return res;
}
