struct Node {
    int id, mask, dist;
};

int shortestPathLength(int** graph, int graphSize, int* graphColSize){
    int n = graphSize;
    int d[n][n];
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            d[i][j] = n + 1;
        }
        for (int j = 0; j < graphColSize[i]; j++) {
            d[i][graph[i][j]] = 1;
        }
    }
    
    for (int k = 0; k < n; ++k) {
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < n; ++j) {
                d[i][j] = fmin(d[i][j], d[i][k] + d[k][j]);
            }
        }
    }

    int f[n][1 << n];
    memset(f, 0x3f, sizeof(f));
    for (int mask = 1; mask < (1 << n); ++mask) {
        
        if ((mask & (mask - 1)) == 0) {
            int u = __builtin_ctz(mask);
            f[u][mask] = 0;
        }
        else {
            for (int u = 0; u < n; ++u) {
                if (mask & (1 << u)) {
                    for (int v = 0; v < n; ++v) {
                        if ((mask & (1 << v)) && u != v) {
                            f[u][mask] = fmin(f[u][mask], f[v][mask ^ (1 << u)] + d[v][u]);
                        }
                    }
                }
            }
        }
    }

    int ans = INT_MAX;
    for (int u = 0; u < n; ++u) {
        ans = fmin(ans, f[u][(1 << n) - 1]);
    }
    return ans;
}
