public class Solution {
    public int CountPairs(IList<int> nums, int target) {
        ((List<int>) nums).Sort();
        int res = 0;
        for (int i = 0, j = nums.Count - 1; i < j; i++) {
            while (i < j && nums[i] + nums[j] >= target) {
                j--;
            }
            res += j - i;
        }
        return res;
    }
}
