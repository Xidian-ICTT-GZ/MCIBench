public class Solution {
    public int DifferenceOfSum(int[] nums) {
        int elementSum = 0, digitSum = 0;
        for (int i = 0; i < nums.Length; i++) {
            int num = nums[i];
            elementSum += num;
            while (num > 0) {
                digitSum += num % 10;
                num /= 10;
            }
        }
        return elementSum - digitSum;
    }
}
