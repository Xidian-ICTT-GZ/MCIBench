int divisorSubstrings(int num, int k) {
    char s[12]; 
    sprintf(s, "%d", num);   
    int n = strlen(s);
    int res = 0;
    for (int i = 0; i <= n - k; ++i) {
        
        char tmpStr[k + 1];
        strncpy(tmpStr, s + i, k);
        tmpStr[k] = '\0';
        int tmp = atoi(tmpStr);
        if (tmp != 0 && num % tmp == 0) {
            ++res;
        }
    }
    return res;
}
