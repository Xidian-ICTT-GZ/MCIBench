var divisorSubstrings = function(num, k) {
    let s = num.toString();   
    let n = s.length;
    let res = 0;
    for (let i = 0; i <= n - k; ++i) {
        
        let tmp = parseInt(s.substring(i, i + k));
        if (tmp !== 0 && num % tmp === 0) {
            ++res;
        }
    }
    return res;
};
