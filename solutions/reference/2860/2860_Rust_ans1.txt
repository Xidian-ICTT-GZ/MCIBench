impl Solution {
    pub fn count_ways(mut nums: Vec<i32>) -> i32 {
        nums.sort();
        let n = nums.len();
        let mut res = 0;
        for k in 0..= n {
            
            if k > 0 && nums[k - 1] >= k as i32 {
                continue;
            }
            
            if k < n && nums[k] <= k as i32 {
                continue;
            }
            res += 1;
        }
        res
    }
}
