public class Solution {
    public int MaximumBeauty(int[] nums, int k) {
        int m = 0;
        foreach (int x in nums) {
            m = Math.Max(m, x);
        }
        int[] diff = new int[m + 2];
        foreach (int x in nums) {
            diff[Math.Max(x - k, 0)]++;
            diff[Math.Min(x + k + 1, m + 1)]--;
        }
        int res = 0, count = 0;
        foreach (int x in diff) {
            count += x;
            res = Math.Max(res, count);
        }
        return res;
    }
}
