int cmp(const void *x1, const void *x2) {
    return *(const int *)x1 - *(const int *)x2;
}

int maximumBeauty(int* nums, int numsSize, int k){
    int res = 0;
    qsort(nums, numsSize, sizeof(int), cmp);
    for (int i = 0, j = 0; i < numsSize; i++) {
        while (nums[i] - 2 * k > nums[j]) {
            j++;
        }
        res = fmax(res, i - j + 1);
    }
    return res;
}
