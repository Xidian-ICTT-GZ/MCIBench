int maximumBeauty(int* nums, int numsSize, int k){
    int m = 0;
    for (int i = 0; i < numsSize; i++) {
        m = fmax(m, nums[i]);
    }
    int diff[m + 2];
    memset(diff, 0, sizeof(diff));
    for (int i = 0; i < numsSize; i++) {
        diff[(int)fmax(nums[i] - k, 0)]++;
        diff[(int)fmin(nums[i] + k + 1, m + 1)]--;
    }
    int res = 0, count = 0;
    for (int i = 0; i < m + 2; i++) {
        count += diff[i];
        res = fmax(res, count);
    }
    return res;
}
