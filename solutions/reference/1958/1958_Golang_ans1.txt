func checkMove(board [][]byte, rMove int, cMove int, color byte) bool {
    
    check := func(dx, dy int) bool {
        x, y := rMove+dx, cMove+dy
        step := 1   
        for x >= 0 && x < 8 && y >= 0 && y < 8 {
            if step == 1 {
                
                if board[x][y] == '.' || board[x][y] == color {
                    return false
                }
            } else {
                
                if board[x][y] == '.' {
                    return false
                }
                
                if board[x][y] == color {
                    return true
                }
            }
            step++
            x += dx
            y += dy
        }
        
        return false
    }

    
    dx := []int{1, 1, 0, -1, -1, -1, 0, 1}   
    dy := []int{0, 1, 1, 1, 0, -1, -1, -1}   
    for k := 0; k < 8; k++ {
        if check(dx[k], dy[k]) {
            return true
        }
    }
    return false
}
