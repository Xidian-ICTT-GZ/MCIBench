impl Solution {
    pub fn check_move(board: Vec<Vec<char>>, r_move: i32, c_move: i32, color: char) -> bool {
         
        fn check(board: &Vec<Vec<char>>, r_move: i32, c_move: i32, dx: i32, dy: i32, color: char) -> bool {
            let mut x = (r_move + dx) as usize;
            let mut y = (c_move + dy) as usize;
            let mut step = 1;   
            while x >= 0 && x < 8 && y >= 0 && y < 8 {
                if step == 1 {
                    
                    if board[x][y] == '.' || board[x][y] == color {
                        return false;
                    }
                } else {
                    
                    if board[x][y] == '.' {
                        return false;
                    }
                    
                    if board[x][y] == color {
                        return true;
                    }
                }
                step += 1;
                x += dx as usize;
                y += dy as usize;
            }
            
            false
        }

        
        let dx = [1, 1, 0, -1, -1, -1, 0, 1];   
        let dy = [0, 1, 1, 1, 0, -1, -1, -1];   
        for (&d_x, &d_y) in dx.iter().zip(dy.iter()) {
            if check(&board, r_move, c_move, d_x, d_y, color) {
                return true;
            }
        }
        false
    }
}
