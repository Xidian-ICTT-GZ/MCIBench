var checkMove = function(board, rMove, cMove, color) {
    
    const check = (dx, dy) => {
        let x = rMove + dx;
        let y = cMove + dy;
        let step = 1;   
        while (x >= 0 && x < 8 && y >= 0 && y < 8) {
            if (step === 1) {
                
                if (board[x][y] === '.' || board[x][y] === color) {
                    return false;
                }
            } else {
                
                if (board[x][y] === '.') {
                    return false;
                }
                
                if (board[x][y] === color) {
                    return true;
                }
            }
            ++step;
            x += dx;
            y += dy;
        }
        
        return false;
    };

    
    const dx = [1, 1, 0, -1, -1, -1, 0, 1];   
    const dy = [0, 1, 1, 1, 0, -1, -1, -1];   
    for (let k = 0; k < 8; ++k) {
        if (check(dx[k], dy[k])) {
            return true;
        }
    }
    return false;
};
