impl Solution {
    pub fn get_winner(arr: Vec<i32>, k: i32) -> i32 {
        let mut prev = arr[0].max(arr[1]);
        if k == 1 {
            return prev;
        }
        let mut consecutive = 1;
        let mut max_num = prev;
        for &num in &arr[2..] {
            let curr = num;
            if prev > curr {
                consecutive += 1;
                if consecutive == k {
                    return prev;
                }
            } else {
                prev = curr;
                consecutive = 1;
            }
            max_num = max_num.max(curr);
        }
        max_num
    }
}
