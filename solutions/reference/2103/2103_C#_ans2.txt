public class Solution {
    const int POLE_NUM = 10;

    public int CountPoints(string rings) {
        int[] state = new int[POLE_NUM];
        int n = rings.Length;
        for (int i = 0; i < n; i += 2) {
            char color = rings[i];
            int poleIndex = rings[i + 1] - '0';
            if (color == 'R') {
                state[poleIndex] |= 1;
            } else if (color == 'G') {
                state[poleIndex] |= 2;
            } else {
                state[poleIndex] |= 4;
            }
        }
        int res = 0;
        for (int i = 0; i < POLE_NUM; i++) {
            res += state[i] == 7 ? 1 : 0;
        }
        return res;
    }
}
