bool canAliceWin(int* nums, int numsSize) {
    int single_digit_sum = 0;
    int double_digit_sum = 0;
    for (int i = 0; i < numsSize; i++) {
        if (nums[i] < 10) {
            single_digit_sum += nums[i];
        } else {
            double_digit_sum += nums[i];
        }
    }
    return single_digit_sum != double_digit_sum;
}
