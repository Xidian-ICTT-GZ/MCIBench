public class Solution {
    public string TruncateSentence(string s, int k) {
        int n = s.Length;
        int end = 0, count = 0;
        for (int i = 1; i <= n; i++) {
            if (i == n || s[i] == ' ') {
                count++;
                if (count == k) {
                    end = i;
                    break;
                }
            }
        }
        return s.Substring(0, end);
    }
}
