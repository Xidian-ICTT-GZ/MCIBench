class Solution:
    def truncateSentence(self, s: str, k: int) -> str:
        n, end, count = len(s), 0, 0
        for i in range(1, n + 1):
            if i == n or s[i] == ' ':
                count += 1
                if count == k:
                    end = i
                    break
        return s[:end]
