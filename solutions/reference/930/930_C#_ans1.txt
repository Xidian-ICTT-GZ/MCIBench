public class Solution {
    public int NumSubarraysWithSum(int[] nums, int goal) {
        int sum = 0;
        Dictionary<int, int> cnt = new Dictionary<int, int>();
        int ret = 0;
        foreach (int num in nums) {
            if (cnt.ContainsKey(sum)) {
                cnt[sum]++;
            } else {
                cnt.Add(sum, 1);
            }
            sum += num;
            int val = 0;
            cnt.TryGetValue(sum - goal, out val);
            ret += val;
        }
        return ret;
    }
}
