impl Solution {
    pub fn final_string(s: String) -> String {
        let mut q = Vec::<char>::new();
        let mut head = false;
        for ch in s.chars() {
            if ch != 'i' {
                if head {
                    q.insert(0, ch);
                } else {
                    q.push(ch);
                }
            } else {
                head = !head;
            }
        }
        let ans: String = if head { q.iter().rev().collect() } else { q.iter().collect() };
        ans
    }
}
