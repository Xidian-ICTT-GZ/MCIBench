var finalString = function(s) {
    let q = [];
    let head = false;
    for (let ch of s) {
        if (ch !== 'i') {
            if (head) {
                q.unshift(ch);
            } else {
                q.push(ch);
            }
        } else {
            head = !head;
        }
    }
    let ans = head ? q.reverse().join('') : q.join('');
    return ans;
};
