var minKBitFlips = function(nums, k) {
    const n = nums.length;
    const diff = new Array(n + 1).fill(0);
    let ans = 0, revCnt = 0;
    for (let i = 0; i < n; i++) {
        revCnt += diff[i];
        if ((nums[i] + revCnt) % 2 === 0) {
            if ((i + k) > n) {
                return -1;
            }
            ++ans;
            ++revCnt;
            --diff[i + k]
        }
    }
    return ans;
};
