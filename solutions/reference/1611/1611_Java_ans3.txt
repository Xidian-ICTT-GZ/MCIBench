class Solution {
    public int minimumOneBitOperations(int n) {
        if (n == 0) {
            return 0;
        }
        int x = Integer.toBinaryString(n).length() - 1;
        
        int ans = 1 << x;
        
        int bit = 1;
        for (int i = x - 1; i >= 0; --i) {
            
            if ((n & (1 << i)) != 0) {
                bit ^= 1;
            }
            
            if (bit != 0) {
                ans |= (1 << i);
            }
        }
        return ans;
    }
}
