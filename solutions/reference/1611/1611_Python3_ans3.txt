class Solution:
    def minimumOneBitOperations(self, n: int) -> int:
        if n == 0:
            return 0
        
        x = len(bin(n)) - 3
        
        ans = 1 << x
        
        bit = 1
        for i in range(x - 1, -1, -1):
            
            if n & (1 << i):
                bit ^= 1
            
            if bit:
                ans |= (1 << i)
        
        return ans
