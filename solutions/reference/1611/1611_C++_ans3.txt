class Solution {
public:
    int minimumOneBitOperations(int n) {
        if (!n) {
            return 0;
        }
        int x = 31 - __builtin_clz(n);
        
        int ans = 1 << x;
        
        int bit = 1;
        for (int i = x - 1; i >= 0; --i) {
            
            if (n & (1 << i)) {
                bit ^= 1;
            }
            
            if (bit) {
                ans |= (1 << i);
            }
        }
        return ans;
    }
};
