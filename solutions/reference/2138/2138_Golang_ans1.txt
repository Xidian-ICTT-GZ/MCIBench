func divideString(s string, k int, fill byte) []string {
	var res []string 
	n := len(s)
	curr := 0 
	
	for curr < n {
		end := curr + k
		if end > n {
			end = n
		}
		res = append(res, s[curr:end])
		curr += k
	}
	
	last := res[len(res) - 1]
	if len(last) < k {
		last += strings.Repeat(string(fill), k - len(last))
		res[len(res) - 1] = last
	}
	return res
}
