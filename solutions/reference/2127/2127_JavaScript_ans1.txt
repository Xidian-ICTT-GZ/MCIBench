var maximumInvitations = function(favorite) {
    const n = favorite.length;
    
    const indeg = new Array(n).fill(0);
    for (x of favorite) {
        indeg[x]++;
    }

    const used = new Array(n).fill(false)
    const f = new Array(n).fill(1)
    const q = []
    for (let i = 0; i < n; i++) {
        if (!indeg[i]) {
            q.push(i);
        }
    }
    while (q.length) {
        let u = q.pop();
        used[u] = true;
        let v = favorite[u];
        
        f[v] = Math.max(f[v], f[u] + 1);
        indeg[v]--;
        if (!indeg[v]) {
            q.unshift(v);
        }
    }
    
    
    let ring = 0, total = 0;
    for (let i = 0; i < n; i++) {
        if (!used[i]) {
            let j = favorite[i];
            
            if (favorite[j] == i) {
                total += f[i] + f[j]
                used[i] = used[j] = true;
            } else {
                
                let u = i, cnt = 0;
                while (true) {
                    cnt++;
                    u = favorite[u];
                    used[u] = true;
                    if (u == i) {
                        break;
                    }
                }
                ring = Math.max(ring, cnt);
            }
        }
    }
    return Math.max(ring, total);
};
