class Solution:
    def nearestValidPoint(self, x: int, y: int, points: List[List[int]]) -> int:
        n = len(points)
        best, bestid = float("inf"), -1
        for i, (px, py) in enumerate(points):
            if x == px:
                if (dist := abs(y - py)) < best:
                    best = dist
                    bestid = i
            elif y == py:
                if (dist := abs(x - px)) < best:
                    best = dist
                    bestid = i
        
        return bestid
