public class Solution {
    public long MaximumSubsequenceCount(string text, string pattern) {
        long res = 0, cnt1 = 0, cnt2 = 0;
        foreach (char c in text) {
            if (c == pattern[1]) {
                res += cnt1;
                cnt2++;
            }
            if (c == pattern[0]) {
                cnt1++;
            }
        }
        return res + Math.Max(cnt1, cnt2);
    }
}
