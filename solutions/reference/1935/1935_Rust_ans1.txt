use std::collections::HashSet;

impl Solution {
    pub fn can_be_typed_words(text: String, broken_letters: String) -> i32 {
        let broken: HashSet<char> = broken_letters.chars().collect();   
        let mut res = 0;   
        let mut flag = true;   
        for ch in text.chars() {
            if ch == ' ' {
                
                if flag {
                    res += 1;
                }
                flag = true;
            } else if broken.contains(&ch) {
                
                flag = false;
            }
        }
        
        if flag {
            res += 1;
        }
        res
    }
}
