public class Solution {
    public int CanBeTypedWords(string text, string brokenLetters) {
        HashSet<char> broken = new HashSet<char>();   
        foreach (char ch in brokenLetters) {
            broken.Add(ch);
        }
        int res = 0;   
        bool flag = true;   
        foreach (char ch in text) {
            if (ch == ' ') {
                
                if (flag) {
                    ++res;
                }
                flag = true;
            } else if (broken.Contains(ch)) {
                
                flag = false;
            }
        }
        
        if (flag) {
            ++res;
        }
        return res;
    }
}
