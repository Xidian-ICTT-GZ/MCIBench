int numberOfAlternatingGroups(int* colors, int colorsSize, int k) {
    int res = 0, cnt = 1;
    for (int i = -k + 2; i < colorsSize; i++) {
        if (colors[(i + colorsSize) % colorsSize] != colors[(i - 1 + colorsSize) % colorsSize]) {
            cnt += 1;
        } else {
            cnt = 1;
        }
        if (cnt >= k) {
            res += 1;
        }
    }
    return res;
}
