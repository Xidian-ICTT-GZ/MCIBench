public class Solution {
    public string MaximumOddBinaryNumber(string s) {
        int cnt = 0;
        foreach (char c in s) {
            cnt += c - '0';
        }
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < cnt - 1; i++) {
            sb.Append('1');
        }
        for (int i = 0; i < s.Length - cnt; i++) {
            sb.Append('0');
        }
        sb.Append('1');
        return sb.ToString();
    }
}
