func fullBloomFlowers(flowers [][]int, people []int) []int {
    n := len(flowers)
    starts := make([]int, n)
    ends := make([]int, n)
    for i, flower := range flowers {
        starts[i] = flower[0]
        ends[i] = flower[1]
    }
    sort.Ints(starts)
    sort.Ints(ends)
    m := len(people)
    ans := make([]int, m)
    for i, p := range people {
        x := sort.SearchInts(starts, p + 1)
        y := sort.SearchInts(ends, p)
        ans[i] = x - y
    }
    return  ans
}
