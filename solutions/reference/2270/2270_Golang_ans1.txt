func waysToSplitArray(nums []int) int {
    n := len(nums)
    left, right := int64(0), int64(0)
    for _, num := range nums {
        right += int64(num)
    }
    ans := 0
    for i := 0; i < n-1; i++ {
        left += int64(nums[i])
        right -= int64(nums[i])
        if left >= right {
            ans++
        }
    }
    return ans
}
