public class Solution {
    public int WaysToSplitArray(int[] nums) {
        int n = nums.Length;
        long left = 0, right = 0;
        foreach (int num in nums) {
            right += num;
        }
        int ans = 0;
        for (int i = 0; i < n - 1; i++) {
            left += nums[i];
            right -= nums[i];
            if (left >= right) {
                ans++;
            }
        }
        return ans;
    }
}
