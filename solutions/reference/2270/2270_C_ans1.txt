int waysToSplitArray(int* nums, int numsSize) {
    int n = numsSize;
    long long left = 0, right = 0;
    for (int i = 0; i < n; ++i) {
        right += nums[i];
    }
    int ans = 0;
    for (int i = 0; i < n - 1; ++i) {
        left += nums[i];
        right -= nums[i];
        if (left >= right) {
            ++ans;
        }
    }
    return ans;
}
