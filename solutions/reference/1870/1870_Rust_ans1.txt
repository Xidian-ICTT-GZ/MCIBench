impl Solution {
    pub fn min_speed_on_time(dist: Vec<i32>, hour: f64) -> i32 {
        let n = dist.len();
        
        let hr = (hour * 100.0).round() as i64;
        
        if hr <= ((n as i64 - 1) * 100) {
            return -1;
        }
        
        let mut l = 1;
        let mut r = 10_000_000;
        while l < r {
            let mid = l + (r - l) / 2;
            
            let mut t = 0i64;
            
            for i in 0..n - 1 {
                t += (dist[i] as i64 - 1) / mid as i64 + 1;
            }
            
            t *= mid as i64;
            t += dist[n - 1] as i64;
            if t * 100 <= hr * mid as i64 { 
                r = mid;
            } else {
                l = mid + 1;
            }
        }
        l 
    }
}
