var minSpeedOnTime = function(dist, hour) {
    const n = dist.length;
    
    const hr = Math.round(hour * 100);
    
    if (hr <= (n - 1) * 100) {
        return -1;
    }
    
    let l = 1;
    let r = 10000000;
    while (l < r) {
        const mid = l + Math.floor((r - l) / 2);
        
        let t = 0;
        
        for (let i = 0; i < n - 1; ++i) {
            t += Math.floor((dist[i] - 1) / mid) + 1;
        }
        
        t *= mid;
        t += dist[n - 1];
        if (t * 100 <= hr * mid) { 
            r = mid;
        } else {
            l = mid + 1;
        }
    }
    return l; 
};
