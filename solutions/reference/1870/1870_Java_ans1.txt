class Solution {
    public int minSpeedOnTime(int[] dist, double hour) {
        int n = dist.length;
        
        long hr = Math.round(hour * 100);
        
        if (hr <= (n - 1) * 100) {
            return -1;
        }
        
        int l = 1;
        int r = 10000000;
        while (l < r) {
            int mid = l + (r - l) / 2;
            
            long t = 0;
            
            for (int i = 0; i < n - 1; ++i) {
                t += (dist[i] - 1) / mid + 1;
            }
            
            t *= mid;
            t += dist[n - 1];
            if (t * 100 <= hr * mid) { 
                r = mid;
            } else {
                l = mid + 1;
            }
        }
        return l; 
    }
}
