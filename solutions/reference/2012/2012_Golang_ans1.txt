func sumOfBeauties(nums []int) int {
    n := len(nums)
    state := make([]int, n)
    pre_max := nums[0]
    for i := 1; i < n - 1; i++ {
        if nums[i] > pre_max {
            state[i] = 1
            pre_max = nums[i]
        }
    }
    suf_min := nums[n-1]
    res := 0
    for i := n - 2; i > 0; i-- {
        if state[i] == 1 && nums[i] < suf_min {
            res += 2
        } else if nums[i - 1] < nums[i] && nums[i] < nums[i + 1] {
            res += 1
        }
        suf_min = min(suf_min, nums[i])
    }
    return res
}
