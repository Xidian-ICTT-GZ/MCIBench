func min(a, b int) int {
    if a < b {
        return a
    }
    return b
}

func minTrioDegree(n int, edges [][]int) int {
    g := make([][]int, n)
    degree := make([]int, n)
    for i := 0; i < n; i++ {
        g[i] = make([]int, n)
    }
    for _, edge := range edges {
        x, y := edge[0] - 1, edge[1] - 1
        g[x][y], g[y][x] = 1, 1
        degree[x]++
        degree[y]++
    }
    ans := 0x3f3f3f3f
    for i := 0; i < n; i++ {
        for j := i + 1; j < n; j++ {
            if g[i][j] != 1 {
                continue
            }
            for k := j + 1; k < n; k++ {
                if g[i][k] == 1 && g[j][k] == 1 {
                    ans = min(ans, degree[i] + degree[j] + degree[k] - 6)
                }
            }
        }
    }
    if ans == 0x3f3f3f3f {
        return -1
    }
    return ans
}
