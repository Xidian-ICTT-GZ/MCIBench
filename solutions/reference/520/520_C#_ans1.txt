public class Solution {
    public bool DetectCapitalUse(string word) {
        
        if (word.Length >= 2 && char.IsLower(word[0]) && char.IsUpper(word[1])) {
            return false;
        }
        
        
        for (int i = 2; i < word.Length; ++i) {
            if (char.IsLower(word[i]) ^ char.IsLower(word[1])) {
                return false;
            }
        }
        return true;
    }
}
