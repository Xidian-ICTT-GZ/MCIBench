class Solution:
    def numberOfPoints(self, nums: List[List[int]]) -> int:
        C = max(y for _, y in nums)
        count = [0] * (C + 1)
        for x, y in nums:
            for i in range(x, y + 1):
                count[i] += 1
        
        ans = sum(1 for i in range(1, C + 1) if count[i] > 0)
        return ans
