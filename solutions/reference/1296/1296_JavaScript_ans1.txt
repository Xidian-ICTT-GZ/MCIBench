var isPossibleDivide = function(nums, k) {
    const n = nums.length;
    if (n % k !== 0) {
        return false;
    }
    nums.sort((a, b) => a - b);
    const cnt = new Map();
    for (const x of nums) {
        cnt.set(x, (cnt.get(x) || 0) + 1);
    }
    for (const x of nums) {
        if (!cnt.has(x)) {
            continue;
        }
        for (let j = 0; j < k; j++) {
            const num = x + j;
            if (!cnt.has(num)) {
                return false;
            }
            cnt.set(num, cnt.get(num) - 1);
            if (cnt.get(num) == 0) {
                cnt.delete(num);
            }
        }
    }
    return true;
};
