var findOriginalArray = function(changed) {
    changed.sort((a, b) => a - b);
    const count = {};
    for (const num of changed) {
        count[num] = (count[num] || 0) + 1;
    }
    const res = [];
    for (const a of changed) {
        if (count[a] === 0) {
            continue;
        }
        count[a]--;
        if (!count[a * 2]) {
            return [];
        }
        count[a * 2]--;
        res.push(a);
    }
    return res;
};
