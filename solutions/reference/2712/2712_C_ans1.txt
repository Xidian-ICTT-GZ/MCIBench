long long minimumCost(char* s) {
    int n = strlen(s);
    long long suf[n + 1][2];
    memset(suf, 0, sizeof(suf));
    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == '1') {
            suf[i][1] = suf[i + 1][1];
            suf[i][0] = suf[i + 1][1] + (n - i);
        } else {
            suf[i][1] = suf[i + 1][0] + (n - i);
            suf[i][0] = suf[i + 1][0];
        }
    }

    long long pre[2] = {0, 0};
    long long res = LLONG_MAX;
    for (int i = 0; i < n; i++) {
        if (s[i] == '1') {
            pre[0] = pre[1] + i + 1;
        } else {
            pre[1] = pre[0] + i + 1;
        }
        res = fmin(res, fmin(pre[0] + suf[i + 1][0], pre[1] + suf[i + 1][1]));
    }
    return res;
}
