impl Solution {
    pub fn minimum_cost(s: String) -> i64 {
        s.chars()
            .collect::<Vec<_>>()
            .windows(2)
            .enumerate()
            .filter(|&(_, w)| w[0] != w[1])
            .map(|(i, _)| (i + 1).min(s.len() - (i + 1)) as i64)
            .sum()
    }
}
