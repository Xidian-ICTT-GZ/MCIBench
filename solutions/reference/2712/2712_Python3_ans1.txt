class Solution:
    def minimumCost(self, s: str) -> int:
        n = len(s)
        suf = [[0] * 2 for _ in range(n + 1)]
        for i in range(n - 1, -1, -1):
            if s[i] == '1':
                suf[i][1] = suf[i + 1][1]
                suf[i][0] = suf[i + 1][1] + (n - i)
            else:
                suf[i][1] = suf[i + 1][0] + (n - i)
                suf[i][0] = suf[i + 1][0]
        
        pre = [0] * 2
        res = float('inf')
        for i in range(n):
            if s[i] == '1':
                pre[0] = pre[1] + i + 1
            else:
                pre[1] = pre[0] + i + 1

            res = min([res, pre[0] + suf[i + 1][0], pre[1] + suf[i + 1][1]])
        return res
