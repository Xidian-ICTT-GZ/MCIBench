public class Solution {
    public int CountOfPairs(int[] nums) {
        int n = nums.Length;
        int[,] dp = new int[n, 51];
        int mod = 1000000007;
        for (int v = 0; v <= nums[0]; v++) {
            dp[0, v] = 1;
        }

        for (int i = 1; i < n; i++) {
            for (int v2 = 0; v2 <= nums[i]; v2++) {
                for (int v1 = 0; v1 <= v2; v1++) {
                    if (nums[i - 1] - v1 >= nums[i] - v2) {
                        dp[i, v2] = (dp[i, v2] + dp[i - 1, v1]) % mod;
                    }
                }
            }
        }

        int res = 0;
        for (int v = 0; v < 51; v++) {
            res = (res + dp[n - 1, v]) % mod;
        }
        return res;
    }
}
