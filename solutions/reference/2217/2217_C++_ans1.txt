class Solution {
public:
    vector<long long> kthPalindrome(vector<int>& queries, int intLength) {
        int l = (intLength + 1) / 2;   
        int start = (int) pow(10, l - 1) - 1;   
        int limit = (int) pow(10, l) - 1;   
        vector<long long> res;
        
        auto recover = [&](int num) -> long long {
            string s = to_string(num);
            if (intLength % 2 == 0) {
                for (int i = l - 1; i >= 0; --i) {
                    s.push_back(s[i]);
                }
            } else {
                for (int i = l - 2; i >= 0; --i) {
                    s.push_back(s[i]);
                }
            }
            return stoll(s);
        };
        
        
        for (int query: queries) {
            if (start + query > limit) {
                
                res.push_back(-1);
                continue;
            }
            res.push_back(recover(start + query));
        }
        return res;
    }
};
