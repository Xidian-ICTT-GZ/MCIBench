#define MAXN 500

int countQuadruplets(int* nums, int numsSize){
    int ans = 0;
    int * cnt = (int *)malloc(sizeof(int) * MAXN);
    memset(cnt, 0, sizeof(int) * MAXN);
    for (int b = numsSize - 3; b >= 1; --b) {
        for (int d = b + 2; d < numsSize; ++d) {
            if (nums[d] >= nums[b + 1]) {
                ++cnt[nums[d] - nums[b + 1]];
            }
        }
        for (int a = 0; a < b; ++a) {
            ans += cnt[nums[a] + nums[b]];
        }
    }
    free(cnt);
    return ans;
}
