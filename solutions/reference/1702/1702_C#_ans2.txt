public class Solution {
    public string MaximumBinaryString(string binary) {
        int n = binary.Length, i = binary.IndexOf('0');
        if (i < 0) {
            return binary;
        }
        int zeros = 0;
        StringBuilder s = new StringBuilder();
        foreach (char c in binary) {
            if (c == '0') {
                zeros++;
            }
            s.Append('1');
        }
        s[i + zeros - 1] = '0';
        return s.ToString();
    }
}
