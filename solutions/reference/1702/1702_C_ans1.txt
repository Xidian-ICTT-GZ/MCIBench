char* maximumBinaryString(char* binary) {
    int n = strlen(binary);
    char *s = malloc(n + 1);
    strcpy(s, binary);
    int j = 0;
    for (int i = 0; i < n; i++) {
        if (s[i] == '0') {
            while (j <= i || (j < n && s[j] == '1')) {
                j++;
            }
            if (j < n) {
                s[j] = '1';
                s[i] = '1';
                s[i + 1] = '0';
            }
        }
    }
    return s;
}
