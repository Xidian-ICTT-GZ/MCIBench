public class Solution {
    public int[] FindBall(int[][] grid) {
        int n = grid[0].Length;
        int[] ans = new int[n];
        for (int j = 0; j < n; j++) {
            int col = j;  
            foreach (int[] row in grid) {
                int dir = row[col];
                col += dir;  
                if (col < 0 || col == n || row[col] != dir) {  
                    col = -1;
                    break;
                }
            }
            ans[j] = col;  
        }
        return ans;
    }
}
