var findBall = function(grid) {
    const n = grid[0].length;
    const ans = new Array(n);
    for (let j = 0; j < n; j++) {
        let col = j; 
        for (const row of grid) {
            const dir = row[col];
            col += dir; 
            if (col < 0 || col === n || row[col] !== dir) { 
                col = -1;
                break;
            }
        }
        ans[j] = col;  
    }
    return ans;
};
