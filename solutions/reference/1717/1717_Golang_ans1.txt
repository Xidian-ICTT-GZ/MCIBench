func maximumGain(s string, x int, y int) int {
    if x < y {
        x, y = y, x
        s2 := []rune(s)
        for i := range s2 {
            if s2[i] == 'a' {
                s2[i] = 'b'
            } else if s2[i] == 'b' {
                s2[i] = 'a'
            }
        }
        s = string(s2)
    }

    ans := 0
    for i:=0; i < len(s); i++ {
        cntA, cntB := 0, 0
        for i < len(s) && (s[i] == 'a' || s[i] == 'b') {
            if s[i] == 'a' {
                cntA++
            } else {
                if cntA > 0 {
                    cntA--
                    ans += x
                } else {
                    cntB++
                }
            }
            i++
        }
        if cntA < cntB {
            ans += cntA * y
        } else {
            ans += cntB * y
        }
    }
    return ans
}
