bool checkIfPangram(char * sentence) {
    int exist[26];
    memset(exist, 0, sizeof(exist));
    for (int i = 0; sentence[i] != '\0'; i++) {
        exist[sentence[i] - 'a'] = 1;
    }
    for (int i = 0; i < 26; i++) {
        if (exist[i] == 0) {
            return false;
        }
    }
    return true;
}
