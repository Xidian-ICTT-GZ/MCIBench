public class Solution {
    int[] nums;
    int maxOr, cnt;

    public int CountMaxOrSubsets(int[] nums) {
        this.nums = nums;
        this.maxOr = 0;
        this.cnt = 0;
        DFS(0, 0);
        return cnt;
    }

    public void DFS(int pos, int orVal) {
        if (pos == nums.Length) {
            if (orVal > maxOr) {
                maxOr = orVal;
                cnt = 1;
            } else if (orVal == maxOr) {
                cnt++;
            }
            return;
        }
        DFS(pos + 1, orVal | nums[pos]);
        DFS(pos + 1, orVal);
    }
}
