public class Solution {
    public int Calculate(string s) {
        Stack<int> stack = new Stack<int>();
        char preSign = '+';
        int num = 0;
        int n = s.Length;
        for (int i = 0; i < n; ++i) {
            if (char.IsDigit(s[i])) {
                num = num * 10 + s[i] - '0';
            }
            if (!char.IsDigit(s[i]) && s[i] != ' ' || i == n - 1) {
                switch (preSign) {
                case '+':
                    stack.Push(num);
                    break;
                case '-':
                    stack.Push(-num);
                    break;
                case '*':
                    stack.Push(stack.Pop() * num);
                    break;
                default:
                    stack.Push(stack.Pop() / num);
                    break;
                }
                preSign = s[i];
                num = 0;
            }
        }
        int ans = 0;
        while (stack.Count > 0) {
            ans += stack.Pop();
        }
        return ans;
    }
}
