int minLength(char *s){
    int n = strlen(s), m = 0;
    char *stack = (char *)malloc(sizeof(char) * n);
    for (int i = 0; i < n; i++) {
        stack[m++] = s[i];
        if (m >= 2 && (
            stack[m - 2] == 'A' && stack[m - 1] == 'B' ||
            stack[m - 2] == 'C' && stack[m - 1] == 'D'
        )) {
            m -= 2;
        }
    }
    free(stack);
    return m;
}
