public class Solution {
    public int Flipgame(int[] fronts, int[] backs) {
        var same = new HashSet<int>();
        for (int i = 0; i < fronts.Length; i++) {
            if (fronts[i] == backs[i]) {
                same.Add(fronts[i]);
            }
        }
        int res = 3000;
        foreach (var x in fronts) {
            if (x < res && !same.Contains(x)) {
                res = x;
            }
        }
        foreach (var x in backs) {
            if (x < res && !same.Contains(x)) {
                res = x;
            }
        }
        return res % 3000;
    }
}
