class Solution:
    def flipgame(self, fronts: List[int], backs: List[int]) -> int:
        n = len(fronts)
        same = set()
        for i in range(n):
            if fronts[i] == backs[i]:
                same.add(fronts[i])
        res = 3000
        for a in fronts:
            if a < res and a not in same:
                res = a
        for a in backs:
            if a < res and a not in same:
                res = a
        return res if res < 3000 else 0
