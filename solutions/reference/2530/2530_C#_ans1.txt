public class Solution {
    public long MaxKelements(int[] nums, int k) {
        PriorityQueue<int, int> q = new PriorityQueue<int, int>();
        foreach (int num in nums) {
            q.Enqueue(num, -num);
        }
        long ans = 0;
        for (int i = 0; i < k; ++i) {
            int x = q.Dequeue();
            ans += x;
            q.Enqueue((x + 2) / 3, -(x + 2) / 3);
        }
        return ans;
    }
}
