class NumberContainers {
private:
    unordered_map<int, int> nums;
    unordered_map<int, set<int>> us;
public:
    NumberContainers() {
        
    }
    
    void change(int index, int number) {
        if (nums[index] != 0) {
            us[nums[index]].erase(index);
        }
        us[number].insert(index);
        nums[index] = number;
    }
    
    int find(int number) {
        if (us[number].empty()) {
            return -1;
        }
        return *us[number].begin();
    }
};
