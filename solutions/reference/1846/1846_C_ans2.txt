int maximumElementAfterDecrementingAndRearranging(int *arr, int arrSize) {
    int cnt[arrSize + 1];
    memset(cnt, 0, sizeof(cnt));
    for (int i = 0; i < arrSize; i++) {
        cnt[(int)fmin(arr[i], arrSize)]++;
    }
    int miss = 0;
    for (int i = 1; i <= arrSize; ++i) {
        if (cnt[i] == 0) {
            ++miss;
        } else {
            miss -= fmin(cnt[i] - 1, miss);  
        }
    }
    return arrSize - miss;
}
