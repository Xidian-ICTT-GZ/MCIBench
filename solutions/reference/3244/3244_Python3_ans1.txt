class Solution:
    def shortestDistanceAfterQueries(self, n: int, queries: List[List[int]]) -> List[int]:
        roads = [i + 1 for i in range(n)]
        res = []
        dist = n - 1
        for query in queries:
            k = roads[query[0]]
            while k != -1 and k < query[1]:
                if k < query[1]:
                    roads[query[0]] = query[1]
                roads[k], k = -1, roads[k]
                dist -= 1
            res.append(dist)
        return res
