long long findTheArrayConcVal(int* nums, int numsSize) {
    long long ans = 0;
    char str[16];
    for (int i = 0, j = numsSize - 1; i <= j; i++, j--) {
        if (i != j) {
            sprintf(str, "%d%d", nums[i], nums[j]);
            ans += atoi(str);
        } else {
            ans += nums[i];
        }
    }
    return ans;
}
