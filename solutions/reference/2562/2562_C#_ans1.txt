public class Solution {
    public long FindTheArrayConcVal(int[] nums) {
        long ans = 0;
        for (int i = 0, j = nums.Length - 1; i <= j; i++, j--) {
            if (i != j) {
                ans += int.Parse(nums[i].ToString() + nums[j].ToString());
            } else {
                ans += nums[i];
            }
        }
        return ans;
    }
}
