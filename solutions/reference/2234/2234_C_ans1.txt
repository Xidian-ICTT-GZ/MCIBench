int cmp(const void* a, const void* b) {
    return *(int*)b - *(int*)a;
}

long long maximumBeauty(int* flowers, int flowersSize, long long newFlowers, int target, int full, int partial) {
    int n = flowersSize;
    for (int i = 0; i < n; i++) {
        if (flowers[i] > target) {
            flowers[i] = target;
        }
    }
    qsort(flowers, n, sizeof(int), cmp);

    long sum = 0;
    for (int i = 0; i < n; i++) {
        sum += flowers[i];
    }
    long ans = 0;
    if ((long)target * n - sum <= newFlowers) {
        ans = (long)full * n;
    }
    long pre = 0;
    int ptr = 0;
    for (int i = 0; i < n; i++) {
        if (i != 0) {
            pre += flowers[i - 1];
        }
        if (flowers[i] == target) {
            continue;
        }
        long rest = newFlowers - ((long)target * i - pre);
        if (rest < 0) {
            break;
        }
        while (!(ptr >= i && (long)flowers[ptr] * (n - ptr) - sum <= rest)) {
            sum -= flowers[ptr];
            ptr++;
        }
        rest -= (long)flowers[ptr] * (n - ptr) - sum;
        ans = fmax(ans, (long)full * i + (long)partial * fmin(flowers[ptr] + rest / (n - ptr), (long)target - 1));
    }

    return ans;
}
