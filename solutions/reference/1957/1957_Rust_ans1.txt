impl Solution {
    pub fn make_fancy_string(s: String) -> String {
        let mut res = Vec::new();   
        
        for ch in s.chars() {
            let n = res.len();
            if n >= 2 && res[n - 1] == ch && res[n - 2] == ch {
                
                continue;
            }
            
            res.push(ch);
        }
        res.into_iter().collect()
    }
}
