public class Solution {
    public string MakeFancyString(string s) {
        StringBuilder res = new StringBuilder();   
        
        foreach (char ch in s) {
            int n = res.Length;
            if (n >= 2 && res[n-1] == ch && res[n-2] == ch) {
                
                continue;
            }
            
            res.Append(ch);
        }
        return res.ToString();
    }
}
