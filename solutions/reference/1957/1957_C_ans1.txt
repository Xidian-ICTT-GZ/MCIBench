char* makeFancyString(char* s) {
    int len = strlen(s);
    char* res = (char*)malloc(len + 1);   
    int pos = 0;
    
    for (int i = 0; i < len; i++) {
        char ch = s[i];
        if (pos >= 2 && res[pos-1] == ch && res[pos-2] == ch) {
            
            continue;
        }
        
        res[pos++] = ch;
    }
    res[pos] = '\0';
    return res;
}
