func checkDistances(s string, distance []int) bool {
    firstIndex := make([]int, 26)
    for i := 0; i < len(s); i++ {
        idx := s[i] - 'a'
        if firstIndex[idx] != 0 && i - firstIndex[idx] != distance[idx] {
            return false
        }
        firstIndex[idx] = i + 1
    }
    return true
}
