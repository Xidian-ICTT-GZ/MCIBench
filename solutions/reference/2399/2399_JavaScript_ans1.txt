var checkDistances = function(s, distance) {
    let n = s.length;
    for (let i = 0; i < n; i++) {
        for (let j = i + 1; j < n; j++) {
            if (s[i] == s[j] && distance[s.charCodeAt(i) - 'a'.charCodeAt(0)] != j - i - 1) {
                return false;
            }
        }
    }
    return true;
};
