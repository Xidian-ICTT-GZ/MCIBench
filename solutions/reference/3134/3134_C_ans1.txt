#define MAX_SIZE 100001


int check(int* nums, int numsSize, int t, long long median) {
    int cnt[MAX_SIZE] = {0};
    long long tot = 0;
    int diff = 0;
    for (int i = 0, j = 0; i < numsSize; i++) {
        cnt[nums[i]]++;
        if (cnt[nums[i]] == 1) {
            diff++;
        }
        while (diff > t) {
            cnt[nums[j]]--;
            if (cnt[nums[j]] == 0) {
                diff--;
            }
            j++;
        }
        tot += i - j + 1;
    }
    return tot >= median;
}

int medianOfUniquenessArray(int* nums, int numsSize) {
    int n = numsSize;
    int lo = 1, hi = numsSize;
    long long median = ((long long)n * (n + 1) / 2 + 1) / 2;
    int res = 0;

    while (lo <= hi) {
        int mid = (lo + hi) / 2;
        if (check(nums, n, mid, median)) {
            res = mid;
            hi = mid - 1;
        } else {
            lo = mid + 1;
        }
    }

    return res;
}
