public class Solution {
    public int MinimumRounds(int[] tasks) {
        IDictionary<int, int> cnt = new Dictionary<int, int>();
        foreach (int task in tasks) {
            cnt.TryAdd(task, 0);
            cnt[task]++;
        }
        int res = 0;
        foreach (int v in cnt.Values) {
            if (v == 1) {
                return -1;
            }
            if (v % 3 == 0) {
                res += v / 3;
            } else {
                res += 1 + v / 3;
            }
        }
        return res;
    }
}
