use std::collections::HashMap;

impl Solution {
    pub fn minimum_rounds(tasks: Vec<i32>) -> i32 {
        let mut cnt = HashMap::new();
        for &t in &tasks {
            *cnt.entry(t).or_insert(0) += 1;
        }
        let mut res = 0;
        for (&_, &v) in &cnt {
            if v == 1 {
                return -1;
            } else if v % 3 == 0 {
                res += v / 3;
            } else {
                res += v / 3 + 1;
            }
        }
        res
    }
}
