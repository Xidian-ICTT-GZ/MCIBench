public class Solution {
    public long MaximumValueSum(int[] nums, int k, int[][] edges) {
        long f0 = 0, f1 = long.MinValue;
        foreach (int x in nums) {
            long t = Math.Max(f1 + x, f0 + (x ^ k));
            f0 = Math.Max(f0 + x, f1 + (x ^ k));
            f1 = t;
        }
        return f0;
    }
}
