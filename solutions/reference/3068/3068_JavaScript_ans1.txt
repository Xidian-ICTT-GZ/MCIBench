var maximumValueSum = function(nums, k, edges) {
    let res = nums.reduce((a, b) => a + b, 0);
    let diff = nums.map(x => (x ^ k) - x);
    diff.sort((a, b) => a - b);
    for (let i = diff.length - 1; i > 0 && diff[i] + diff[i - 1] >= 0; i -= 2) {
        res += diff[i] + diff[i - 1];
    }
    return res;
};
