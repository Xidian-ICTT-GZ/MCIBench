impl Solution {
    pub fn min_changes(n: i32, k: i32) -> i32 {
        let mut res = 0;
        let mut n = n;
        let mut k = k;

        while n > 0 || k > 0 {
            if (n & 1) == 0 && (k & 1) == 1 {
                return -1;
            }
            if (n & 1) == 1 && (k & 1) == 0 {
                res += 1;
            }
            n >>= 1;
            k >>= 1;
        }

        return res;
    }
}
