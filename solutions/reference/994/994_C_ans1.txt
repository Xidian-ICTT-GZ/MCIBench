const int dr[4] = {0, 1, 0, -1};
const int dc[4] = {1, 0, -1, 0};

int orangesRotting(int** grid, int gridSize, int* gridColSize) {
    int R = gridSize, C = gridColSize[0];
    int queue[R * C];
    int head = 0, tail = 0;
    int depth[R * C];
    memset(depth, 0, sizeof(depth));

    for (int r = 0; r < R; ++r) {
        for (int c = 0; c < C; ++c) {
            if (grid[r][c] == 2) {
                int code = r * C + c;
                queue[tail++] = code;
            }
        }
    }
    int ans = 0;
    while (head != tail) {
        int code = queue[head++];
        int r = code / C, c = code % C;
        for (int k = 0; k < 4; ++k) {
            int nr = r + dr[k];
            int nc = c + dc[k];
            if (0 <= nr && nr < R && 0 <= nc && nc < C && grid[nr][nc] == 1) {
                grid[nr][nc] = 2;
                int ncode = nr * C + nc;
                queue[tail++] = ncode;
                depth[ncode] = depth[code] + 1;
                ans = depth[ncode];
            }
        }
    }
    for (int i = 0; i < R; i++) {
        for (int j = 0; j < C; j++) {
            if (grid[i][j] == 1) {
                return -1;
            }
        }
    }
    return ans;
}
