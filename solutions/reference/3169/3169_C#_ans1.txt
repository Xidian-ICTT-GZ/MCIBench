public class Solution {
    public int CountDays(int days, int[][] meetings) {
        Array.Sort(meetings, (a, b) => a[0].CompareTo(b[0]));

        int l = 1, r = 0;
        foreach (var m in meetings) {
            if (m[0] > r) {
                days -= (r - l + 1);
                l = m[0];
            }
            r = Math.Max(r, m[1]);
        }
        days -= (r - l + 1);
        return days;
    }
}
