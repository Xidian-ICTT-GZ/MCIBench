class Solution:
    def countDistinct(self, nums: List[int], k: int, p: int) -> int:
        arrs = set()   
        n = len(nums)
        
        for i in range(n):
            cnt = 0   
            arr = []   
            for j in range(i, n):
                if nums[j] % p == 0:
                    cnt += 1
                    if cnt > k:
                        break
                arr.append(nums[j])
                
                arrs.add(tuple(arr))
        return len(arrs)
