public class Solution {
    public int NumRescueBoats(int[] people, int limit) {
        int ans = 0;
        Array.Sort(people);
        int light = 0, heavy = people.Length - 1;
        while (light <= heavy) {
            if (people[light] + people[heavy] <= limit) {
                ++light;
            }
            --heavy;
            ++ans;
        }
        return ans;
    }
}
