var countQuadruplets = function(nums) {
    const n = nums.length;
    const pre = Array(n + 1).fill(0);
    let ans = 0;

    for (let j = 0; j < n; ++j) {
        let suf = 0;
        for (let k = n - 1; k > j; --k) {
            if (nums[j] > nums[k]) {
                ans += pre[nums[k]] * suf;
            } else {
                ++suf;
            }
        }
        for (let x = nums[j] + 1; x <= n; ++x) {
            ++pre[x];
        }
    }
    return ans;
};
