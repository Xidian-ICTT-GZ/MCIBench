public class Solution {
    public bool IsSubstringPresent(string s) {
        for (int i = 0; i + 1 < s.Length; i++) {
            StringBuilder sb = new StringBuilder();
            sb.Append(s[i + 1]);
            sb.Append(s[i]);
            string substr = sb.ToString();
            if (s.Contains(substr)) {
                return true;
            }
        }
        return false;
    }
}
