class Solution {
public:
    string removeOccurrences(string s, string part) {
        int m = part.size();
        string res;
        for (const char ch: s){
            
            res.push_back(ch);
            int n = res.size();
            if (n >= m && res.substr(n - m, n) == part){
                
                res = res.substr(0, n - m);
            }
        }
        return res;
    }
};
