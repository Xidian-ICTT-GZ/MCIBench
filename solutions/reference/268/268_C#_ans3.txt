public class Solution {
    public int MissingNumber(int[] nums) {
        int xor = 0;
        int n = nums.Length;
        for (int i = 0; i < n; i++) {
            xor ^= nums[i];
        }
        for (int i = 0; i <= n; i++) {
            xor ^= i;
        }
        return xor;
    }
}
