impl Solution {
    pub fn count_k_constraint_substrings(s: String, k: i32) -> i32 {
        let n = s.len();
        let s: Vec<u8> = s.bytes().collect(); 
        let mut count = [0, 0];
        let mut res:i32 = 0;
        let mut i = 0;

        for j in 0..n {
            count[s[j] as usize - b'0' as usize] += 1; 
            while count[0] > k && count[1] > k {
                count[s[i] as usize - b'0' as usize] -= 1;
                i += 1;
            }
            res += (j - i + 1) as i32;
        }
        res
    }
}
