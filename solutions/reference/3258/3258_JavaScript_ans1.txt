var countKConstraintSubstrings = function(s, k) {
    const n = s.length;
    let res = 0;
    for (let i = 0; i < n; ++i) {
        const count = [0, 0];
        for (let j = i; j < n; ++j) {
            count[parseInt(s[j], 10)]++;
            if (count[0] > k && count[1] > k) {
                break;
            }
            res++;
        }
    }
    return res;
};
