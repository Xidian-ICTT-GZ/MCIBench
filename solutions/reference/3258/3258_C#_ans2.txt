public class Solution {
    public int CountKConstraintSubstrings(string s, int k) {
        int n = s.Length;
        int[] count = new int[2];
        int res = 0;
        for (int i = 0, j = 0; j < n; ++j) {
            count[s[j] - '0']++;
            while (count[0] > k && count[1] > k) {
                count[s[i] - '0']--;
                i++;
            }
            res += j - i + 1;
        }
        return res;
    }
}
