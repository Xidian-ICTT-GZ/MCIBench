public class Solution {
    public int DeleteGreatestValue(int[][] grid) {
        int m = grid.Length, n = grid[0].Length;
        for (int i = 0; i < m; i++) {
            Array.Sort(grid[i]);
        }
        int res = 0;
        for (int j = 0; j < n; j++) {
            int mx = 0;
            for (int i = 0; i < m; i++) {
                mx = Math.Max(mx, grid[i][j]);
            }
            res += mx;
        }
        return res;
    }
}
