public class Solution {
    public int MinOperations(string[] logs) {
        int depth = 0;
        foreach (string log in logs) {
            if ("./".Equals(log)) {
                continue;
            } else if ("../".Equals(log)) {
                if (depth > 0) {
                    depth--;
                }
            } else {
                depth++;
            }
        }
        return depth;
    }
}
