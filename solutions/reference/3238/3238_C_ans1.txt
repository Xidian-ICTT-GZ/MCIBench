int winningPlayerCount(int n, int** pick, int pickSize, int* pickColSize) {
    int cnt[n][11];
    memset(cnt, 0, sizeof(cnt));
    for (int i = 0; i < pickSize; i++) {
        cnt[pick[i][0]][pick[i][1]]++;
    }
    int ans = 0;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j <= 10; j++) {
            if (cnt[i][j] > i) {
                ans++;
                break;
            }
        }
    }
    return ans;
}
