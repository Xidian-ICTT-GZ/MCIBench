class Solution:
    def kSum(self, nums: List[int], k: int) -> int:
        n = len(nums)
        total, total2 = 0, 0
        for i in range(n):
            if nums[i] >= 0:
                total += nums[i]
            else:
                nums[i] = -nums[i]
            total2 += nums[i]
        nums.sort()

        cnt = 0
        def dfs(i: int, t: int, limit: int) -> int:
            nonlocal cnt
            if i == n or cnt >= k - 1 or t + nums[i] > limit:
                return
            cnt += 1
            dfs(i + 1, t + nums[i], limit)
            dfs(i + 1, t, limit)
        
        left, right = 0, total2
        while left <= right:
            mid = (left + right) // 2
            cnt = 0
            dfs(0, 0, mid)
            if cnt >= k - 1:
                right = mid - 1
            else:
                left = mid + 1
        return total - left
