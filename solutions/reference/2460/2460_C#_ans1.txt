public class Solution {
    public int[] ApplyOperations(int[] nums) {
        int n = nums.Length;
        for (int i = 0, j = 0; i < n; i++) {
            if (i + 1 < n && nums[i] == nums[i + 1]) {
                nums[i] *= 2;
                nums[i + 1] = 0;
            }
            if (nums[i] != 0) {
                Swap(nums, i, j);
                j++;
            }
        }
        return nums;
    }

    public void Swap(int[] nums, int i, int j) {
        int temp = nums[i];
        nums[i] = nums[j];
        nums[j] = temp;
    }
}
