var maxCoins = function(piles) {
    piles.sort((a, b) => a - b);
    let length = piles.length;
    let rounds = Math.floor(length / 3);
    let coins = 0;
    let index = length - 2;
    for (let i = 0; i < rounds; i++) {
        coins += piles[index];
        index -= 2;
    }
    return coins;
};
