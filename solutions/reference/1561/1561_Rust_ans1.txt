impl Solution {
    pub fn max_coins(piles: Vec<i32>) -> i32 {
        let mut piles = piles;
        piles.sort();
        let length = piles.len();
        let rounds = length / 3;
        let mut coins = 0;
        let mut index = length - 2;
        for _ in 0..rounds {
            coins += piles[index];
            index -= 2;
        }
        coins
    }
}
