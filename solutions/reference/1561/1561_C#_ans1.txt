public class Solution {
    public int MaxCoins(int[] piles) {
        Array.Sort(piles);
        int length = piles.Length;
        int rounds = length / 3;
        int coins = 0;
        int index = length - 2;
        for (int i = 0; i < rounds; i++) {
            coins += piles[index];
            index -= 2;
        }
        return coins;
    }
}
