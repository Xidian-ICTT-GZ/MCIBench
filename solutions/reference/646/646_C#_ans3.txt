public class Solution {
    public int FindLongestChain(int[][] pairs) {
        int curr = int.MinValue, res = 0;
        Array.Sort(pairs, (a, b) => a[1] - b[1]);
        foreach (int[] p in pairs) {
            if (curr < p[0]) {
                curr = p[1];
                res++;
            }
        }
        return res;
    }
}
