class Solution:
    def rangeSum(self, nums: List[int], n: int, left: int, right: int) -> int:
        MODULO = 10**9 + 7
        sumsLength = n * (n + 1) // 2
        sums = list()
        index = 0

        for i in range(n):
            total = 0
            for j in range(i, n):
                total += nums[j]
                sums.append(total)
        
        sums.sort()
        ans = sum(sums[left-1:right]) % MODULO
        return ans
