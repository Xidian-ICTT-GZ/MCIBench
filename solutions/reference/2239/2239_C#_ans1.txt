public class Solution {
    public int FindClosestNumber(int[] nums) {
        int res = nums[0];   
        int dis = Math.Abs(nums[0]);   
        foreach (int num in nums) {
            if (Math.Abs(num) < dis) {
                dis = Math.Abs(num);
                res = num;
            } else if (Math.Abs(num) == dis) {
                res = Math.Max(res, num);
            }
        }
        return res;
    }
}
