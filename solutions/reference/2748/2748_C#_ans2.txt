public class Solution {
    public int CountBeautifulPairs(int[] nums) {
        int res = 0;
        int[] cnt = new int[10];
        foreach (int x in nums) {
            for (int y = 1; y <= 9; y++) {
                if (GCD(x % 10, y) == 1) {
                    res += cnt[y];
                }
            }
            int x2 = x;
            while (x2 >= 10) {
                x2 /= 10;
            }
            cnt[x2]++;
        }
        return res;
    }

    private int GCD(int a, int b) {
        return b == 0 ? a : GCD(b, a % b);
    }
}
