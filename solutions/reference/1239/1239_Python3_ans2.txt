class Solution:
    def maxLength(self, arr: List[str]) -> int:
        ans = 0
        masks = [0]
        for s in arr:
            mask = 0
            for ch in s:
                idx = ord(ch) - ord("a")
                if ((mask >> idx) & 1):   
                    mask = 0
                    break
                mask |= 1 << idx   
            if mask == 0:
                continue
            
            n = len(masks)
            for i in range(n):
                m = masks[i]
                if (m & mask) == 0:   
                    masks.append(m | mask)
                    ans = max(ans, bin(m | mask).count("1"))
        
        return ans
