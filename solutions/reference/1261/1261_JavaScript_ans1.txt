var FindElements = function(root) {
    this.valSet = new Set();
    const dfs = (node, val) => {
        if (node == null) {
            return;
        }
        node.val = val;
        this.valSet.add(val);
        dfs(node.left, val * 2 + 1)
        dfs(node.right, val * 2 + 2)
    }
    dfs(root, 0);
};

FindElements.prototype.find = function(target) {
    return this.valSet.has(target);
};
