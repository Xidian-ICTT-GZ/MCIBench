public class Solution {
    public long CountBadPairs(int[] nums) {
        Dictionary<int, int> mp = new Dictionary<int, int>();
        long res = 0;
        for (int i = 0; i < nums.Length; i++) {
            int key = nums[i] - i;
            mp.TryGetValue(key, out int count);
            res += i - count;
            mp[key] = count + 1;
        }
        return res;
    }
}
