public class Solution {
    public long MaximumTripletValue(int[] nums) {
        int n = nums.Length;
        long res = 0;
        for (int k = 2; k < n; k++) {
            int m = nums[0];
            for (int j = 1; j < k; j++) {
                res = Math.Max(res, (long)(m - nums[j]) * nums[k]);
                m = Math.Max(m, nums[j]);
            }
        }
        return res;
    }
}
