func closestMeetingNode(edges []int, node1 int, node2 int) int {
    n := len(edges)
    get := func(node int) []int {
        n := len(edges)
        dist := make([]int, n)
        for i := range dist {
            dist[i] = -1
        }
        distance := 0
        for node != -1 && dist[node] == -1 {
            dist[node] = distance
            distance++
            node = edges[node]
        }
        return dist
    }

    d1 := get(node1)
    d2 := get(node2)
    res := -1
    for i := 0; i < n; i++ {
        if d1[i] != -1 && d2[i] != -1 {
            if res == -1 || max(d1[res], d2[res]) > max(d1[i], d2[i]) {
                res = i
            }
        }
    }
    return res
}
