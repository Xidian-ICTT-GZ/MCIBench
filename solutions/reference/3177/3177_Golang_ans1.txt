func maximumLength(nums []int, k int) int {
    lenNums := len(nums)
	dp := make(map[int][]int)
	zd := make([]int, k + 1)

	for i := 0; i < lenNums; i++ {
		v := nums[i]
		if _, ok := dp[v]; !ok {
			dp[v] = make([]int, k + 1)
		}

		tmp := dp[v]
		for j := 0; j <= k; j++ {
			tmp[j]++
			if j > 0 {
				tmp[j] = max(tmp[j], zd[j - 1] + 1)
			}
		}

		for j := 0; j <= k; j++ {
			zd[j] = max(zd[j], tmp[j])
		}
	}
	return zd[k]
}
