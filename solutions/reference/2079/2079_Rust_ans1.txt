impl Solution {
    pub fn watering_plants(plants: Vec<i32>, capacity: i32) -> i32 {
        let mut ans = 0;
        let mut rest = capacity;
        for (i, &plant) in plants.iter().enumerate() {
            if rest >= plant {
                ans += 1;
                rest -= plant;
            } else {
                ans += i as i32 * 2 + 1;
                rest = capacity - plant;
            }
        }
        ans
    }
}
