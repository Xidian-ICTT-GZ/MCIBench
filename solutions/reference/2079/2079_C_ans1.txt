int wateringPlants(int* plants, int plantsSize, int capacity) {
    int ans = 0;
    int rest = capacity;
    for (int i = 0; i < plantsSize; ++i) {
        if (rest >= plants[i]) {
            ++ans;
            rest -= plants[i];
        } else {
            ans += i * 2 + 1;
            rest = capacity - plants[i];
        }
    }
    return ans;
}
