public class Solution {
    const int MOD = 1337;

    public int SuperPow(int a, int[] b) {
        int ans = 1;
        foreach (int e in b) {
            ans = (int) ((long) Pow(ans, 10) * Pow(a, e) % MOD);
        }
        return ans;
    }

    public int Pow(int x, int n) {
        int res = 1;
        while (n != 0) {
            if (n % 2 != 0) {
                res = (int) ((long) res * x % MOD);
            }
            x = (int) ((long) x * x % MOD);
            n /= 2;
        }
        return res;
    }
}
