use std::collections::HashSet;

impl Solution {
    pub fn dest_city(paths: Vec<Vec<String>>) -> String {
        let mut citiesA: HashSet<String> = HashSet::new();
        for path in &paths {
            citiesA.insert(path[0].clone());
        }
        for path in &paths {
            if !citiesA.contains(&path[1]) {
                return path[1].clone();
            }
        }
        String::from("")
    }
}
