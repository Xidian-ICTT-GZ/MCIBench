class Solution:
    def largestCombination(self, candidates: List[int]) -> int:
        
        def maxlen(k: int):
            res = 0
            for num in candidates:
                if num & (1 << k):
                    res += 1
            return res
        
        res = 0
        for i in range(24):
            
            res = max(res, maxlen(i))
        return res
