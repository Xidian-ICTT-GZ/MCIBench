class Solution {
    
    public int maxlen(int[] candidates, int k) {
        int res = 0;
        for (int num : candidates) {
            if ((num & (1 << k)) != 0) {
                res++;
            }
        }
        return res;
    }

    public int largestCombination(int[] candidates) {
        int res = 0;
        for (int i = 0; i < 24; i++) {
            
            res = Math.max(res, maxlen(candidates, i));
        }
        return res;
    }
}
