public class Solution {
    public bool CheckValidString(string s) {
        int minCount = 0, maxCount = 0;
        int n = s.Length;
        for (int i = 0; i < n; i++) {
            char c = s[i];
            if (c == '(') {
                minCount++;
                maxCount++;
            } else if (c == ')') {
                minCount = Math.Max(minCount - 1, 0);
                maxCount--;
                if (maxCount < 0) {
                    return false;
                }
            } else {
                minCount = Math.Max(minCount - 1, 0);
                maxCount++;
            }
        }
        return minCount == 0;
    }
}
