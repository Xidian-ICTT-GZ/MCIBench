int maxPointsInsideSquare(int** points, int pointsSize, int* pointsColSize, char* s) {
    int min1[26];
    for(int i = 0; i < 26; i++) {
        min1[i] = INT_MAX;
    }
    int min2 = INT_MAX;
    int n = pointsSize;
    for (int i = 0; i < n; ++i) {
        int x = points[i][0], y = points[i][1], j = s[i] - 'a';
        int d = (abs(x) > abs(y)) ? abs(x) : abs(y);
        if (d < min1[j]) {
            min2 = (min2 < min1[j]) ? min2 : min1[j];
            min1[j] = d;
        } else if (d < min2) {
            min2 = d;
        }
    }
    int res = 0;
    for (int i = 0; i < 26; i++) {
        if (min1[i] < min2) {
            ++res;
        }
    }
    return res;
}
