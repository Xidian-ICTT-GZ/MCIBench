var reverse = function(head) {
    let dummy = new ListNode();
    while (head != null) {
        let p = head;
        head = head.next;
        p.next = dummy.next;
        dummy.next = p;
    }
    return dummy.next;
};

var removeNodes = function(head) {
    head = reverse(head);
    for (let p = head; p.next != null; ) {
        if (p.val > p.next.val) {
            p.next = p.next.next;
        } else {
            p = p.next;
        }
    }
    return reverse(head);
};
