class Solution:
    def numSteps(self, s: str) -> int:
        n, ans = len(s), 0
        
        meet1 = False
        for i in range(n - 1, -1, -1):
            if s[i] == '0':
                
                
                
                ans += (2 if meet1 else 1)
            else:
                
                
                
                
                if not meet1:
                    if i != 0:
                        ans += 2
                    meet1 = True
                else:
                    ans += 1
        return ans
