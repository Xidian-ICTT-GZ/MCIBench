var countKConstraintSubstrings = function(s, k) {
    const n = s.length;
    const count = [0, 0];
    let res = 0;
    let i = 0;
    for (let j = 0; j < n; ++j) {
        count[s[j] - '0']++;
        while (count[0] > k && count[1] > k) {
            count[s[i] - '0']--;
            i++;
        }
        res += j - i + 1;
    }
    return res;
};
