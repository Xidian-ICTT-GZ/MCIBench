class Solution:
    def countKConstraintSubstrings(self, s: str, k: int) -> int:
        n = len(s)
        count = [0, 0]
        res = 0
        i = 0
        for j in range(n):
            count[int(s[j])] += 1
            while count[0] > k and count[1] > k:
                count[int(s[i])] -= 1
                i += 1
            res += j - i + 1
        return res
