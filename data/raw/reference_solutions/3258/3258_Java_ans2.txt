class Solution {
    public int countKConstraintSubstrings(String s, int k) {
        int n = s.length();
        int[] count = new int[2];
        int res = 0;
        for (int i = 0, j = 0; j < n; ++j) {
            count[s.charAt(j) - '0']++;
            while (count[0] > k && count[1] > k) {
                count[s.charAt(i) - '0']--;
                i++;
            }
            res += j - i + 1;
        }
        return res;
    }
}
