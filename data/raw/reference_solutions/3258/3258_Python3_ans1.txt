class Solution:
    def countKConstraintSubstrings(self, s: str, k: int) -> int:
        n = len(s)
        res = 0
        for i in range(n):
            count = [0, 0]
            for j in range(i, n):
                count[int(s[j])] += 1
                if count[0] > k and count[1] > k:
                    break
                res += 1
        return res
