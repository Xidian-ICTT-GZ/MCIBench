int countKConstraintSubstrings(char* s, int k) {
    int n = strlen(s);
    int count[2] = {0};
    int res = 0;
    int i = 0;
    for (int j = 0; j < n; ++j) {
        count[s[j] - '0']++;
        while (count[0] > k && count[1] > k) {
            count[s[i] - '0']--;
            i++;
        }
        res += j - i + 1;
    }
    return res;
}
