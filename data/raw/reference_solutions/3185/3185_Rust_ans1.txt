impl Solution {
    pub fn count_complete_day_pairs(hours: Vec<i32>) -> i64 {
        let mut ans: i64 = 0;
        let mut cnt = vec![0; 24];
        for hour in hours {
            ans += cnt[(24 - hour % 24) as usize % 24] as i64;
            cnt[hour as usize % 24] += 1;
        }
        ans
    }
}
