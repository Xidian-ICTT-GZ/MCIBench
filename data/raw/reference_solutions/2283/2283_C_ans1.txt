bool digitCount(char * num) {
    int h[16];
    memset(h, 0, sizeof(h));
    int n = strlen(num);
    for (int i = 0; i < n; i++) {
        h[num[i] - '0']++;
    }
    for (int i = 0; i < n; i++) {
        int v = num[i] - '0';
        if (h[i] != v) {
            return false;
        }
    }
    return true;
}
