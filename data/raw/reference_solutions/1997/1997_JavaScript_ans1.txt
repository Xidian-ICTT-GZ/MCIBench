var firstDayBeenInAllRooms = function(nextVisit) {
    const mod = 1e9 + 7;
    const len = nextVisit.length;
    const dp = new Array(len).fill(0);

    
    dp[0] = 2; 
    for (let i = 1; i < len; i++) {
        const to = nextVisit[i];
        dp[i] = 2 + dp[i - 1];
        if (to) {
            
            dp[i] = (dp[i] - dp[to - 1] + mod) % mod;
        }
        dp[i] = (dp[i] + dp[i - 1]) % mod;
    }
    return dp[len - 2]; 
};
