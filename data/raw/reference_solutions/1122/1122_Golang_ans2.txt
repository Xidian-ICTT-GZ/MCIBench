func relativeSortArray(arr1 []int, arr2 []int) []int {
    rank := map[int]int{}
    for i, v := range arr2 {
        rank[v] = i - len(arr2)
    }
    sort.Slice(arr1, func(i, j int) bool {
        x, y := arr1[i], arr1[j]
        if r, has := rank[x]; has {
            x = r
        }
        if r, has := rank[y]; has {
            y = r
        }
        return x < y
    })
    return arr1
}
