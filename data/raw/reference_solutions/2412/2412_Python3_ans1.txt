class Solution:
    def minimumMoney(self, transactions: List[List[int]]) -> int:
        total_lose = 0
        res = 0
        for t in transactions:
            cost, cashback = t[0], t[1]
            total_lose += max(cost - cashback, 0)
            res = max(res, min(cost, cashback))
        return total_lose + res
