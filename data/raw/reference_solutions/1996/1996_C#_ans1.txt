public class Solution {
    public int NumberOfWeakCharacters(int[][] properties) {
        Array.Sort(properties, (o1, o2) => {
            return o1[0] == o2[0] ? (o1[1] - o2[1]) : (o2[0] - o1[0]);
        });
        int maxDef = 0;
        int ans = 0;
        foreach (int[] p in properties) {
            if (p[1] < maxDef) {
                ans++;
            } else {
                maxDef = p[1];
            }
        }
        return ans;
    }
}
