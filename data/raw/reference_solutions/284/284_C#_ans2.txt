class PeekingIterator<T> {
    private IEnumerator<T> iterator;
    private bool flag;

    public PeekingIterator(IEnumerator<T> iterator) {
        this.iterator = iterator;
        flag = true;
    }
    
    public T Peek() {
        return iterator.Current;
    }
    
    public T Next() {
        T ret = iterator.Current;
        flag = iterator.MoveNext();
        return ret;
    }
    
    public bool HasNext() {
        return flag;
    }
}
