class Solution {
    public List<String> validStrings(int n) {
        List<String> res = new ArrayList<String>();
        int mask = (1 << n) - 1;
        for (int i = 0; i < 1 << n; i++) {
            int t = mask ^ i;
            if (((t >> 1) & t) == 0) {
                String str = Integer.toBinaryString(i);
                StringBuilder sb = new StringBuilder();
                for (int j = n - str.length(); j > 0; j--) {
                    sb.append('0');
                }
                sb.append(str);
                res.add(sb.toString());
            }
        }
        return res;
    }
}
