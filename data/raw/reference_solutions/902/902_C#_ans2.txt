public class Solution {
    public int AtMostNGivenDigitSet(string[] digits, int n) {
        string s = n.ToString();
        int m = digits.Length, k = s.Length;
        IList<int> bits = new List<int>();
        bool isLimit = true;
        for (int i = 0; i < k; i++) {
            if (!isLimit) {
                bits.Add(m - 1);
            } else {
                int selectIndex = -1;
                for (int j = 0; j < m; j++) {
                    if (digits[j][0] <= s[i]) {
                        selectIndex = j;
                    } else {
                        break;
                    }
                }
                if (selectIndex >= 0) {
                    bits.Add(selectIndex);
                    if (digits[selectIndex][0] < s[i]) {
                        isLimit = false;
                    }
                } else {
                    int len = bits.Count;
                    while (bits.Count > 0 && bits[bits.Count - 1] == 0) {
                        bits.RemoveAt(bits.Count - 1);
                    }
                    if (bits.Count > 0) {
                        bits[bits.Count - 1]--;
                    } else {
                        len--;
                    }
                    while (bits.Count <= len) {
                        bits.Add(m - 1);
                    }
                    isLimit = false;
                }
            }
        }
        int ans = 0;
        for (int i = 0; i < bits.Count; i++) {
            ans = ans * m + (bits[i] + 1);
        }
        return ans;
    }
}
