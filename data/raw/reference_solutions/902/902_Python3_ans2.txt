class Solution:
    def atMostNGivenDigitSet(self, digits: List[str], n: int) -> int:
        m = len(digits)
        s = str(n)
        bits = []
        is_limit = True
        for c in s:
            if not is_limit:
                bits.append(m - 1)
                continue
            select_index = -1
            for j, d in enumerate(digits):
                if d > c:
                    break
                select_index = j
            if select_index >= 0:
                bits.append(select_index)
                if digits[select_index] < c:
                    is_limit = False
            else:
                sz = len(bits)
                while bits and bits[-1] == 0:
                    bits.pop()
                if bits:
                    bits[-1] -= 1
                else:
                    sz -= 1
                while len(bits) <= sz:
                    bits.append(m - 1)
                is_limit = False
        ans = 0
        for b in bits:
            ans = ans * m + b + 1
        return ans
