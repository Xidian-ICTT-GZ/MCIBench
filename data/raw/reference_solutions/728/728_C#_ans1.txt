public class Solution {
    public IList<int> SelfDividingNumbers(int left, int right) {
        IList<int> ans = new List<int>();
        for (int i = left; i <= right; i++) {
            if (IsSelfDividing(i)) {
                ans.Add(i);
            }
        }
        return ans;
    }

    public bool IsSelfDividing(int num) {
        int temp = num;
        while (temp > 0) {
            int digit = temp % 10;
            if (digit == 0 || num % digit != 0) {
                return false;
            }
            temp /= 10;
        }
        return true;
    }
}
