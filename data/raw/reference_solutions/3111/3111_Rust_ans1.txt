impl Solution {
    pub fn min_rectangles_to_cover_points(points: Vec<Vec<i32>>, w: i32) -> i32 {
        let mut points = points.clone();
        points.sort_by(|a, b| a[0].cmp(&b[0]));
        let mut res = 0;
        let mut bound = -1;
        for p in points {
            if p[0] > bound {
                bound = p[0] + w;
                res += 1;
            }
        }
        res
    }
}
