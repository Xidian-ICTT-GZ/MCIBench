var maximumBeauty = function(nums, k) {
    let m = Math.max(...nums);
    let diff = new Array(m + 2).fill(0);
    for (let i = 0; i < nums.length; i++) {
        diff[Math.max(nums[i] - k, 0)]++;
        diff[Math.min(nums[i] + k + 1, m + 1)]--;
    }
    let res = 0, count = 0;
    for (let i = 0; i < diff.length; i++) {
        count += diff[i];
        res = Math.max(res, count);
    }
    return res;
};
