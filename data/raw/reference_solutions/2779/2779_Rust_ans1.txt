impl Solution {
    pub fn maximum_beauty(nums: Vec<i32>, k: i32) -> i32 {
        let mut nums = nums.clone();
        nums.sort();
        let mut res = 0;
        let mut i = 0;
        let mut j = 0;
        while i < nums.len() {
            while nums[i] - 2 * k > nums[j] {
                j += 1;
            }
            res = std::cmp::max(res, i - j + 1);
            i += 1;
        }
        res as i32
    }
}
