impl Solution {
    pub fn max_num_of_marked_indices(mut nums: Vec<i32>) -> i32 {
        nums.sort();
        let n = nums.len();
        let m = n / 2;
        let mut res = 0;
        let mut i = 0;
        let mut j = m;
        while i < m && j < n {
            while j < n && 2 * nums[i] > nums[j] {
                j += 1;
            }
            if j < n {
                res += 2;
                j += 1;
            }
            i += 1;
        }

        res
    }
}
