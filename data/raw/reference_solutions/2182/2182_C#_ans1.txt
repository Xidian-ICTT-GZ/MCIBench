public class Solution {
    const int N = 26;

    public string RepeatLimitedString(string s, int repeatLimit) {
        int[] count = new int[N];
        foreach (char c in s) {
            count[c - 'a']++;
        }
        StringBuilder ret = new StringBuilder();
        int m = 0;
        for (int i = N - 1, j = N - 2; i >= 0 && j >= 0;) {
            if (count[i] == 0) { 
                m = 0;
                i--;
            } else if (m < repeatLimit) { 
                count[i]--;
                ret.Append((char) ('a' + i));
                m++;
            } else if (j >= i || count[j] == 0) { 
                j--;
            } else { 
                count[j]--;
                ret.Append((char) ('a' + j));
                m = 0;
            }
        }
        return ret.ToString();
    }
}
