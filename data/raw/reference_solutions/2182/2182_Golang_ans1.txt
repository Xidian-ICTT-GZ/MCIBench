const N = 26

func repeatLimitedString(s string, repeatLimit int) string {
    count := make([]int, 26)
    for _, c := range s {
        count[c - 'a']++
    }
    var ret []byte
    var m int
    for i, j := N - 1, N - 2; i >= 0 && j >= 0; {
        switch {
        case count[i] == 0: 
            m, i = 0, i - 1
        case m < repeatLimit: 
            count[i]--
            ret = append(ret, 'a' + byte(i))
            m++
        case j >= i ||count[j] == 0: 
            j--
        default: 
            count[j]--
            ret = append(ret, 'a' + byte(j))
            m = 0
        }
    }
    return string(ret)
}
