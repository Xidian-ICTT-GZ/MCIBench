public class Solution {
    public int MaximumUniqueSubarray(int[] nums) {
        int n = nums.Length;
        HashSet<int> seen = new HashSet<int>();
        int ans = 0, psum = 0;
        for (int i = 0, j = 0; i < n; ++i) {
            psum += nums[i];
            while (seen.Contains(nums[i])) {
                seen.Remove(nums[j]);
                psum -= nums[j];
                j++;
            }
            seen.Add(nums[i]);
            ans = Math.Max(ans, psum);
        }
        return ans;
    }
}
