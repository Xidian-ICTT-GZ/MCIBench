class Solution {
    public int maximumUniqueSubarray(int[] nums) {
        int n = nums.length;
        int[] psum = new int[n + 1];
        HashMap<Integer, Integer> cnt = new HashMap<>();
        int ans = 0, pre = 0;
        for (int i = 0; i < n; ++i) {
            psum[i + 1] = psum[i] + nums[i];
            pre = Math.max(pre, cnt.getOrDefault(nums[i], 0));
            ans = Math.max(ans, psum[i + 1] - psum[pre]);
            cnt.put(nums[i], i + 1);
        }
        return ans;
    }
}
