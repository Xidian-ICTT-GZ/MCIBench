class Solution:
    def maximumUniqueSubarray(self, nums: List[int]) -> int:
        n = len(nums)
        psum = [0] * (n + 1)
        cnt = {}
        ans = pre = 0
        for i in range(n):
            psum[i + 1] = psum[i] + nums[i]
            pre = max(pre, cnt.get(nums[i], 0))
            ans = max(ans, psum[i + 1] - psum[pre])
            cnt[nums[i]] = i + 1
        return ans
