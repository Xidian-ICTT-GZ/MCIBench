class Solution {
public:
    int maximumUniqueSubarray(vector<int>& nums) {
        int n = nums.size();
        unordered_set<int> seen;
        int ans = 0, psum = 0;
        for(int i = 0, j = 0; i < n; ++i) {
            psum += nums[i];
            while (seen.count(nums[i])) {
                seen.erase(nums[j]);
                psum -= nums[j];
                j++;
            }
            seen.emplace(nums[i]);
            ans = max(ans, psum);
        }
        
        return ans;
    }
};
