public class Solution {
    public int AverageValue(int[] nums) {
        int total = 0, k = 0, n = nums.Length;
        for (int i = 0; i < n; i++) {
            int a = nums[i];
            if (a % 6 == 0) {
                total += a;
                k++;
            }
        }
        return k > 0 ? total / k : 0;
    }
}
