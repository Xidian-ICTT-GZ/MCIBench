var maximumGain = function (s, x, y) {
    if (x < y) {
        [x, y] = [y, x];
        s = s.replace(/./g, c => c === 'a' ? 'b' : c === 'b' ? 'a' : c);
    }

    let ans = 0, i = 0;
    while (i < s.length) {
        let cntA = 0, cntB = 0;
        while (i < s.length && (s[i] === 'a' || s[i] === 'b')) {
            if (s[i] === 'a') {
                cntA++;
            }
            else {
                if (cntA > 0) {
                    cntA--;
                    ans += x;
                } else {
                    cntB++;
                }
            }
            i++;
        }
        ans += Math.min(cntA, cntB) * y;
        i++;
    }
    return ans;
};
