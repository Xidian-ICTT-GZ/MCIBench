int maximumGain(char* s, int x, int y) {
    if (x < y) {
        int tmp = x;
        x = y;
        y = tmp;
        for (int i = 0; s[i]; i++) {
            if (s[i] == 'a'){
                s[i] = 'b';
            }
            else if (s[i] == 'b'){
                s[i] = 'a';
            }
        }
    }

    int ans = 0;
    for (int i = 0; i < strlen(s); i++) {
        int cntA = 0, cntB = 0;
        while (s[i] && (s[i] == 'a' || s[i] == 'b')) {
            if (s[i] == 'a'){
                cntA++;
            }
            else {
                if (cntA > 0) {
                    cntA--;
                    ans += x;
                } else{
                    cntB++;
                }
            }
            i++;
        }
        ans += (cntA < cntB ? cntA : cntB) * y;
    }
    return ans;
}
