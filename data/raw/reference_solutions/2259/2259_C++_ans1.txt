class Solution {
public:
    string removeDigit(string number, char digit) {
        int n = number.size();
        string res;   
        for (int i = 0; i < n; ++i) {
            if (number[i] == digit) {
                string tmp = number.substr(0, i);
                tmp.append(number.substr(i + 1, n - i));
                res = max(res, tmp);
            }
        }
        return res;
    }
};
