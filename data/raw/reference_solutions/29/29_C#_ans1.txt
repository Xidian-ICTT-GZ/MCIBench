public class Solution {
    public int Divide(int dividend, int divisor) {
        
        if (dividend == int.MinValue) {
            if (divisor == 1) {
                return int.MinValue;
            }
            if (divisor == -1) {
                return int.MaxValue;
            }
        }
        
        if (divisor == int.MinValue) {
            return dividend == int.MinValue ? 1 : 0;
        }
        
        if (dividend == 0) {
            return 0;
        }
        
        
        
        bool rev = false;
        if (dividend > 0) {
            dividend = -dividend;
            rev = !rev;
        }
        if (divisor > 0) {
            divisor = -divisor;
            rev = !rev;
        }
        
        int left = 1, right = int.MaxValue, ans = 0;
        while (left <= right) {
            
            int mid = left + ((right - left) >> 1);
            bool check = quickAdd(divisor, mid, dividend);
            if (check) {
                ans = mid;
                
                if (mid == int.MaxValue) {
                    break;
                }
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }

        return rev ? -ans : ans;
    }

    
    public bool quickAdd(int y, int z, int x) {
        
        
        int result = 0, add = y;
        while (z != 0) {
            if ((z & 1) != 0) {
                
                if (result < x - add) {
                    return false;
                }
                result += add;
            }
            if (z != 1) {
                
                if (add < x - add) {
                    return false;
                }
                add += add;
            }
            
            z >>= 1;
        }
        return true;
    }
}
