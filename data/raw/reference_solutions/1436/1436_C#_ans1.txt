public class Solution {
    public string DestCity(IList<IList<string>> paths) {
        ISet<string> citiesA = new HashSet<string>();
        foreach (IList<string> path in paths) {
            citiesA.Add(path[0]);
        }
        foreach (IList<string> path in paths) {
            if (!citiesA.Contains(path[1])) {
                return path[1];
            }
        }
        return "";
    }
}
