class Solution {
public:
    vector<int> recoverArray(int n, vector<int>& sums) {
        
        sort(sums.begin(), sums.end());

        vector<int> ans;
        for (int i = n; i >= 2; --i) {
            int d = sums[1] - sums[0];
            
            int left = 0, right = 0;
            vector<int> s, t;
            
            vector<int> used(1 << i);
            while (true) {
                
                while (left < (1 << i) && used[left]) {
                    ++left;
                }
                if (left == (1 << i)) {
                    break;
                }
                s.push_back(sums[left]);
                used[left] = true;
                
                while (used[right] || sums[right] != sums[left] + d) {
                    ++right;
                }
                t.push_back(sums[right]);
                used[right] = true;
            }
            if (find(s.begin(), s.end(), 0) != s.end()) {
                
                ans.push_back(d);
                sums = move(s);
            }
            else {
                
                ans.push_back(-d);
                sums = move(t);
            }
        }
        
        ans.push_back(sums[0] + sums[1]);
        return ans;
    }
};
