class Solution {
public:
    bool checkValid(vector<vector<int>>& matrix) {
        int n = matrix.size();
        unordered_set<int> occur;   
        
        for (int i = 0; i < n; ++i) {
            occur.clear();   
            for (int j = 0; j < n; ++j) {
                if (occur.count(matrix[i][j])) {
                    
                    return false;
                }
                occur.insert(matrix[i][j]);
            }
        }
        
        for (int i = 0; i < n; ++i) {
            occur.clear();   
            for (int j = 0; j < n; ++j) {
                if (occur.count(matrix[j][i])) {
                    
                    return false;
                }
                occur.insert(matrix[j][i]);
            }
        }
        return true;
    }
};
