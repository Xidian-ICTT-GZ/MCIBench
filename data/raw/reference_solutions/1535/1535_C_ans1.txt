int getWinner(int* arr, int arrSize, int k){
    int prev = fmax(arr[0], arr[1]);
    if (k == 1) {
        return prev;
    }
    int consecutive = 1;
    int maxNum = prev;
    for (int i = 2; i < arrSize; i++) {
        int curr = arr[i];
        if (prev > curr) {
            consecutive++;
            if (consecutive == k) {
                return prev;
            }
        } else {
            prev = curr;
            consecutive = 1;
        }
        maxNum = fmax(maxNum, curr);
    }
    return maxNum;
}
