public class Solution {
    public int MaxMoves(int[][] grid) {
        int m = grid.Length, n = grid[0].Length;
        ISet<int> q = new HashSet<int>();
        for (int i = 0; i < m; i++) {
            q.Add(i);
        }
        for (int j = 1; j < n; j++) {
            ISet<int> q2 = new HashSet<int>();
            foreach (int i in q) {
                for (int i2 = i - 1; i2 <= i + 1; i2++) {
                    if (0 <= i2 && i2 < m && grid[i][j - 1] < grid[i2][j]) {
                        q2.Add(i2);
                    }
                }
            }
            q = q2;
            if (q.Count == 0) {
                return j - 1;
            }
        }
        return n - 1;
    }
}
