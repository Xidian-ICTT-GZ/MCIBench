#define MAX(a, b) ((a) > (b) ? (a) : (b))

int maxScore(char * s){
    int score = 0;
    int n = strlen(s);
    if (s[0] == '0') {
        score++;
    }
    for (int i = 1; i < n; i++) {
        if (s[i] == '1') {
            score++;
        }
    }
    int ans = score;
    for (int i = 1; i < n - 1; i++) {
        if (s[i] == '0') {
            score++;
        } else {
            score--;
        }
        ans = MAX(ans, score);
    }
    return ans;
}
