long long maximumSumOfHeights(int* maxHeights, int maxHeightsSize){
    int n = maxHeightsSize;
    long long res = 0;
    for (int i = 0; i < n; i++) {
        int pre = maxHeights[i];
        long long sum = pre;
        for (int j = i - 1; j >= 0; j--) {
            pre = fmin(pre, maxHeights[j]);
            sum += pre;
        }
        int suf = maxHeights[i];
        for (int j = i + 1; j < n; j++) {
            suf = fmin(suf, maxHeights[j]);
            sum += suf;
        }
        res = fmax(res, sum);
    }
    return res;
}
