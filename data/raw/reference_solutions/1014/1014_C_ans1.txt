int maxScoreSightseeingPair(int* values, int valuesSize) {
    int ans = 0, mx = values[0];
    for (int j = 1; j < valuesSize; ++j) {
        ans = (ans > mx + values[j] - j) ? ans : mx + values[j] - j;
        
        mx = (mx > values[j] + j) ? mx : values[j] + j;
    }
    return ans;
}
