var maxTurbulenceSize = function(arr) {
    const n = arr.length;
    const dp = new Array(n).fill(0).map(() => new Array(2).fill(0));
    dp[0][0] = dp[0][1] = 1;
    for (let i = 1; i < n; i++) {
        dp[i][0] = dp[i][1] = 1;
        if (arr[i - 1] > arr[i]) {
            dp[i][0] = dp[i - 1][1] + 1;
        } else if (arr[i - 1] < arr[i]) {
            dp[i][1] = dp[i - 1][0] + 1;
        }
    }

    let ret = 1;
    for (let i = 0; i < n; i++) {
        ret = Math.max(ret, dp[i][0]);
        ret = Math.max(ret, dp[i][1]);
    }
    return ret;
};
