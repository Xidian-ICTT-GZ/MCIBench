var largestVariance = function(s) {
    const pos = {};
    for (let i = 0; i < s.length; i++) {
        const ch = s[i];
        if (!pos[ch]) pos[ch] = [];
        pos[ch].push(i);
    }

    let ans = 0;
    for (let c0 in pos) {
        for (let c1 in pos) {
            if (c0 !== c1) {
                const pos0 = pos[c0];
                const pos1 = pos[c1];
                let i = 0, j = 0;
                let f = 0, g = -Infinity;
                while (i < pos0.length || j < pos1.length) {
                    if (j === pos1.length || (i < pos0.length && pos0[i] < pos1[j])) {
                        f = Math.max(f, 0) + 1;
                        g = g + 1;
                        i++;
                    } else {
                        g = Math.max(f, Math.max(g, 0)) - 1;
                        f = Math.max(f, 0) - 1;
                        j++;
                    }
                    ans = Math.max(ans, g);
                }
            }
        }
    }
    return ans;
};
