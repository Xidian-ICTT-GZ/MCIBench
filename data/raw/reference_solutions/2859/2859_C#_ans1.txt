public class Solution {
    public int SumIndicesWithKSetBits(IList<int> nums, int k) {
        int ans = 0;
        for (int i = 0; i < nums.Count; ++i) {
            if (BitCount(i) == k) {
                ans += nums[i];
            }
        }
        return ans;
    }

    public int BitCount(int x) {
        int cnt = 0;
        while (x != 0) {
            cnt += (x % 2);
            x /= 2;
        }
        return cnt;
    }
}
