int bitCount(int x) {
    int cnt = 0;
    while (x) {
        cnt += (x % 2);
        x /= 2;
    }
    return cnt;
};

int sumIndicesWithKSetBits(int* nums, int numsSize, int k) {
    int ans = 0;
    for (int i = 0; i < numsSize; ++i) {
        if (bitCount(i) == k) {
            ans += nums[i];
        }
    }
    return ans;
}
