func sumIndicesWithKSetBits(nums []int, k int) int {
    ans := 0
    for i := 0; i < len(nums); i++ {
        if bitCount(i) == k {
            ans += nums[i]
        }
    }
    return ans
}

func bitCount(x int) int {
    x = (x & 0b0101010101) + ((x & 0b1010101010) >> 1)
    x = ((x & 0b0011001100) >> 2) + (x & 0b1100110011)
    x = (x >> 8) + ((x >> 4) & 0b1111) + (x & 0b1111)
    return x
}
