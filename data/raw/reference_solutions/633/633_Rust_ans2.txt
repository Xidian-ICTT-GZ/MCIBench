impl Solution {
    pub fn judge_square_sum(c: i32) -> bool {
        let mut left = 0 as i64;
        let mut right = (c as f64).sqrt() as i64;
        while left <= right {
            let sum = (left * left + right * right) as i64;
            if sum == c as i64 {
                return true;
            } else if sum > c as i64 {
                right -= 1;
            } else {
                left += 1;
            }
        }
        false
    }
}
