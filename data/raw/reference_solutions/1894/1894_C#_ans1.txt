public class Solution {
    public int ChalkReplacer(int[] chalk, int k) {
        int n = chalk.Length;
        long total = 0;
        foreach (int num in chalk) {
            total += num;
        }
        if (k >= total) {
            k %= (int) total;
        }
        int res = -1;
        for (int i = 0; i < n; ++i) {
            if (chalk[i] > k) {
                res = i;
                break;
            }
            k -= chalk[i];
        }
        return res;
    }
}
