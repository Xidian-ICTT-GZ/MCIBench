public class Solution {
    public int ChalkReplacer(int[] chalk, int k) {
        int n = chalk.Length;
        if (chalk[0] > k) {
            return 0;
        }
        for (int i = 1; i < n; ++i) {
            chalk[i] += chalk[i - 1];
            if (chalk[i] > k) {
                return i;
            }
        }

        k %= chalk[n - 1];
        return BinarySearch(chalk, k);
    }

    public int BinarySearch(int[] arr, int target) {
        int low = 0, high = arr.Length - 1;
        while (low < high) {
            int mid = (high - low) / 2 + low;
            if (arr[mid] <= target) {
                low = mid + 1;
            } else {
                high = mid;
            }
        }
        return low;
    }
}
