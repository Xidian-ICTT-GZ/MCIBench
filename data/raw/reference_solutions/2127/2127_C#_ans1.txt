public class Solution {
    public int MaximumInvitations(int[] favorite) {
        int n = favorite.Length;
        
        int[] indeg = new int[n];
        for (int i = 0; i < n; ++i) {
            ++indeg[favorite[i]];
        }
        bool[] used = new bool[n];
        int[] f = new int[n];
        Array.Fill(f, 1);
        Queue<int> queue = new Queue<int>();
        for (int i = 0; i < n; ++i) {
            if (indeg[i] == 0) {
                queue.Enqueue(i);
            }
        }
        while (queue.Count > 0) {
            int u = queue.Dequeue();
            used[u] = true;
            int v = favorite[u];
            
            f[v] = Math.Max(f[v], f[u] + 1);
            --indeg[v];
            if (indeg[v] == 0) {
                queue.Enqueue(v);
            }
        }
        
        
        int ring = 0, total = 0;
        for (int i = 0; i < n; ++i) {
            if (!used[i]) {
                int j = favorite[i];
                
                if (favorite[j] == i) {
                    total += f[i] + f[j];
                    used[i] = used[j] = true;
                }
                
                else {
                    int u = i, cnt = 0;
                    while (true) {
                        ++cnt;
                        u = favorite[u];
                        used[u] = true;
                        if (u == i) {
                            break;
                        }
                    }
                    ring = Math.Max(ring, cnt);
                }
            }
        }
        return Math.Max(ring, total);
    }
}
