var maxDistance = function(arrays) {
    let res = 0;
    let n = arrays.length;
    for (let i = 0; i < n - 1; i++) {
        for (let j = 0; j < arrays[i].length; j++) {
            for (let k = i + 1; k < n; k++) {
                for (let l = 0; l < arrays[k].length; l++) {
                    res = Math.max(res, Math.abs(arrays[i][j] - arrays[k][l]));
                }
            }
        }
    }
    return res;
};
