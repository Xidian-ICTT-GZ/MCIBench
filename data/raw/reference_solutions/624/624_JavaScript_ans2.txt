var maxDistance = function(arrays) {
    let res = 0;
    let n = arrays.length;
    for (let i = 0; i < n - 1; i++) {
        for (let j = i + 1; j < n; j++) {
            let array1 = arrays[i];
            let array2 = arrays[j];
            res = Math.max(res, Math.abs(array1[0] - array2[array2.length - 1]));
            res = Math.max(res, Math.abs(array2[0] - array1[array1.length - 1]));
        }
    }
    return res;
};
