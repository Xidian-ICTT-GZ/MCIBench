impl Solution {
    pub fn max_distance(arrays: Vec<Vec<i32>>) -> i32 {
        let mut res = 0;
        let n = arrays.len();
        for i in 0..n - 1 {
            for j in i + 1..n {
                let array1 = &arrays[i];
                let array2 = &arrays[j];
                res = res.max((array1[0] - array2[array2.len() - 1]).abs());
                res = res.max((array2[0] - array1[array1.len() - 1]).abs());
            }
        }
        res
    }
}
