var reverseOddLevels = function(root) {
    let q = [root];
    let isOdd = false;
    while (q.length) {
        if (isOdd) {
            const n = q.length;
            for (let i = 0; i < n / 2; i++) {
                [q[i].val, q[n - 1 - i].val] = [q[n - 1 - i].val, q[i].val];
            }
        }
        const tmp = [...q];
        q = [];
        for (const node of tmp) {
            if (node.left) {
                q.push(node.left);
            }
            if (node.right) {
                q.push(node.right);
            }
        }
        isOdd ^= true;
    }
    return root;
};
