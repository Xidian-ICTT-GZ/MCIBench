class Solution {
public:
    int numOfMinutes(int n, int headID, vector<int>& manager, vector<int>& informTime) {
        unordered_map<int, int> memo;
        function<int(int)> dfs = [&](int cur) -> int {
            if (cur == headID) {
                return 0;
            }
            if (memo.count(cur)) {
                return memo[cur];
            }
            int res = dfs(manager[cur]) + informTime[manager[cur]];
            memo[cur] = res;
            return res;
        };
        int res = 0;
        for (int i = 0; i < n; i++) {
            res = max(res, dfs(i));
        }
        return res;
    }
};
