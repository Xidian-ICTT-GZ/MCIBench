func countServers(grid [][]int) int {
    m := len(grid)
    n := len(grid[0])
    rows := make(map[int]int)
    cols := make(map[int]int)
    for i := 0; i < m; i++ {
        for j := 0; j < n; j++ {
            if grid[i][j] == 1 {
                rows[i] = rows[i] + 1
                cols[j] = cols[j] + 1
            }
        }
    }
    ans := 0
    for i := 0; i < m; i++ {
        for j := 0; j < n; j++ {
            if grid[i][j] == 1 && (rows[i] > 1 || cols[j] > 1) {
                ans++
            }
        }
    }
    return ans
}
