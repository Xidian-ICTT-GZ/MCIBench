long long countQuadruplets(int* nums, int numsSize) {
    int* pre = (int*)calloc(numsSize + 1, sizeof(int));
    long long ans = 0;

    for (int j = 0; j < numsSize; ++j) {
        int suf = 0;
        for (int k = numsSize - 1; k > j; --k) {
            if (nums[j] > nums[k]) {
                ans += (long long)pre[nums[k]] * suf;
            } else {
                ++suf;
            }
        }
        for (int x = nums[j] + 1; x <= numsSize; ++x) {
            ++pre[x];
        }
    }
    
    free(pre);
    return ans;
}
