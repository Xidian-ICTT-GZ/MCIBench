bool check(int dx, int dy, char **board, int rMove, int cMove, char color) {
    int x = rMove + dx;
    int y = cMove + dy;
    int step = 1;   
    while (x >= 0 && x < 8 && y >= 0 && y < 8) {
        if (step == 1) {
            
            if (board[x][y] == '.' || board[x][y] == color) {
                return false;
            }
        } else {
            
            if (board[x][y] == '.') {
                return false;
            }
            
            if (board[x][y] == color) {
                return true;
            }
        }
        ++step;
        x += dx;
        y += dy;
    }
    
    return false;
}

bool checkMove(char** board, int boardSize, int* boardColSize, int rMove, int cMove, char color) {
     
    int dx[] = {1, 1, 0, -1, -1, -1, 0, 1};   
    int dy[] = {0, 1, 1, 1, 0, -1, -1, -1};   
    for (int k = 0; k < 8; ++k) {
        if (check(dx[k], dy[k], board, rMove, cMove, color)) {
            return true;
        }
    }
    return false;
}
