bool isCovered(int** ranges, int rangesSize, int* rangesColSize, int left, int right) {
    int diff[52];  
    memset(diff, 0, sizeof(diff));
    for (int i = 0; i < rangesSize; i++) {
        ++diff[ranges[i][0]];
        --diff[ranges[i][1] + 1];
    }
    
    int curr = 0;
    for (int i = 1; i <= 50; ++i) {
        curr += diff[i];
        if (i >= left && i <= right && curr <= 0) {
            return false;
        }
    }
    return true;
}
