class Solution:
    def knightDialer(self, n: int) -> int:
        mod = 10**9 + 7
        moves = [
            [4, 6],
            [6, 8],
            [7, 9],
            [4, 8],
            [3, 9, 0],
            [], 
            [1, 7, 0],
            [2, 6],
            [1, 3],
            [2, 4]
        ]
        
        d = [[0] * 10 for _ in range(2)]
        d[1] = [1] * 10
        
        for i in range(2, n + 1):
            x = i & 1  
            for j in range(10):
                d[x][j] = 0
                for k in moves[j]:
                    d[x][j] = (d[x][j] + d[x ^ 1][k]) % mod
        
        return sum(d[n % 2]) % mod
