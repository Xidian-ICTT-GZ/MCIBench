public class Solution {
    const int MOD = 1000000007;

    public int KnightDialer(int n) {
        int[][] moves = {
            new int[]{4, 6},
            new int[]{6, 8},
            new int[]{7, 9},
            new int[]{4, 8},
            new int[]{3, 9, 0},
            new int[]{},
            new int[]{1, 7, 0},
            new int[]{2, 6},
            new int[]{1, 3},
            new int[]{2, 4}
        };
        int[][] d = new int[2][];
        for (int i = 0; i < 2; i++) {
            d[i] = new int[10];
        }
        Array.Fill(d[1], 1);
        for (int i = 2; i <= n; i++) {
            int x = i & 1;
            for (int j = 0; j < 10; j++) {
                d[x][j] = 0;
                foreach (int k in moves[j]) {
                    d[x][j] = (d[x][j] + d[x ^ 1][k]) % MOD;
                }
            }
        }
        int res = 0;
        foreach (int x in d[n % 2]) {
            res = (res + x) % MOD;
        }
        return res;
    }
}
