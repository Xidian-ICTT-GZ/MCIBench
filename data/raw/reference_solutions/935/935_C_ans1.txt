#define MOD 1000000007

int knightDialer(int n) {
    int moves[10][4] = {
        {4, 6, -1, -1},
        {6, 8, -1, -1},
        {7, 9, -1, -1},
        {4, 8, -1, -1},
        {3, 9, 0, -1},
        {-1, -1, -1, -1},
        {1, 7, 0, -1},
        {2, 6, -1, -1},
        {1, 3, -1, -1},
        {2, 4, -1, -1}
    };
    int d[2][10] = {0};
    for (int i = 0; i < 10; i++) {
        d[1][i] = 1;
    }
    for (int i = 2; i <= n; i++) {
        int x = i & 1;
        for (int j = 0; j < 10; j++) {
            d[x][j] = 0;
            for (int k = 0; moves[j][k] != -1; k++) {
                d[x][j] = (d[x][j] + d[x ^ 1][moves[j][k]]) % MOD;
            }
        }
    }
    int res = 0;
    for (int i = 0; i < 10; i++) {
        res = (res + d[n % 2][i]) % MOD;
    }
    return res;
}
