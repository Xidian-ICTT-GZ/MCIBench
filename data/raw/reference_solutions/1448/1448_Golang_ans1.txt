func goodNodes(root *TreeNode) int {
    return dfs(root, math.MinInt)
}

func dfs(root *TreeNode, path_max int) int {
    if root == nil {
        return 0
    }
    res := 0
    if root.Val >= path_max {
        res++
        path_max = root.Val
    }
    res += dfs(root.Left, path_max) + dfs(root.Right, path_max)
    return res
}
