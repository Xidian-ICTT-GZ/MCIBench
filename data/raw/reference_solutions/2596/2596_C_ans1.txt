bool checkValidGrid(int** grid, int gridSize, int* gridColSize) {
    if (grid[0][0] != 0) {
        return false;
    }
    int n = gridSize;
    int indices[n * n][2];
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            indices[grid[i][j]][0] = i;
            indices[grid[i][j]][1] = j;
        }
    }
    for (int i = 1; i < n * n; i++) {
        int dx = abs(indices[i][0] - indices[i - 1][0]);
        int dy = abs(indices[i][1] - indices[i - 1][1]);
        if (dx * dy != 2) {
            return false;
        }
    }
    return true;
}
