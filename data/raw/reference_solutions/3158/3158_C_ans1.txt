#define MAX_NUM 64

int duplicateNumbersXOR(int* nums, int numsSize) {
    int cnt[MAX_NUM] = {0};
    int res = 0;
    for (int i = 0; i < numsSize; ++i) {
        if (cnt[nums[i]] == 1) {
            res ^= nums[i];
        } else {
            cnt[nums[i]] = 1;
        }
    }
    return res;
}
