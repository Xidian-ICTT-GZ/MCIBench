public class Solution {
    public int DuplicateNumbersXOR(int[] nums) {
        HashSet<int> cnt = new HashSet<int>();
        int res = 0;
        foreach (int num in nums) {
            if (cnt.Contains(num)) {
                res ^= num;
            } else {
                cnt.Add(num);
            }
        }
        return res;
    }
}
