public class Solution {
    public long PickGifts(int[] gifts, int k) {
        PriorityQueue<int, int> pq = new PriorityQueue<int, int>();
        foreach (int gift in gifts) {
            pq.Enqueue(gift, -gift);
        }
        while (k > 0) {
            k--;
            int x = pq.Dequeue();
            pq.Enqueue((int) Math.Sqrt(x), -(int) Math.Sqrt(x));
        }
        long res = 0;
        while (pq.Count > 0) {
            res += pq.Dequeue();
        }
        return res;
    }
}
