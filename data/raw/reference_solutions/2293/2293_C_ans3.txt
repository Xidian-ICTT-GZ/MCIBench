#define MAX(a, b) ((a) > (b) ? (a) : (b))
#define MIN(a, b) ((a) < (b) ? (a) : (b))

int minMaxGame(int* nums, int numsSize) {
    while (numsSize != 1) {
        numsSize /= 2;
        for (int i = 0; i < numsSize; i++) {
            if (i % 2 == 0) {
                nums[i] = MIN(nums[2 * i], nums[2 * i + 1]);
            } else {
                nums[i] = MAX(nums[2 * i], nums[2 * i + 1]);
            }
        }
    }
    return nums[0];
}
