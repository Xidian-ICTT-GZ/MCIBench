public class Solution {
    public int MinMaxGame(int[] nums) {
        int n = nums.Length;
        while (n != 1) {
            int m = n / 2;
            for (int i = 0; i < m; i++) {
                if (i % 2 == 0) {
                    nums[i] = Math.Min(nums[2 * i], nums[2 * i + 1]);
                } else {
                    nums[i] = Math.Max(nums[2 * i], nums[2 * i + 1]);
                }
            }
            n = m;
        }
        return nums[0];
    }
}
