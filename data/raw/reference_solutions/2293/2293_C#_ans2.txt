public class Solution {
    public int MinMaxGame(int[] nums) {
        int n = nums.Length;
        while (n != 1) {
            int[] newNums = new int[n / 2];
            for (int i = 0; i < newNums.Length; i++) {
                if (i % 2 == 0) {
                    newNums[i] = Math.Min(nums[2 * i], nums[2 * i + 1]);
                } else {
                    newNums[i] = Math.Max(nums[2 * i], nums[2 * i + 1]);
                }
            }
            nums = newNums;
            n /= 2;
        }
        return nums[0];
    }
}
