static int cmp(const void *a, const void *b) {
    return *(char *)a - *(char *)b;
}

int splitNum(int num) {
    char stnum[16];
    sprintf(stnum, "%d", num);
    int len = strlen(stnum);
    qsort(stnum, len, sizeof(char), cmp);
    int num1 = 0, num2 = 0;
    for (int i = 0; i < len; ++i) {
        if (i % 2 == 0) {
            num1 = num1 * 10 + (stnum[i] - '0');
        } else {
            num2 = num2 * 10 + (stnum[i] - '0');
        }
    }
    return num1 + num2;
}
