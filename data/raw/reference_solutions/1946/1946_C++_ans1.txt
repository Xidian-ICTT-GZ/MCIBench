class Solution {
public:
    string maximumNumber(string num, vector<int>& change) {
        int n = num.size();
        for (int i = 0; i < n; ++i){
            
            if (change[num[i]-'0'] > num[i] - '0'){
                
                while (i < n && change[num[i]-'0'] >= num[i] - '0'){
                    num[i] = '0' + change[num[i]-'0'];
                    ++i;
                }
                break;
            }
        }
        return num;
    }
};
