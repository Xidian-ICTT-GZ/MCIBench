use std::collections::{VecDeque, HashMap};

impl Solution {
    pub fn oranges_rotting(grid: Vec<Vec<i32>>) -> i32 {
        let mut grid = grid.clone();
        let (R, C) = (grid.len(), grid[0].len());
        const DR: [i32; 4] = [-1, 0, 1, 0];
        const DC: [i32; 4] = [0, -1, 0, 1];
        let mut queue = VecDeque::new();
        let mut depth = HashMap::new();
        
        for r in 0..R {
            for c in 0..C {
                if grid[r][c] == 2 {
                    let code = r * C + c;
                    queue.push_back(code);
                    depth.insert(code, 0);
                }
            }
        }
        
        let mut ans = 0;
        while let Some(code) = queue.pop_front() {
            let r = code / C;
            let c = code % C;
            for k in 0..4 {
                let nr = r as i32 + DR[k];
                let nc = c as i32 + DC[k];
                if nr >= 0 && nr < R as i32 && nc >= 0 && nc < C as i32 {
                    let (ur, uc) = (nr as usize, nc as usize);
                    if grid[ur][uc] == 1 {
                        grid[ur][uc] = 2;
                        let ncode = ur * C + uc;
                        queue.push_back(ncode);
                        depth.insert(ncode, *depth.get(&code).unwrap() + 1);
                        ans = *depth.get(&ncode).unwrap();
                    }
                }
            }
        }
        
        for row in grid {
            for v in row {
                if v == 1 {
                    return -1;
                }
            }
        }
        ans
    }
}
