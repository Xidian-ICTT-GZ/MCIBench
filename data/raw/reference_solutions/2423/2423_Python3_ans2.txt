class Solution:
    def equalFrequency(self, word: str) -> bool:
        charCount = [0] * 26
        for c in word:
            charCount[ord(c) - ord('a')] += 1
        freqCount = Counter(c for c in charCount if c > 0)
        for c in charCount:
            if c == 0: continue
            freqCount[c] -= 1
            if freqCount[c] == 0:
                del freqCount[c]
            if c - 1 > 0:
                freqCount[c - 1] += 1
            if len(freqCount) == 1:
                return True
            if c - 1 > 0:
                freqCount[c - 1] -= 1
            if freqCount[c - 1] == 0:
                del freqCount[c - 1]
            freqCount[c] += 1
        return False
