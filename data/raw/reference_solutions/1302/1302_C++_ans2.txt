class Solution {
public:
    int deepestLeavesSum(TreeNode* root) {
        int sum = 0;
        queue<TreeNode*> qu;
        qu.emplace(root);
        while (!qu.empty()) {
            sum = 0;
            int size = qu.size();
            for (int i = 0; i < size; i++) {
                TreeNode *node = qu.front();
                qu.pop();
                sum += node->val;
                if (node->left != nullptr) {
                    qu.emplace(node->left);
                }
                if (node->right != nullptr) {
                    qu.emplace(node->right);
                }
            }
        }
        return sum;
    }
};
