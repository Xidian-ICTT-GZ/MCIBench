bool isMonotonic(int* nums, int numsSize) {
    bool inc = true, dec = true;
    int n = numsSize;
    for (int i = 0; i < n - 1; ++i) {
        if (nums[i] > nums[i + 1]) {
            inc = false;
        }
        if (nums[i] < nums[i + 1]) {
            dec = false;
        }
    }
    return inc || dec;
}
