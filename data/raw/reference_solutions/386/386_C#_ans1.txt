public class Solution {
    public IList<int> LexicalOrder(int n) {
        IList<int> ret = new List<int>();
        int number = 1;
        for (int i = 0; i < n; i++) {
            ret.Add(number);
            if (number * 10 <= n) {
                number *= 10;
            } else {
                while (number % 10 == 9 || number + 1 > n) {
                    number /= 10;
                }
                number++;
            }
        }
        return ret;
    }
}
