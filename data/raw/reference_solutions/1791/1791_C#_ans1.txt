public class Solution {
    public int FindCenter(int[][] edges) {
        int n = edges.Length + 1;
        int[] degrees = new int[n + 1];
        foreach (int[] edge in edges) {
            degrees[edge[0]]++;
            degrees[edge[1]]++;
        }
        for (int i = 1; ; i++) {
            if (degrees[i] == n - 1) {
                return i;
            }
        }
    }
}
