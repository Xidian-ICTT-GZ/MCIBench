var medianOfUniquenessArray = function(nums) {
    const n = nums.length;
    const median = Math.floor((n * (n + 1) / 2 + 1) / 2);

    
    function check(t) {
        const cnt = new Map();
        let tot = 0;
        for (let i = 0, j = 0; i < n; i++) {
            cnt.set(nums[i], (cnt.get(nums[i]) || 0) + 1);
            while (cnt.size > t) {
                cnt.set(nums[j], cnt.get(nums[j]) - 1);
                if (cnt.get(nums[j]) === 0) {
                    cnt.delete(nums[j]);
                }
                j++;
            }
            tot += i - j + 1;
        }
        return tot >= median;
    }

    let res = 0;
    let lo = 1, hi = n;
    while (lo <= hi) {
        const mid = Math.floor((lo + hi) / 2);
        if (check(mid)) {
            res = mid;
            hi = mid - 1;
        } else {
            lo = mid + 1;
        }
    }
    
    return res;
};
