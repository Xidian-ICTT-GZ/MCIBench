int* mostCompetitive(int* nums, int numsSize, int k, int* returnSize) {
    int n = numsSize;
    int *res = (int *)malloc(sizeof(int) * n);
    *returnSize = k;
    int pos = 0;
    for (int i = 0; i < n; i++) {
        while (pos > 0 && n - i + pos > k && res[pos - 1] > nums[i]) {
            pos--;
        }
        res[pos++] = nums[i];
    }
    return res;
}
