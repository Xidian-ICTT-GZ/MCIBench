class Solution:
    def decodeCiphertext(self, encodedText: str, rows: int) -> str:
        cols = len(encodedText) // rows   
        res = []   
        for i in range(cols):
            
            r = 0
            c = i
            while r < rows and c < cols:
                res.append(encodedText[r*cols+c])
                r += 1
                c += 1
        
        while res and res[-1] == ' ':
            res.pop()
        return "".join(res)
