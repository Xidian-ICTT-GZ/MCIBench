var edgeScore = function(edges) {
    const n = edges.length;
    const points = new Array(n).fill(0);
    for (let i = 0; i < n; i++) {
        points[edges[i]] += i;
    }
    let maxPoints = -1;
    let res = -1;
    for (let i = 0; i < n; i++) {
        if (points[i] > maxPoints) {
            maxPoints = points[i];
            res = i;
        }
    }
    return res;
};
