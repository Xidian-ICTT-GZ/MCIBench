class Solution:
    def uniquePathsIII(self, grid: List[List[int]]) -> int:
        r, c = len(grid), len(grid[0])
        si, sj, st = 0, 0, 0
        for i in range(r):
            for j in range(c):
                if grid[i][j] in [0, 2]:
                    st |= (1 << (i * c + j))
                elif grid[i][j] == 1:
                    si, sj = i, j

        @lru_cache(None)
        def dp(i: int, j: int, st: int) -> int:
            if grid[i][j] == 2:
                if st == 0:
                    return 1
                return 0
            res = 0
            for di, dj in [[-1, 0], [1, 0], [0, -1], [0, 1]]:
                ni, nj = i + di, j + dj
                if 0 <= ni < r and 0 <= nj < c and (st & (1 << (ni * c + nj))) > 0:
                    res += dp(ni, nj, st ^ (1 << (ni * c + nj)))
            return res

        return dp(si, sj, st)
