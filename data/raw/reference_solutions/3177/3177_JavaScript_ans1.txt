var maximumLength = function(nums, k) {
    const dp = new Map();
    const zd = Array(k + 1).fill(0);
    nums.forEach(v => {
        if (!dp.has(v)) {
            dp.set(v, Array(k + 1).fill(0));
        }

        const tmp = dp.get(v);
        for (let j = 0; j <= k; j++) {
            tmp[j]++;
            if (j > 0) {
                tmp[j] = Math.max(tmp[j], zd[j - 1] + 1);
            }
        }

        for (let j = 0; j <= k; j++) {
            zd[j] = Math.max(zd[j], tmp[j]);
        }
    });

    return zd[k];
};
