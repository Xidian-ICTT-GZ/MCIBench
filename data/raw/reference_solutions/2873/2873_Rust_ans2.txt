impl Solution {
    pub fn maximum_triplet_value(nums: Vec<i32>) -> i64 {
        let n = nums.len();
        let mut res = 0;
        for k in 2..n {
            let mut m = nums[0];
            for j in 1..k {
                res = res.max((m - nums[j]) as i64 * nums[k] as i64);
                m = m.max(nums[j]);
            }
        }
        res
    }
}
