class Solution:
    def maximumTripletValue(self, nums: List[int]) -> int:
        n = len(nums)
        res = 0
        for k in range(2, n):
            m = nums[0]
            for j in range(1, k):
                res = max(res, (m - nums[j]) * nums[k])
                m = max(m, nums[j])
        return res
