class Solution:
    def minimizedMaximum(self, n: int, quantities: List[int]) -> int:
        
        def check(x: int) -> bool:
            
            cnt = 0
            for q in quantities:
                cnt += (q - 1) // x + 1
            return cnt <= n
        
        l, r = 1, max(quantities) + 1
        
        while l < r:
            mid = l + (r - l) // 2
            if check(mid):
                r = mid
            else:
                l = mid + 1
        return l
