double average(int* salary, int salarySize) {
    double maxValue = INT_MIN;
    double minValue = INT_MAX;
    double sum = 0;

    for (int i = 0; i < salarySize; ++i) {
        if (salary[i] > maxValue) {
            maxValue = salary[i];
        }
        if (salary[i] < minValue) {
            minValue = salary[i];
        }
        sum += salary[i];
    }
    sum -= maxValue + minValue;
    return sum / (salarySize - 2);
}
