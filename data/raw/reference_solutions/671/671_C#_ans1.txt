public class Solution {
    int ans;
    int rootvalue;

    public int FindSecondMinimumValue(TreeNode root) {
        ans = -1;
        rootvalue = root.val;
        DFS(root);
        return ans;
    }

    public void DFS(TreeNode node) {
        if (node == null) {
            return;
        }
        if (ans != -1 && node.val >= ans) {
            return;
        }
        if (node.val > rootvalue) {
            ans = node.val;
        }
        DFS(node.left);
        DFS(node.right);
    }
}
