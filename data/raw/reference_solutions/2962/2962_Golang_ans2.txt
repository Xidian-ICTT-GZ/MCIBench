func countSubarrays(nums []int, k int) int64 {
	mx := nums[0]
	for _, v := range nums {
		if v > mx {
			mx = v
		}
	}
	var ans int64 = 0
	cnt := 0
	left := 0
	for _, x := range nums {
		if x == mx {
			cnt++
		}
		for cnt == k {
			if nums[left] == mx {
				cnt--
			}
			left++
		}
		ans += int64(left)
	}
	return ans
}
