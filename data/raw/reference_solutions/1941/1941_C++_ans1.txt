class Solution {
public:
    bool areOccurrencesEqual(string s) {
        unordered_map<char, int> freq;   
        for (const char ch : s){
            if (!freq.count(ch)){
                freq[ch] = 0;
            }
            ++freq[ch];
        }
        int occ = s.size() / freq.size();   
        for (auto&& [_, v] : freq){
            if (v != occ){
                return false;
            }
        }
        return true;
    }
};
