class Solution {
public:
    int longestEqualSubarray(vector<int>& nums, int k) {
        int n = nums.size();
        int ans = 0;
        unordered_map<int, int> cnt;
        for (int i = 0, j = 0; j < n; j++) {
            cnt[nums[j]]++;
            
            while (j - i + 1 - cnt[nums[i]] > k) {
                cnt[nums[i]]--;
                i++;
            }
            ans = max(ans, cnt[nums[j]]);
        }
        return ans;
    }
};
