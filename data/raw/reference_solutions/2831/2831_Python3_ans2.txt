class Solution:
    def longestEqualSubarray(self, nums: List[int], k: int) -> int:
        ans = 0
        cnt = defaultdict(int)
        i = 0
        for j, x in enumerate(nums):
            cnt[x] += 1
            
            while j - i + 1 - cnt[nums[i]] > k:
                cnt[nums[i]] -= 1
                i += 1
            ans = max(ans, cnt[nums[j]])
        return ans
