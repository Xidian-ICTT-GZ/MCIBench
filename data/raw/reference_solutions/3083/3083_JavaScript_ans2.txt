var isSubstringPresent = function(s) {
    let h = new Array(26).fill(0);
    for (let i = 0; i + 1 < s.length; i++) {
        let x = s.charCodeAt(i) - 'a'.charCodeAt(0);
        let y = s.charCodeAt(i + 1) - 'a'.charCodeAt(0);
        h[x] |= (1 << y);
        if ((h[y] >> x) & 1) {
            return true;
        }
    }
    return false;
};
