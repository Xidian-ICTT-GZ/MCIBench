bool isSubstringPresent(char* s) {
    int h[26] = {0};
    int len = strlen(s);
    for (int i = 0; i + 1 < len; i++) {
        int x = s[i] - 'a';
        int y = s[i + 1] - 'a';
        h[x] |= (1 << y);
        if ((h[y] >> x) & 1) {
            return true;
        }
    }
    return false;
}
