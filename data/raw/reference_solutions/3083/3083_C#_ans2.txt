public class Solution {
    public bool IsSubstringPresent(string s) {
        int[] h = new int[26];
        for (int i = 0; i + 1 < s.Length; i++) {
            int x = s[i] - 'a';
            int y = s[i + 1] - 'a';
            h[x] |= 1 << y;
            if ((h[y] >> x & 1) != 0) {
                return true;
            }
        }
        return false;
    }
}
