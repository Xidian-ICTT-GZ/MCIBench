impl Solution {
    pub fn get_row(row_index: i32) -> Vec<i32> {
        let mut pre: Vec<i32> = Vec::new();
        let mut cur: Vec<i32> = Vec::new();
        for i in 0..=row_index {
            cur = vec![1; (i + 1) as usize];
            for j in 1..i {
                cur[j as usize] = pre[(j - 1) as usize] + pre[j as usize];
            }
            pre = cur.clone();
        }
        pre
    }
}
