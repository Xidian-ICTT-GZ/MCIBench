public class Solution {
    public IList<int> GetRow(int rowIndex) {
        List<int> row = new List<int>(new int[rowIndex + 1]);
        row[0] = 1;
        for (int i = 1; i <= rowIndex; ++i) {
            row[i] = (int)((long)row[i - 1] * (rowIndex - i + 1) / i);
        }
        return row;
    }
}
