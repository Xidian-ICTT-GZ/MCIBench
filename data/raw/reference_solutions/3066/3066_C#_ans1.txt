public class Solution {
    public int MinOperations(int[] nums, int k) {
        int res = 0;
        PriorityQueue<long, long> pq = new PriorityQueue<long, long>();
        foreach (int num in nums) {
            pq.Enqueue(num, num);
        }
        while (pq.Peek() < k) {
            long x = pq.Dequeue(), y = pq.Dequeue();
            pq.Enqueue(x + x + y, x + x + y);
            res++;
        }
        return res;
    }
}
