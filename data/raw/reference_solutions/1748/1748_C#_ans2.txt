public class Solution {
    public int SumOfUnique(int[] nums) {
        int ans = 0;
        Dictionary<int, int> state = new Dictionary<int, int>();
        foreach (int num in nums) {
            if (!state.ContainsKey(num)) {
                ans += num;
                state.Add(num, 1);
            } else if (state[num] == 1) {
                ans -= num;
                state[num] = 2;
            }
        }
        return ans;
    }
}
