class Solution {
public:
    int minOperations(vector<int>& nums) {
        int ret = 0, maxn = 0;
        for (auto num : nums) {
            maxn = max(maxn, num);
            ret += __builtin_popcount(num);
        }
        if (maxn > 0) {
            ret += 31 - __builtin_clz(maxn);
        }
        return ret;
    }
};
