#define MAXN 500

int countQuadruplets(int* nums, int numsSize){
    int ans = 0;
    int * cnt = (int *)malloc(sizeof(int) * MAXN);
    memset(cnt, 0, sizeof(int) * MAXN);
    for (int c = numsSize - 2; c >= 2; --c) {
        cnt[nums[c + 1]]++;
        for (int a = 0; a < c; ++a) {
            for (int b = a + 1; b < c; ++b) {
                ans += cnt[nums[a] + nums[b] + nums[c]];
            }
        }
    }
    free(cnt);
    return ans;
}
