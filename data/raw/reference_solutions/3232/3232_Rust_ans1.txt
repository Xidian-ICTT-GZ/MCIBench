impl Solution {
    pub fn can_alice_win(nums: Vec<i32>) -> bool {
        let mut single_digit_sum : i32 = 0;
        let mut double_digit_sum : i32 = 0;
        for &num in &nums {
            if num < 10 {
                single_digit_sum += num;
            } else {
                double_digit_sum += num;
            }
        }
        (single_digit_sum != double_digit_sum) as bool
    }
}
