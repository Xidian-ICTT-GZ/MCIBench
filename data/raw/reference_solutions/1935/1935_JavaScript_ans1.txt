var canBeTypedWords = function(text, brokenLetters) {
    const broken = new Set();   
    for (const ch of brokenLetters) {
        broken.add(ch);
    }
    let res = 0;   
    let flag = true;   
    for (const ch of text) {
        if (ch === ' ') {
            
            if (flag) {
                ++res;
            }
            flag = true;
        } else if (broken.has(ch)) {
            
            flag = false;
        }
    }
    
    if (flag) {
        ++res;
    }
    return res;
};
