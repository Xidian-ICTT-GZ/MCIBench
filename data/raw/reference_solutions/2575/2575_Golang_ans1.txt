func divisibilityArray(word string, m int) []int {
    res := make([]int, 0)
    cur := 0
    for _, c := range word {
        cur = (cur * 10 + int(c - '0')) % m
        if cur == 0 {
            res = append(res, 1)
        } else {
            res = append(res, 0)
        }
    }
    return res
}
