public class Solution {
    public int[] DivisibilityArray(string word, int m) {
        int[] res = new int[word.Length];
        long cur = 0;
        for (int i = 0; i < word.Length; i++) {
            char c = word[i];
            cur = (cur * 10 + (c - '0')) % m;
            res[i] = (cur == 0) ? 1 : 0;
        }
        return res;
    }
}
