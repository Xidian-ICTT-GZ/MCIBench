impl Solution {
    pub fn count_fair_pairs(mut nums: Vec<i32>, lower: i32, upper: i32) -> i64 {
        nums.sort();
        let mut ans = 0;
        for i in 0..nums.len() {
            let l = nums[0..i].partition_point(|num| * num + nums[i] < lower);
            let r = nums[0..i].partition_point(|num| * num + nums[i] <= upper );
            ans += r - l
        }
        ans as i64
    }
}
