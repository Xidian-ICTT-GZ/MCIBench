var longestContinuousSubstring = function(s) {
    let res = 1;
    let cur = 1;
    for (let i = 1; i < s.length; i++) {
        if (s[i] == String.fromCharCode(s.charCodeAt(i - 1) + 1)) {
            cur++;
        } else {
            cur = 1;
        }
        res = Math.max(res, cur);
    }
    return res;
};
