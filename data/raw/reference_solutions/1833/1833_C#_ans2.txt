public class Solution {
    public int MaxIceCream(int[] costs, int coins) {
        int[] freq = new int[100001];
        foreach (int cost in costs) {
            freq[cost]++;
        }
        int count = 0;
        for (int i = 1; i <= 100000; i++) {
            if (coins >= i) {
                int curCount = Math.Min(freq[i], coins / i);
                count += curCount;
                coins -= i * curCount;
            } else {
                break;
            }
        }
        return count;
    }
}
