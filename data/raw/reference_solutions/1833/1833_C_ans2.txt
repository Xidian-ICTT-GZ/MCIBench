int maxIceCream(int *costs, int costsSize, int coins) {
    int freq[100001];
    memset(freq, 0, sizeof(freq));
    for (int i = 0; i < costsSize; i++) {
        freq[costs[i]]++;
    }
    int count = 0;
    for (int i = 1; i <= 100000; i++) {
        if (coins >= i) {
            int curCount = fmin(freq[i], coins / i);
            count += curCount;
            coins -= i * curCount;
        } else {
            break;
        }
    }
    return count;
}
