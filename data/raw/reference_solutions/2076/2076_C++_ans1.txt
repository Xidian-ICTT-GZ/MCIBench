class UnionFind {
public:
    vector<int> parent;
    vector<int> size;
    int n;
    
    int setCount;
    
public:
    UnionFind(int _n): n(_n), setCount(_n), parent(_n), size(_n, 1) {
        iota(parent.begin(), parent.end(), 0);
    }
    
    int findset(int x) {
        return parent[x] == x ? x : parent[x] = findset(parent[x]);
    }
    
    bool unite(int x, int y) {
        x = findset(x);
        y = findset(y);
        if (x == y) {
            return false;
        }
        if (size[x] < size[y]) {
            swap(x, y);
        }
        parent[y] = x;
        size[x] += size[y];
        --setCount;
        return true;
    }
    
    bool connected(int x, int y) {
        x = findset(x);
        y = findset(y);
        return x == y;
    }
};

class Solution {
public:
    vector<bool> friendRequests(int n, vector<vector<int>>& restrictions, vector<vector<int>>& requests) {
        UnionFind uf(n);
        vector<bool> ans;
        for (const auto& req: requests) {
            int x = uf.findset(req[0]), y = uf.findset(req[1]);
            if (x != y) {
                bool check = true;
                for (const auto& res: restrictions) {
                    int u = uf.findset(res[0]), v = uf.findset(res[1]);
                    if ((x == u && y == v) || (x == v && y == u)) {
                        check = false;
                        break;
                    }
                }
                if (check) {
                    ans.push_back(true);
                    uf.unite(x, y);
                }
                else {
                    ans.push_back(false);
                }
            }
            else {
                ans.push_back(true);
            }
        }
        return ans;
    }
};
