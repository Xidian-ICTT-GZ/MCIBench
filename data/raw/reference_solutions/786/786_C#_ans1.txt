public class Solution {
    public int[] KthSmallestPrimeFraction(int[] arr, int k) {
        int n = arr.Length;
        List<int[]> frac = new List<int[]>();
        for (int i = 0; i < n; ++i) {
            for (int j = i + 1; j < n; ++j) {
                frac.Add(new int[]{arr[i], arr[j]});
            }
        }
        frac.Sort((x, y) => x[0] * y[1] - y[0] * x[1]);
        return frac[k - 1];
    }
}
