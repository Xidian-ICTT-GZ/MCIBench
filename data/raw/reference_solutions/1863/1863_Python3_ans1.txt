class Solution:
    def subsetXORSum(self, nums: List[int]) -> int:
        res = 0
        n = len(nums)
        def dfs(val, idx):
            nonlocal res
            if idx == n:
                
                res += val
                return
            
            dfs(val ^ nums[idx], idx + 1)
            
            dfs(val, idx + 1)
        
        dfs(0, 0)
        return res
