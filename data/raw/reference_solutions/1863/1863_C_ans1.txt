int dfs(int val, int idx, int* nums, int numsSize) {
    if (idx == numsSize) {
        
        return val;
    }
    
    return dfs(val ^ nums[idx], idx + 1, nums, numsSize) + dfs(val, idx + 1, nums, numsSize);
}

int subsetXORSum(int* nums, int numsSize) {
    return dfs(0, 0, nums, numsSize);
}
