class Solution {
    public int subsetXORSum(int[] nums) {
        int res = 0, n = nums.length;
        for (int i = 0; i < (1 << n); i++) {
            int tmp = 0;
            for (int j = 0; j < n; j++) {
                if ((i & (1 << j)) != 0) {
                    tmp ^= nums[j];
                }
            }
            res += tmp;
        }
        return res;
    }
}
