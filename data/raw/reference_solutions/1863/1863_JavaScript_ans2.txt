var subsetXORSum = function(nums) {
    let res = 0;
    let n = nums.length;
    for (let i = 0; i < (1 << n); i++) {
        let tmp = 0;
        for (let j = 0; j < n; j++) {
            if (i & (1 << j)) {
                tmp ^= nums[j];
            }
        }
        res += tmp;
    }
    return res;
};
