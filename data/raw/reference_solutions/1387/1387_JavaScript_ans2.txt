const f = new Map();

function getF(x) {
    if (f.has(x)) {
        return f.get(x);
    }
    if (x === 1) {
        return f.set(x, 0).get(x);
    }
    if (x & 1) {
        return f.set(x, getF(x * 3 + 1) + 1).get(x);
    }
    return f.set(x, getF(x / 2) + 1).get(x);
}

var getKth = function(lo, hi, k) {
    let v = [];
    for (let i = lo; i <= hi; i++) {
        v.push(i);
    }
    v.sort((u, v) => {
        let f1 = getF(u);
        let f2 = getF(v);
        if (f1 !== f2) {
            return f1 - f2;
        }
        return u - v;
    });
    return v[k - 1];
};
