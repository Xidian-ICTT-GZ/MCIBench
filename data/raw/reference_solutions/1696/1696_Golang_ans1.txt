func maxResult(nums []int, k int) int {
    n := len(nums)
    dp := make([]int, n)
    dp[0] = nums[0]
    queue := make([]int, n) 
    qi, qj := 0, 1
    for i := 1; i < n; i++ {
        for qi < qj && queue[qi] < i - k {
            qi++
        }
        dp[i] = dp[queue[qi]] + nums[i]
        for qi < qj && dp[queue[qj - 1]] <= dp[i] {
            qj--
        }
        queue[qj] = i
        qj++
    }
    return dp[n - 1]
}
