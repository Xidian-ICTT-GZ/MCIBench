int* findBall(int** grid, int gridSize, int* gridColSize, int* returnSize) {
    int n = gridColSize[0];
    int * ans = (int *)malloc(sizeof(int) * n);
    for (int j = 0; j < n; ++j) {
        int col = j; 
        for (int i = 0; i < gridSize; i++) {
            int dir = grid[i][col];
            col += dir; 
            if (col < 0 || col == n || grid[i][col] != dir) { 
                col = -1;
                break;
            }
        }
        ans[j] = col; 
    }
    *returnSize = n; 
    return ans;
}
