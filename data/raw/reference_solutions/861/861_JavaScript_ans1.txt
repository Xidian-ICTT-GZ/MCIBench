var matrixScore = function(grid) {
    const m = grid.length, n = grid[0].length;
    let ret = m * (1 << (n - 1));

    for (let j = 1; j < n; j++) {
        let nOnes = 0;
        for (let i = 0; i < m; i++) {
            if (grid[i][0] === 1) {
                nOnes += grid[i][j];
            } else {
                nOnes += (1 - grid[i][j]); 
            }
        }
        const k = Math.max(nOnes, m - nOnes);
        ret += k * (1 << (n - j - 1));
    }
    return ret;
};
