bool PredictTheWinner(int* nums, int numsSize) {
    int dp[numsSize];
    for (int i = 0; i < numsSize; i++) {
        dp[i] = nums[i];
    }
    for (int i = numsSize - 2; i >= 0; i--) {
        for (int j = i + 1; j < numsSize; j++) {
            dp[j] = fmax(nums[i] - dp[j], nums[j] - dp[j - 1]);
        }
    }
    return dp[numsSize - 1] >= 0;
}
