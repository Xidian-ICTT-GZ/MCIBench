int maxScore(int** grid, int gridSize, int* gridColSize) {
    int m = gridSize, n = gridColSize[0];
    int premin[m][n];
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            premin[i][j] = INT_MAX;
        }
    }
    int ans = INT_MIN;

    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            int pre = INT_MAX;
            if (i > 0) {
                pre = fmin(pre, premin[i - 1][j]);
            }
            if (j > 0) {
                pre = fmin(pre, premin[i][j - 1]);
            }
            
            if (i + j > 0) {
                ans = fmax(ans, grid[i][j] - pre);
            }
            premin[i][j] = fmin(pre, grid[i][j]);
        }
    }

    return ans;
}
