var findMaximumXOR = function(nums) {
    const HIGH_BIT = 30;
    let x = 0;
    for (let k = HIGH_BIT; k >= 0; --k) {
        const seen = new Set();
        
        for (const num of nums) {
            
            
            seen.add(num >> k);
        }

        
        
        const xNext = x * 2 + 1;
        let found = false;
        
        
        for (const num of nums) {
            if (seen.has(xNext ^ (num >> k))) {
                found = true;
                break;
            }
        }

        if (found) {
            x = xNext;
        } else {
            
            
            x = xNext - 1;
        }
    }
    return x; 
};
