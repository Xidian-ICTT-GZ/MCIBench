int compare(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

int countWays(int* nums, int numsSize){
    int res = 0;
    qsort(nums, numsSize, sizeof(int), compare);
    for (int k = 0; k <= numsSize; k++) {
        
        if (k > 0 && nums[k - 1] >= k) {
            continue;
        }
        
        if (k < numsSize && nums[k] <= k) {
            continue;
        }
        res++;
    }
    return res; 
}
