func max(a, b int) int {
    if a > b {
        return a
    }
    return b
}

func findMaxK(nums []int) int {
    k, s := -1, map[int]struct{}{}
    for _, x := range nums {
        s[x] = struct{}{}
    }
    for _, x := range nums {
        if _, ok := s[-x]; ok {
            k = max(x, k)
        }
    }
    return k
}
