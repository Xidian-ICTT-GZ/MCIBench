static int cmp(const void *pa, const void *pb) {
    return *(int *)pa - *(int *)pb;
}

int findMaxK(int* nums, int numsSize) {
    qsort(nums, numsSize, sizeof(int), cmp);
    for (int i = 0, j = numsSize - 1; i < j; j--) {
        while (i < j && nums[i] < -nums[j]) {
            i++;
        }
        if (nums[i] == -nums[j]) {
            return nums[j];
        }
    }
    return -1;
}
