class Solution:
    def findMaxK(self, nums: List[int]) -> int:
        nums.sort()
        i, j = 0, len(nums) - 1
        while i < j:
            while i < j and nums[i] < -nums[j]:
                i += 1
            if nums[i] == -nums[j]:
                return nums[j]
            j -= 1
        return -1
