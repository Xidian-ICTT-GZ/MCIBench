var numberOfRightTriangles = function(grid) {
    const n = grid.length;
    const m = grid[0].length;
    const col = Array(m).fill(0);

    for (let i = 0; i < n; i++) {
        for (let j = 0; j < m; j++) {
            col[j] += grid[i][j];
        }
    }

    let res = 0;
    for (let i = 0; i < n; i++) {
        let row = grid[i].reduce((a, b) => a + b, 0);
        for (let j = 0; j < m; j++) {
            if (grid[i][j] === 1) {
                res +=(row - 1) * (col[j] - 1);
            }
        }
    }

    return res;
};
