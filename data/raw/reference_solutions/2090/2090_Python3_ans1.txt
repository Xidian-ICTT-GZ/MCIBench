class Solution:
    def getAverages(self, nums: List[int], k: int) -> List[int]:
        n = len(nums)
        ans = [-1] * n
        if k * 2 + 1 <= n:
            total = sum(nums[:k * 2 + 1])
            for i in range(k, n - k):
                if i != k:
                    total += nums[i + k] - nums[i - k - 1]
                ans[i] = total // (k * 2 + 1)
        return ans
