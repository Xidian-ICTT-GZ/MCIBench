public class Solution {
    public int[] DistinctDifferenceArray(int[] nums) {
        int n = nums.Length;
        ISet<int> set = new HashSet<int>();
        int[] sufCnt = new int[n + 1];
        for (int i = n - 1; i > 0; i--) {
            set.Add(nums[i]);
            sufCnt[i] = set.Count;
        }
        int[] res = new int[n];
        set.Clear();
        for (int i = 0; i < n; i++) {
            set.Add(nums[i]);
            res[i] = set.Count - sufCnt[i + 1];
        }
        return res;
    }
}
