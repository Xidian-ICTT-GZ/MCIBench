var generateKey = function(num1, num2, num3) {
    let key = 0;
    for (let p = 1; num1 > 0 && num2 > 0 && num3 > 0; p *= 10) {
        key += Math.min(num1 % 10, num2 % 10, num3 % 10) * p;
        num1 = Math.floor(num1 / 10);
        num2 = Math.floor(num2 / 10);
        num3 = Math.floor(num3 / 10);
    }
    return key;
};
