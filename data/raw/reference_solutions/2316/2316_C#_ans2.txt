public class Solution {
    public long CountPairs(int n, int[][] edges) {
        IList<int>[] graph = new IList<int>[n];
        for (int i = 0; i < n; i++) {
            graph[i] = new List<int>();
        }
        foreach (int[] edge in edges) {
            graph[edge[0]].Add(edge[1]);
            graph[edge[1]].Add(edge[0]);
        }

        bool[] visited = new bool[n];

        long res = 0;
        for (int i = 0; i < n; i++) {
            if (!visited[i]) {
                long count = DFS(i, graph, visited);
                res += count * (n - count);
            }
        }
        return res / 2;
    }

    public int DFS(int x, IList<int>[] graph, bool[] visited) {
        visited[x] = true;
        int count = 1;
        foreach (int y in graph[x]) {
            if (!visited[y]) {
                count += DFS(y, graph, visited);
            }
        }
        return count;
    }
}
