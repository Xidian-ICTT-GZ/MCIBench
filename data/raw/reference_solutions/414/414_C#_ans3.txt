public class Solution {
    public int ThirdMax(int[] nums) {
        long a = long.MinValue, b = long.MinValue, c = long.MinValue;
        foreach (long num in nums) {
            if (num > a) {
                c = b;
                b = a;
                a = num;
            } else if (a > num && num > b) {
                c = b;
                b = num;
            } else if (b > num && num > c) {
                c = num;
            }
        }
        return c == long.MinValue ? (int) a : (int) c;
    }
}
