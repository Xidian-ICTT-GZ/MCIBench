public class Solution {
    public int ThirdMax(int[] nums) {
        int? a = null, b = null, c = null;
        foreach (int num in nums) {
            if (a == null || num > a) {
                c = b;
                b = a;
                a = num;
            } else if (a > num && (b == null || num > b)) {
                c = b;
                b = num;
            } else if (b != null && b > num && (c == null || num > c)) {
                c = num;
            }
        }
        return c == null ? (int) a : (int) c;
    }
}
