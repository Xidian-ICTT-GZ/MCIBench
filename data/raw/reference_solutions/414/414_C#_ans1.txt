public class Solution {
    public int ThirdMax(int[] nums) {
        Array.Sort(nums);
        Array.Reverse(nums);
        for (int i = 1, diff = 1; i < nums.Length; ++i) {
            if (nums[i] != nums[i - 1] && ++diff == 3) { 
                return nums[i];
            }
        }
        return nums[0];
    }
}
