func sumOfPower(nums []int) int {
    n := len(nums)
    sort.Ints(nums)
    dp := make([]int, n)
    preSum := make([]int, n + 1)
    res := 0
    mod := int(1e9 + 7)
    for i := 0; i < n; i++ {
        dp[i] = (nums[i] + preSum[i]) % mod
        preSum[i+1] = (preSum[i] + dp[i]) % mod
        res = (res + (nums[i] * nums[i] % mod * dp[i]) % mod) % mod
    }
    return res
}
