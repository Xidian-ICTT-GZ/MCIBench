impl Solution {
    pub fn count_days(mut days: i32, mut meetings: Vec<Vec<i32>>) -> i32 {
        meetings.sort_by_key(|m| m[0]);
        let (mut l, mut r) = (1, 0);
        for m in meetings {
            let (start, end) = (m[0], m[1]);
            if start > r {
                days -= r - l + 1;
                l = start;
            }
            r = r.max(end);
        }
        days -= r - l + 1;
        days
    }
}
