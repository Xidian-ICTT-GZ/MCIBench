func countDays(days int, meetings [][]int) int {
    sort.Slice(meetings, func(i, j int) bool {
        return meetings[i][0] < meetings[j][0]
    })
    l, r := 1, 0
    for _, m := range meetings {
        if m[0] > r {
            days -= r - l + 1
            l = m[0]
        }
        if m[1] > r {
            r = m[1]
        }
    }
    days -= r - l + 1
    return days
}
