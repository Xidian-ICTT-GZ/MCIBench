var findIntersectionValues = function(nums1, nums2) {
    let res = new Array(2).fill(0);
    for (let i = 0; i < nums1.length; i++) {
        for (let j = 0; j < nums2.length; j++) {
            if (nums1[i] == nums2[j]) {
                res[0]++;
                break;
            }
        }
    }
    for (let i = 0; i < nums2.length; i++) {
        for (let j = 0; j < nums1.length; j++) {
            if (nums2[i] == nums1[j]) {
                res[1]++;
                break;
            }
        }
    }
    return res;
};
