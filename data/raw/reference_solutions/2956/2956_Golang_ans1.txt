func findIntersectionValues(nums1 []int, nums2 []int) []int {
    res := make([]int, 2)
    for _, x1 := range nums1 {
        for _, x2 := range nums2 {
            if x1 == x2 {
                res[0]++
                break
            }
        }
    }
    for _, x2 := range nums2 {
        for _, x1 := range nums1 {
            if x2 == x1 {
                res[1]++
                break
            }
        }
    }
    return res
}
