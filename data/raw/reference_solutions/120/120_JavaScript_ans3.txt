var minimumTotal = function(triangle) {
    const n = triangle.length;
    const f = new Array(n).fill(0);
    f[0] = triangle[0][0];
    
    for (let i = 1; i < n; ++i) {
        f[i] = f[i - 1] + triangle[i][i];
        for (let j = i - 1; j > 0; --j) {
            f[j] = Math.min(f[j - 1], f[j]) + triangle[i][j];
        }
        f[0] += triangle[i][0];
    }
    
    return Math.min(...f);
};
