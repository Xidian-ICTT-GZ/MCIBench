class Solution:
    def rowAndMaximumOnes(self, mat: List[List[int]]) -> List[int]:
        max_ones = 0
        row_index = 0
        for i, row in enumerate(mat):
            tot = sum(row)
            if tot > max_ones:
                max_ones = tot
                row_index = i
        return [row_index, max_ones]
