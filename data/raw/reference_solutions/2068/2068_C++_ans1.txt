class Solution {
public:
    bool checkAlmostEquivalent(string word1, string word2) {
        unordered_map<char, int> freq;   
        for (auto ch: word1){
            ++freq[ch];
        }
        for (auto ch: word2){
            --freq[ch];
        }
        
        return all_of(freq.cbegin(), freq.cend(), [](auto&& x) { return abs(x.second) <= 3; });
    }
};
