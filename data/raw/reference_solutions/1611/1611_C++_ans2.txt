class Solution {
public:
    int minimumOneBitOperations(int n) {
        int ans = 0;
        int sign = 1;
        for (int i = 29; i >= 0; --i) {
            if (n & (1 << i)) {
                ans += sign * ((1 << (i + 1)) - 1);
                sign = -sign;
            }
        }
        return ans;
    }
};
