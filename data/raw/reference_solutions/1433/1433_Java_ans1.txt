class Solution {
    public boolean checkIfCanBreak(String s1, String s2) {
        return check(s1, s2) || check(s2, s1);
    }

    public boolean check(String x, String y) {
        int[] h = new int[26];
        int length = x.length();
        for (int i = 0; i < length; ++i) {
            h[x.charAt(i) - 'a']++;
        }
        for (int i = 0; i < length; ++i) {
            char c = y.charAt(i);
            boolean canFind = false;
            for (int j = c - 'a'; j < 26; ++j) {
                if (h[j] > 0) {
                    --h[j];
                    canFind = true;
                    break;
                }
            }
            if (!canFind) {
                return false;
            }
        }
        return true;
    }
}
