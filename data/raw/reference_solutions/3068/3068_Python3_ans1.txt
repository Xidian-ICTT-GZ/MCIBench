class Solution:
    def maximumValueSum(self, nums: List[int], k: int, edges: List[List[int]]) -> int:
        res = sum(nums)
        diff = [(x ^ k) - x for x in nums]
        diff.sort()
        i = len(diff) - 1
        while i > 0 and diff[i] + diff[i - 1] >= 0:
            res += diff[i] + diff[i - 1]
            i -= 2
        return res
