func maximumValueSum(nums []int, k int, edges [][]int) int64 {
	res := int64(0)
	diff := make([]int, len(nums))
	for i, x := range nums {
		res += int64(x)
		diff[i] = (x ^ k) - x
	}
	sort.Ints(diff)
	for i := len(diff) - 1; i > 0 && diff[i]+diff[i-1] >= 0; i -= 2 {
		res += int64(diff[i] + diff[i-1])
	}
	return res
}
