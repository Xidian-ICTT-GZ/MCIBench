class Solution {
public:
    bool splitString(string s) {
        long long start = 0;
        long long INF = 1e10; 
        int n = s.size();
        
        
        for (int i = 0; i < n - 1 && start < INF; ++i){
            start = start * 10 + (s[i] - '0');
            
            long long pval = start;
            long long cval = 0;
            int cidx = i + 1;
            for (int j = i + 1; j < n && cval < INF; ++j){
                if (pval == 1){
                    
                    if (all_of(s.begin() + cidx, s.end(), [](char c){ return c == '0'; })){
                        return true;
                    }
                    else{
                        break;
                    }
                }
                cval = cval * 10 + (s[j] - '0');
                if (cval > pval - 1){
                    
                    break;
                }
                else if (cval == pval - 1){
                    if (j + 1 == n){
                        
                        return true;
                    }
                    pval = cval;
                    cval = 0;
                    cidx = j + 1;
                }
            }
        }
        return false;
    }
};
