class Solution:
    def splitString(self, s: str) -> bool:
        n = len(s)
        start = 0
        
        
        for i in range(n - 1):
            start = 10 * start + int(s[i])
            
            pval = start
            cval = 0
            cidx = i + 1
            for j in range(i + 1, n):
                if pval == 1:
                    
                    if all(s[k] == '0' for k in range(cidx, n)):
                        return True
                    else:
                        break
                cval = 10 * cval + int(s[j])
                if cval > pval - 1:
                    
                    break
                elif cval == pval - 1:
                    if j + 1 == n:
                        
                        return True
                    pval = cval
                    cval = 0
                    cidx = j + 1     
        return False
