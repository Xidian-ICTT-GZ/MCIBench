func longestMountain(arr []int) int {
    n := len(arr)
    ans := 0
    left := 0
    for left+2 < n {
        right := left + 1
        if arr[left] < arr[left+1] {
            for right+1 < n && arr[right] < arr[right+1] {
                right++
            }
            if right < n-1 && arr[right] > arr[right+1] {
                for right+1 < n && arr[right] > arr[right+1] {
                    right++
                }
                if right-left+1 > ans {
                    ans = right - left + 1
                }
            } else {
                right++
            }
        }
        left = right
    }
    return ans
}
