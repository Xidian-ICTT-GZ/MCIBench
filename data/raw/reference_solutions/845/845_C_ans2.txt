int longestMountain(int* arr, int arrSize) {
    int ans = 0;
    int left = 0;
    while (left + 2 < arrSize) {
        int right = left + 1;
        if (arr[left] < arr[left + 1]) {
            while (right + 1 < arrSize && arr[right] < arr[right + 1]) {
                ++right;
            }
            if (right < arrSize - 1 && arr[right] > arr[right + 1]) {
                while (right + 1 < arrSize && arr[right] > arr[right + 1]) {
                    ++right;
                }
                ans = fmax(ans, right - left + 1);
            } else {
                ++right;
            }
        }
        left = right;
    }
    return ans;
}
