class Solution:
    def maxSumMinProduct(self, nums: List[int]) -> int:
        mod = 10**9 + 7

        n = len(nums)
        
        
        left, right = [0] * n, [n - 1] * n
        
        s = list()
        for i, num in enumerate(nums):
            while s and nums[s[-1]] >= num:
                
                right[s[-1]] = i - 1
                s.pop()
            if s:
                
                left[i] = s[-1] + 1
            s.append(i)
        
        
        pre = [0]
        for i, num in enumerate(nums):
            pre.append(pre[-1] + num)
        
        best = max((pre[right[i] + 1] - pre[left[i]]) * num for i, num in enumerate(nums))
        return best % mod
