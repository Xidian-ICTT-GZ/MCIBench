class Solution {
public:
    string kthLargestNumber(vector<string>& nums, int k) {
        
        auto cmp = [](const string& s1, const string& s2) -> bool{
            
            if (s1.size() > s2.size()){
                return true;
            }
            else if (s1.size() < s2.size()){
                return false;
            }
            else{
                
                return s1 > s2;
            }
        };
        
        sort(nums.begin(), nums.end(), cmp);
        return nums[k-1];
    }
};
