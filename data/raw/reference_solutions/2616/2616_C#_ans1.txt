public class Solution {
    public int MinimizeMax(int[] nums, int p) {
        Array.Sort(nums);
        int left = 0, right = nums[nums.Length - 1] - nums[0];

        while (left < right) {
            int mid = (left + right) / 2;
            int count = 0;
            for (int i = 0; i < nums.Length - 1; ) {
                if (nums[i + 1] - nums[i] <= mid) {
                    count++;
                    i += 2;
                } else {
                    i++;
                }
            }

            if (count >= p) {
                right = mid;
            } else {
                left = mid + 1;
            }
        }

        return left;
    }
}
