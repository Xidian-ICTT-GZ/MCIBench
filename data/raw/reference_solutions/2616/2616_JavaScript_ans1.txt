function minimizeMax(nums, p) {
    nums.sort((a, b) => a - b);
    let left = 0, right = nums[nums.length - 1] - nums[0];

    const check = (mx) => {
        let cnt = 0;
        for (let i = 0; i < nums.length - 1;) {
            if (nums[i + 1] - nums[i] <= mx) {
                cnt++;
                i += 2;
            } else {
                i++;
            }
        }
        return cnt >= p;
    };

    while (left < right) {
        let mid = Math.floor((left + right) / 2);
        if (check(mid)) {
            right = mid;
        } else {
            left = mid + 1;
        }
    }

    return left;
}
