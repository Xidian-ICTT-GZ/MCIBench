class Solution:
    def processQueries(self, queries: List[int], m: int) -> List[int]:
        p = [x + 1 for x in range(m)]
        ans = list()
        for query in queries:
            pos = -1
            for i in range(m):
                if p[i] == query:
                    pos = i
                    break
            ans.append(pos)
            p.pop(pos)
            p.insert(0, query)
        return ans
