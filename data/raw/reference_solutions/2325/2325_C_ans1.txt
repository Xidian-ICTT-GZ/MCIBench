char * decodeMessage(char * key, char * message){
    char cur = 'a';
    char rules[26];
    memset(rules, 0, sizeof(rules));

    for (int i = 0; key[i] != '\0'; i++) {
        char c = key[i];
        if (c != ' ' && !rules[c - 'a']) {
            rules[c - 'a'] = cur;
            ++cur;
        }
    }

    for (int i = 0; message[i] != '\0'; i++) {
        char c = message[i];
        if (c != ' ') {
            message[i] = rules[c - 'a'];
        }
    }

    return message;
}
