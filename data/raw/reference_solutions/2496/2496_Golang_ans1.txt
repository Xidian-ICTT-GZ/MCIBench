func max(a int, b int) int {
    if a > b {
        return a
    }
    return b
}

func maximumValue(strs []string) int {
    res := 0
    for _, s := range strs {
        isDigits := true
        for _, c := range s {
            isDigits = isDigits && (c >= '0' && c <= '9')
        }
        if isDigits {
            v, _ := strconv.Atoi(s)
            res = max(res, v)
        } else {
            res = max(res, len(s))
        }
    }
    return res
}
