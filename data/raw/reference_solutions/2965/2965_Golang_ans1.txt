func findMissingAndRepeatedValues(grid [][]int) []int {
    n := len(grid)
    count := make([]int, n * n + 1)
    for i := 0; i < n; i++ {
        for j := 0; j < n; j++ {
            count[grid[i][j]]++
        }
    }

    res := make([]int, 2)
    for i := 1; i <= n * n; i++ {
        if count[i] == 2 {
            res[0] = i;
        }
        if count[i] == 0 {
            res[1] = i;
        }
    }
    return res
}
