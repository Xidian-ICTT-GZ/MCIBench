class Solution {
public:
    int minSwaps(string s) {
        int n = s.size();
        int n0 = count(s.begin(), s.end(), '0');
        int n1 = count(s.begin(), s.end(), '1');
        int res = INT_MAX;
        
        if (n1 == (n + 1) / 2 && n0 == n / 2){   
            int diff1 = 0;
            for (int i = 0; i < n; ++i){
                if (s[i] - '0' == i % 2){   
                    ++diff1;
                }
            }
            res = min(res, diff1 / 2);
        }
        
        if (n0 == (n + 1) / 2 && n1 == n / 2){   
            int diff2 = 0;
            for (int i = 0; i < n; ++i){
                if (s[i] - '0' != i % 2){   
                    ++diff2;
                }
            }
            res = min(res, diff2 / 2);
        }
        if (res == INT_MAX){
            return -1;   
        }
        else {
            return res;
        }
    }
};
