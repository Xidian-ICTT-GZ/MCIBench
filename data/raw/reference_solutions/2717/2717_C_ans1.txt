int semiOrderedPermutation(int* nums, int numsSize) {
    int first = 0, last = 0;
    for (int i = 0; i < numsSize; i++) {
        if (nums[i] == 1) {
            first = i;
        }
        if (nums[i] == numsSize) {
            last = i;
        }
    }
    return first + numsSize - 1 - last - (last < first ? 1 : 0);
}
