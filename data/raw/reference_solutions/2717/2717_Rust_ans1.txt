impl Solution {
    pub fn semi_ordered_permutation(nums: Vec<i32>) -> i32 {
        let n = nums.len() as i32;
        let mut first = 0;
        let mut last = 0;
        for (i, &num) in nums.iter().enumerate() {
            if num == 1 {
                first = i as i32;
            }
            if num == n {
                last = i as i32;
            }
        }
        first + n - 1 - last - if last < first { 1 } else { 0 }
    }
}
