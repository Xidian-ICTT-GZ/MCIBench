func semiOrderedPermutation(nums []int) int {
    n := len(nums)
    first, last := 0, 0
    for i := 0; i < n; i++ {
        if nums[i] == 1 {
            first = i
        }
        if nums[i] == n {
            last = i
        }
    }
    if last < first {
        return first + n - 1 - last - 1
    }
    return first + n - 1 - last
}
