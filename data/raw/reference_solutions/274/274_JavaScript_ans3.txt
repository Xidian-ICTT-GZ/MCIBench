var hIndex = function(citations) {
    let left = 0, right = citations.length
    while (left<right){
        
        let mid = Math.floor((left + right + 1) / 2)
        let cnt = 0
        for (let v of citations){
            if (v >= mid){
                cnt+=1
            }
        }
        if(cnt>=mid){
            
            left=mid
        }else{
            
            right=mid-1
        }
    }
    return left
};
