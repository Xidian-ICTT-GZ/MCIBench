public class Solution {
    public bool AreNumbersAscending(string s) {
        int pre = 0, pos = 0;
        while (pos < s.Length) {
            if (char.IsDigit(s[pos])) {
                int cur = 0;
                while (pos < s.Length && char.IsDigit(s[pos])) {
                    cur = cur * 10 + s[pos] - '0';
                    pos++;
                }
                if (cur <= pre) {
                    return false;
                }
                pre = cur;
            } else {
                pos++;
            }
        }
        return true;
    }
}
