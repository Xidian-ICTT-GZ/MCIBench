def count_one(x):
    res = 0
    sum = 0

    for i in range(60, -1, -1):
        if (1 << i) & x:
            res += sum * (1 << i)
            sum += 1

            if i > 0:
                res += i * (1 << (i - 1))
    
    res += sum
    return res


def count_pow(x):
    res = 0
    sum = 0

    for i in range(60, -1, -1):
        if (1 << i) & x:
            res += sum * (1 << i)
            sum += i

            if i > 0:
                res += i * (i - 1) // 2 * (1 << (i - 1))
    
    res += sum
    return res

def pow_mod(x, y, mod):
    res = 1
    while y:
        if y & 1:
            res = res * x % mod
        x = x * x % mod
        y >>= 1
    return res

def mid_check(x):
    l, r = 1, int(1e15)
    while l < r:
        mid = (l + r) >> 1
        if count_one(mid) >= x:
            r = mid
        else:
            l = mid + 1
    return r

class Solution:
    def findProductsOfElements(self, queries: List[List[int]]) -> List[int]:
        ans = []
        for query in queries:
            
            query[0] += 1
            query[1] += 1
            l = mid_check(query[0])
            r = mid_check(query[1])
            mod = query[2]

            res = 1
            pre = count_one(l - 1)
            for j in range(60):
                if (1 << j) & l:
                    pre += 1
                    if query[0] <= pre <= query[1]:
                        res = res * (1 << j) % mod

            if r > l:
                bac = count_one(r - 1)
                for j in range(60):
                    if (1 << j) & r:
                        bac += 1
                        if query[0] <= bac <= query[1]:
                            res = res * (1 << j) % mod

            if r - l > 1:
                xs = count_pow(r - 1) - count_pow(l)
                res = res * pow_mod(2, xs, mod) % mod
            
            ans.append(res)

        return ans
