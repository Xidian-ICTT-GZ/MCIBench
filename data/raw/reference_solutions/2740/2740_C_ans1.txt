int cmp(const void *p1, const void *p2) {
    return *(const int *)p1 - *(const int *)p2;
}

int findValueOfPartition(int *nums, int numsSize){
    qsort(nums, numsSize, sizeof(int), cmp);
    int res = INT_MAX;
    for (int i = 1; i < numsSize; i++) {
        res = fmin(res, nums[i] - nums[i - 1]);
    }
    return res;
}
