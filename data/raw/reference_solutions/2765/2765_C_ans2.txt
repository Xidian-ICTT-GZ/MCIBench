int alternatingSubarray(int* nums, int numsSize) {
    int res = -1;
    int firstIndex = 0;
    for (int i = 1; i < numsSize; i++) {
        int length = i - firstIndex + 1;
        if (nums[i] - nums[firstIndex] == (length - 1) % 2) {
            res = fmax(res, length);
        } else {
            if (nums[i] - nums[i - 1] == 1) {
                firstIndex = i - 1;
                res = fmax(res, 2);
            } else {
                firstIndex = i;
            }
        }
    }

    return res;
}
