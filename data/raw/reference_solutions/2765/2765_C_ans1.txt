int alternatingSubarray(int* nums, int numsSize) {
    int res = -1;
    for (int firstIndex = 0; firstIndex < numsSize; firstIndex++) {
        for (int i = firstIndex + 1; i < numsSize; i++) {
            int length = i - firstIndex + 1;
            if (nums[i] - nums[firstIndex] == (length - 1) % 2) {
                res = fmax(res, length);
            } else {
                break;
            }
        }
    }
    return res;
}
