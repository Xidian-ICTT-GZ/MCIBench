public class Solution {
    public int[] SortArrayByParityII(int[] nums) {
        int n = nums.Length;
        int[] ans = new int[n];
        int i = 0;
        foreach (int x in nums) {
            if (x % 2 == 0) {
                ans[i] = x;
                i += 2;
            }
        }
        i = 1;
        foreach (int x in nums) {
            if (x % 2 == 1) {
                ans[i] = x;
                i += 2;
            }
        }
        return ans;
    }
}
