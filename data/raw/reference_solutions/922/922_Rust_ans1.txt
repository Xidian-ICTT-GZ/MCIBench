impl Solution {
    pub fn sort_array_by_parity_ii(nums: Vec<i32>) -> Vec<i32> {
        let n = nums.len();
        let mut ans = vec![0; n];
        let mut i = 0;
        for &x in &nums {
            if x % 2 == 0 {
                ans[i] = x;
                i += 2;
            }
        }
        i = 1;
        for &x in &nums {
            if x % 2 == 1 {
                ans[i] = x;
                i += 2;
            }
        }
        ans
    }
}
