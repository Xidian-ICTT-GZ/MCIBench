class Solution:
    def sortArrayByParityII(self, nums: List[int]) -> List[int]:
        n = len(nums)
        ans = [0] * n
        i = 0
        for x in nums:
            if x % 2 == 0:
                ans[i] = x
                i += 2
        i = 1
        for x in nums:
            if x % 2 == 1:
                ans[i] = x
                i += 2
        return ans
