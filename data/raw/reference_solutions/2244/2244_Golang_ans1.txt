func minimumRounds(tasks []int) int {
    cnt := map[int]int{}
    for _, t := range tasks {
        cnt[t]++
    }
    res := 0
    for _, v := range cnt {
        if v == 1 {
            return -1
        } else if v % 3 == 0 {
            res += v / 3
        } else {
            res += v / 3 + 1
        }
    }
    return res
}
