class Solution {
    public int numberOfArrays(String s, int k) {
        final int MOD = 1000000007;
        int n = s.length();
        
        long kLong = k;
        int[] f = new int[n + 1];
        
        f[0] = 1;
        for (int i = 1; i <= n; ++i) {
            long num = 0, base = 1;
            
            for (int j = i - 1; j >= 0 && i - j <= 10; --j) {
                
                
                num += (s.charAt(j) - '0') * base;
                if (num > kLong) {
                    break;
                }
                
                if (s.charAt(j) != '0') {
                    f[i] += f[j];
                    f[i] %= MOD;
                }
                base *= 10;
            }
        }
        return f[n];
    }
}
