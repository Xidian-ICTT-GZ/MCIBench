using LL = long long;

class Solution {
private:
    static constexpr int mod = 1000000007;

public:
    int numberOfArrays(string s, int k) {
        int n = s.size();
        
        LL kll = k;
        vector<int> f(n + 1, 0);
        
        f[0] = 1;
        for (int i = 1; i <= n; ++i) {
            LL num = 0, base = 1;
            
            for (int j = i - 1; j >= 0 && i - j <= 10; --j) {
                
                
                num += (s[j] - '0') * base;
                if (num > kll) {
                    break;
                }
                
                if (s[j] != '0') {
                    f[i] += f[j];
                    f[i] %= mod;
                }
                base *= 10;
            }
        }
        return f[n];
    }
};
