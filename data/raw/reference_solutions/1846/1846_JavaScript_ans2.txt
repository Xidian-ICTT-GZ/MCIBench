var maximumElementAfterDecrementingAndRearranging = function(arr) {
    const n = arr.length;
    const cnt = new Array(n + 1).fill(0);
    for (const v of arr) {
        ++cnt[Math.min(v, n)];
    }
    let miss = 0;
    for (let i = 1; i <= n; ++i) {
        if (cnt[i] == 0) {
            ++miss;
        } else {
            miss -= Math.min(cnt[i] - 1, miss); 
        }
    }
    return n - miss;
};
