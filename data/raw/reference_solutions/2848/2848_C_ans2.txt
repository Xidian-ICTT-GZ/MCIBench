int numberOfPoints(int** nums, int numsSize, int* numsColSize){
    int C = 0;
    for (int i = 0; i < numsSize; i++) {
        if (nums[i][1] > C) {
            C = nums[i][1];
        }
    }

    int diff[C + 2];
    memset(diff, 0, sizeof(diff));
    for (int i = 0; i < numsSize; i++) {
        diff[nums[i][0]]++;
        diff[nums[i][1] + 1]--;
    }

    int ans = 0, count = 0;
    for (int i = 1; i <= C; i++) {
        count += diff[i];
        if (count > 0) {
            ans++;
        }
    }

    return ans;
}
