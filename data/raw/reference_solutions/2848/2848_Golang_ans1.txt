func numberOfPoints(nums [][]int) int {
    C := 0
	for _, interval := range nums {
		if interval[1] > C {
			C = interval[1]
		}
	}

	count := make([]int, C + 1)
	for _, interval := range nums {
		for i := interval[0]; i <= interval[1]; i++ {
			count[i]++
		}
	}

	ans := 0
	for i := 1; i <= C; i++ {
		if count[i] > 0 {
			ans++
		}
	}
	return ans
}
