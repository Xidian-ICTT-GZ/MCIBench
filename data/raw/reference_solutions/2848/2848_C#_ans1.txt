public class Solution {
    public int NumberOfPoints(IList<IList<int>> nums) {
        int C = 0;
        foreach (IList<int> interval in nums) {
            C = Math.Max(C, interval[1]);
        }

        int[] count = new int[C + 1];
        foreach (IList<int> interval in nums) {
            for (int i = interval[0]; i <= interval[1]; ++i) {
                ++count[i];
            }
        }

        int ans = 0;
        for (int i = 1; i <= C; ++i) {
            if (count[i] > 0) {
                ++ans;
            }
        }
        return ans;
    }
}
