impl Solution {
    pub fn find_champion(n: i32, edges: Vec<Vec<i32>>) -> i32 {
        let mut degree = vec![0; n as usize];
        for e in edges.iter() {
            degree[e[1] as usize] += 1;
        }
        let mut champion = -1;
        for i in 0..n {
            if degree[i as usize] == 0 {
                if champion == -1 {
                    champion = i;
                } else {
                    return -1;
                }
            }
        }
        champion
    }
}
