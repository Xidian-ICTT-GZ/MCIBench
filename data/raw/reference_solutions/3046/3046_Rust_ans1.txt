use std::collections::HashMap;

impl Solution {
    pub fn is_possible_to_split(nums: Vec<i32>) -> bool {
        let mut count = HashMap::new();
        for num in nums {
            let counter = count.entry(num).or_insert(0);
            *counter += 1;
            if *counter > 2 {
                return false;
            }
        }
        true
    }
}
