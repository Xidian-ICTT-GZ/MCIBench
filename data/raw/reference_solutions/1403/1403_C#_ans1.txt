public class Solution {
    public IList<int> MinSubsequence(int[] nums) {
        int total = nums.Sum();
        Array.Sort(nums);
        IList<int> ans = new List<int>();
        int curr = 0;
        for (int i = nums.Length - 1; i >= 0; --i) {
            curr += nums[i];
            ans.Add(nums[i]);
            if (total - curr < curr) {
                break;
            }
        }
        return ans;
    }
}
