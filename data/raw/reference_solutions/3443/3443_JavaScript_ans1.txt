var maxDistance = function(s, k) {
    let ans = 0;
    let north = 0, south = 0, east = 0, west = 0;
    for (const it of s) {
        switch (it) {
            case 'N':
                north++;
                break;
            case 'S':
                south++;
                break;
            case 'E':
                east++;
                break;
            case 'W':
                west++;
                break;
        }

        const count = (drt1, drt2, times) => {
            return Math.abs(drt1 - drt2) + times * 2; 
        };

        let times1 = Math.min(north, south, k); 
        let times2 = Math.min(east, west, k - times1); 
        ans = Math.max(ans, count(north, south, times1) + count(east, west, times2));
    }
    return ans;
};
