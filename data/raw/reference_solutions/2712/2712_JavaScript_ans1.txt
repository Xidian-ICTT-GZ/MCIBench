var minimumCost = function(s) {
    const n = s.length;
    const suf = Array.from({ length: n + 1 }, () => [0, 0]);
    for (let i = n - 1; i >= 0; i--) {
        if (s[i] === '1') {
            suf[i][1] = suf[i + 1][1];
            suf[i][0] = suf[i + 1][1] + (n - i);
        } else {
            suf[i][1] = suf[i + 1][0] + (n - i);
            suf[i][0] = suf[i + 1][0];
        }
    }

    let pre = [0, 0];
    let res = Infinity;
    for (let i = 0; i < n; i++) {
        if (s[i] === '1') {
            pre[0] = pre[1] + i + 1;
        } else {
            pre[1] = pre[0] + i + 1;
        }
        res = Math.min(res, Math.min(pre[0] + suf[i + 1][0], pre[1] + suf[i + 1][1]));
    }
    return res;
};
