impl Solution {
    pub fn min_operations(nums: Vec<i32>) -> i32 {
        let mut operation = 0;
        for i in (0..nums.len() - 1).rev() {
            if nums[i] != nums[i + 1] {
                operation += 1;
            }
        }
        if nums[0] == 1 {
            operation
        } else {
            operation + 1
        }
    }
}
