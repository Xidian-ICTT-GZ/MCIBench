public class Solution {
    public string GetSmallestString(string s) {
        char[] arr = s.ToCharArray();
        for (int i = 0; i + 1 < arr.Length; i++) {
            if (arr[i] > arr[i + 1] && arr[i] % 2 == arr[i + 1] % 2) {
                char temp = arr[i];
                arr[i] = arr[i + 1];
                arr[i + 1] = temp;
                break;
            }
        }
        return new string(arr);
    }
}
