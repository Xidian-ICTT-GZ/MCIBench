var canMakeSquare = function(grid) {
    for (let i = 0; i <= 1; i++) {
        for (let j = 0; j <= 1; j++) {
            if (check(grid, i, j)) {
                return true;
            }
        }
    }
    return false;
};

var check = function(grid, x, y) {
    let count = 0;
    for (let i = 0; i <= 1; i++) {
        for (let j = 0; j <= 1; j++) {
            if (grid[x + i][y + j] == 'B') {
                count++;
            }
        }
    }
    return count != 2;
};
