int findKOr(int* nums, int numsSize, int k) {
    int ans = 0;
    for (int i = 0; i < 31; ++i) {
        int cnt = 0;
        for (int j = 0; j < numsSize; j++) {
            int num = nums[j];
            if ((num >> i) & 1) {
                ++cnt;
            }
        }
        if (cnt >= k) {
            ans |= 1 << i;
        }
    }
    return ans;
}
