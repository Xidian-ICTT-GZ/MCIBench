const POLE_NUM = 10;

var countPoints = function(rings) {
    const state = new Array(POLE_NUM).fill(0);
    const n = rings.length;
    for (let i = 0; i < n; i += 2) {
        const color = rings[i];
        const pole_index = rings[i + 1] - '0';
        if (color == 'R') {
            state[pole_index] |= 1;
        } else if (color == 'G') {
            state[pole_index] |= 2;
        } else {
            state[pole_index] |= 4;
        }
    }
    let res = 0;
    for (let i = 0; i < POLE_NUM; i++) {
        res += state[i] == 7;
    }
    return res;
};
