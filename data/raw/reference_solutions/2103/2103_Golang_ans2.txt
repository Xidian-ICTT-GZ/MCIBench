var POLE_NUM = 10

func countPoints(rings string) int {
    state := make([]int, POLE_NUM)
    n := len(rings)
    for i := 0; i < n; i += 2 {
        color := rings[i]
        pole_index := rings[i + 1] - '0'
        if color == 'R' {
            state[pole_index] |= 1
        } else if color == 'G' {
            state[pole_index] |= 2
        } else {
            state[pole_index] |= 4
        }
    }
    res := 0
    for i := 0; i < POLE_NUM; i++ {
        if state[i] == 7 {
            res++
        }
    }
    return res
}
