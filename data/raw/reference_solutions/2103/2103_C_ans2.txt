const int POLE_NUM = 10;

int countPoints(char * rings) {
    int state[POLE_NUM];
    memset(state, 0, sizeof(state));
    int n = strlen(rings);
    for (int i = 0; i < n; i += 2) {
        char color = rings[i];
        int pole_index = rings[i + 1] - '0';
        if (color == 'R') {
            state[pole_index] |= 1;
        } else if (color == 'G') {
            state[pole_index] |= 2;
        } else {
            state[pole_index] |= 4;
        }
    }
    int res = 0;
    for (int i = 0; i < POLE_NUM; i++) {
        res += state[i] == 7;
    }
    return res;
}
