public class Solution {
    public int BalancedStringSplit(string s) {
        int ans = 0, d = 0;
        foreach (char ch in s) {
            if (ch == 'L') {
                ++d;
            } else {
                --d;
            }
            if (d == 0) {
                ++ans;
            }
        }
        return ans;
    }
}
