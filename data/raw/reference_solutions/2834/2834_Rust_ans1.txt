impl Solution {
    pub fn minimum_possible_sum(n: i32, target: i32) -> i32 {
        const MOD: i64 = 1_000_000_007;
        let n: i64 = n as i64;
        let target: i64 = target as i64;
        let m: i64 = target / 2;
        if n <= m {
            return (((1 + n) * n / 2) % MOD) as i32
        }
        return ((((1 + m) * m / 2) + (((target + target + (n - m) - 1) * (n - m) / 2))) % MOD) as i32;
    }
}
