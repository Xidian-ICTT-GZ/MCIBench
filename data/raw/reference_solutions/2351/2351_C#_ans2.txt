public class Solution {
    public char RepeatedCharacter(string s) {
        int seen = 0;
        foreach (char ch in s) {
            int x = ch - 'a';
            if ((seen & (1 << x)) != 0) {
                return ch;
            }
            seen |= (1 << x);
        }
        
        return ' ';
    }
}
