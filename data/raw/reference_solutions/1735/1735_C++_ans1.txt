class Solution {
public:
    using LL = long long;
    const long long MOD = 1e9 + 7;
    const int MAXN = 1e4 + 14;
    const int MAXM = 14;
    vector<int> waysToFillArray(vector<vector<int>>& queries) {
        vector<vector<LL>> comb(MAXN, vector<LL>(MAXM));
        vector<int> ans;

        comb[0][0] = 1;
        for (int i = 1; i < MAXN; i++) {
            comb[i][0] = 1;
            for (int j = 1; j <= i && j < MAXM; j++) {
                comb[i][j] = (comb[i - 1][j - 1] + comb[i - 1][j]) % MOD;
            }
        }

        for(auto &q : queries) {
            int n = q[0], k = q[1];
            long long tot = 1;
            for (int i = 2; i * i <= k; i++) {
                if (k % i == 0) {
                    int cnt = 0;
                    while (k % i == 0) {
                        k /= i;
                        cnt++;
                    }
                    tot = (tot * comb[n + cnt - 1][cnt]) % MOD;
                }
            }
            
            if(k > 1) {
                tot = tot * n % MOD;
            }
            ans.push_back(tot);
        }
        return ans;
    }
};
