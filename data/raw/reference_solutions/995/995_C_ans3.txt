int minKBitFlips(int* nums, int numsSize, int k) {
    int diff[numsSize + 1];
    memset(diff, 0, sizeof(diff));
    int ans = 0, revCnt = 0;
    for (int i = 0; i < numsSize; ++i) {
        revCnt ^= diff[i];
        if (nums[i] == revCnt) {  
            if (i + k > numsSize) {
                return -1;
            }
            ++ans;
            revCnt ^= 1;
            diff[i + k] ^= 1;
        }
    }
    return ans;
}
