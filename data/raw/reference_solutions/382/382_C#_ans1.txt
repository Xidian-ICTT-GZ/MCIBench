public class Solution {
    IList<int> list;
    Random random;

    public Solution(ListNode head) {
        list = new List<int>();
        while (head != null) {
            list.Add(head.val);
            head = head.next;
        }
        random = new Random();
    }

    public int GetRandom() {
        return list[random.Next(list.Count)];
    }
}
