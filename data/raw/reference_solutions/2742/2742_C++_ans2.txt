class Solution {
public:
    int paintWalls(vector<int>& cost, vector<int>& time) {
        int n = cost.size();
        vector<int> f(n + 2, INT_MAX / 2);
        f[0] = 0;
        for (int i = 0; i < n; ++i) {
            for (int j = n + 1; j >= 0; --j) {
                f[min(j + time[i], n) + 1] = min(f[min(j + time[i], n) + 1], f[j] + cost[i]);
            }
        }
        return min(f[n], f[n + 1]);
    }
};
