func paintWalls(cost []int, time []int) int {
    n := len(cost)
    f := make([]int, n + 2)
    for k := range f {
        f[k] = math.MaxInt / 2
    }
    f[0] = 0
    for i := 0; i < n; i++ {
        for j := n + 1; j >= 0; j-- {
            f[min(j + time[i], n) + 1] = min(f[min(j + time[i], n) + 1], f[j] + cost[i])
        }
    }
    return min(f[n], f[n + 1])
}
