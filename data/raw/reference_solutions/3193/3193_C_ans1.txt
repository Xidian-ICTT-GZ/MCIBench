typedef struct {
    int key;
    int val;
    UT_hash_handle hh;
} HashItem; 

HashItem *hashFindItem(HashItem **obj, int key) {
    HashItem *pEntry = NULL;
    HASH_FIND_INT(*obj, &key, pEntry);
    return pEntry;
}

bool hashAddItem(HashItem **obj, int key, int val) {
    if (hashFindItem(obj, key)) {
        return false;
    }
    HashItem *pEntry = (HashItem *)malloc(sizeof(HashItem));
    pEntry->key = key;
    pEntry->val = val;
    HASH_ADD_INT(*obj, key, pEntry);
    return true;
}

bool hashSetItem(HashItem **obj, int key, int val) {
    HashItem *pEntry = hashFindItem(obj, key);
    if (!pEntry) {
        hashAddItem(obj, key, val);
    } else {
        pEntry->val = val;
    }
    return true;
}

int hashGetItem(HashItem **obj, int key, int defaultVal) {
    HashItem *pEntry = hashFindItem(obj, key);
    if (!pEntry) {
        return defaultVal;
    }
    return pEntry->val;
}

void hashFree(HashItem **obj) {
    HashItem *curr = NULL, *tmp = NULL;
    HASH_ITER(hh, *obj, curr, tmp) {
        HASH_DEL(*obj, curr);  
        free(curr);
    }
}

const long long MOD = 1e9 + 7;

int dfs(int end, int cnt, HashItem **reqMap, int **dp) {
    if (end == 0) {
        return 1;
    }
    if (dp[end][cnt] != -1) {
        return dp[end][cnt];
    }

    if (hashFindItem(reqMap, end - 1)) {
        int r = hashGetItem(reqMap, end - 1, 0);
        if (r <= cnt && cnt <= end + r) {
            return dp[end][cnt] = dfs(end - 1, r, reqMap, dp);
        } else {
            return dp[end][cnt] = 0;
        }
    } else {
        long long sum = 0;
        for (int i = 0; i <= fmin(end, cnt); ++i) {
            sum = (sum + dfs(end - 1, cnt - i, reqMap, dp)) % MOD;
        }
        return dp[end][cnt] = sum;
    }
}

int numberOfPermutations(int n, int** requirements, int requirementsSize, int* requirementsColSize) {
    HashItem *reqMap = NULL;
    int maxCnt = 0;
    hashAddItem(&reqMap, 0, 0);
    for (int i = 0; i < requirementsSize; i++) {
        int end = requirements[i][0];
        int cnt = requirements[i][1];
        hashSetItem(&reqMap, end, cnt);
        maxCnt = fmax(maxCnt, cnt);
    }
    if (hashGetItem(&reqMap, 0, 0)) {
        return 0;
    }
    int *dp[n];
    for (int i = 0; i < n; i++) {
        dp[i] = (int *)malloc(sizeof(int) * (maxCnt + 1));
        memset(dp[i], 0xff, sizeof(int) * (maxCnt + 1));
    }
    
    int ret = dfs(n - 1, hashGetItem(&reqMap, n - 1, 0), &reqMap, dp);
    hashFree(&reqMap);
    return ret;
}
