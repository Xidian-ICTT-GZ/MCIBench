class Solution {
public:
    vector<int> rearrangeArray(vector<int>& nums) {
        int n = nums.size();
        int pos = 0, neg = 0;
        vector<int> ans;
        for (int i = 0; i + i < n; ++i) {
            while (nums[pos] < 0) {
                ++pos;
            }
            ans.push_back(nums[pos]);
            ++pos;
            while (nums[neg] > 0) {
                ++neg;
            }
            ans.push_back(nums[neg]);
            ++neg;
        }
        return ans;
    }
};
