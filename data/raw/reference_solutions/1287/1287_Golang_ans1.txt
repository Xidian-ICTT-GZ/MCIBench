func findSpecialInteger(arr []int) int {
    n := len(arr)
    cur := arr[0]
    cnt := 0
    for i := 0; i < n; i++ {
        if arr[i] == cur {
            cnt++
            if cnt*4 > n {
                return cur
            }
        } else {
            cur = arr[i]
            cnt = 1
        }
    }
    return -1
}
