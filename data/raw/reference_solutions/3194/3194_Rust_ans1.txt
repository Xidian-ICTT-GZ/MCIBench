impl Solution {
    pub fn minimum_average(nums: Vec<i32>) -> f64 {
        let mut nums = nums;
        nums.sort();
        let n = nums.len();
        let mut res = f64::INFINITY;
        for i in 0..n/2 {
            let average = (nums[i] as f64 + nums[n - 1 - i] as f64) / 2.0;
            res = res.min(average);
        }
        res
    }
}
