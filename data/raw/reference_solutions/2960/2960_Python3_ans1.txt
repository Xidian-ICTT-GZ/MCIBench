class Solution:
    def countTestedDevices(self, batteryPercentages: List[int]) -> int:
        n = len(batteryPercentages)
        need = 0
        for i in range(n):
            if batteryPercentages[i] > 0:
                need += 1
                for j in range(i + 1, n):
                    batteryPercentages[j] = max(batteryPercentages[j] - 1, 0)
        return need
