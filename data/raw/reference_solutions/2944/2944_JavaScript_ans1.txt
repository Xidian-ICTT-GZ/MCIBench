var minimumCoins = function(prices) {
    const memo = new Map();

    const dp = (index) => {
        if (2 * index + 2 >= prices.length) {
            return prices[index];
        }
        if (!memo.has(index)) {
            let minValue = Infinity;
            for (let i = index + 1; i <= 2 * index + 2; i++) {
                minValue = Math.min(minValue, dp(i));
            }
            memo.set(index, prices[index] + minValue);
        }
        return memo.get(index);
    };

    return dp(0);
};
