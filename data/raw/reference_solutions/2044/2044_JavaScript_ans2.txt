var countMaxOrSubsets = function(nums) {
    this.nums = nums;
    this.maxOr = 0;
    this.cnt = 0;
    dfs(0, 0);
    return cnt;
};

const dfs = (pos, orVal) => {
    if (pos === nums.length) {
        if (orVal > maxOr) {
            maxOr = orVal;
            cnt = 1;
        } else if (orVal === maxOr) {
            cnt++;
        }
        return;
    }
    dfs(pos + 1, orVal | nums[pos]);
    dfs(pos + 1, orVal);
}
