func miceAndCheese(reward1 []int, reward2 []int, k int) int {
    ans := 0
    n := len(reward1)
    diffs := make([]int, n)
    for i:= 0; i < n; i++ {
        ans += reward2[i]
        diffs[i] = reward1[i] - reward2[i]
    }
    sort.Ints(diffs)
    for i:=1; i <= k; i++ {
        ans += diffs[n - i]
    }
    return ans
}
