public class Solution {
    public int MiceAndCheese(int[] reward1, int[] reward2, int k) {
        int ans = 0;
        int n = reward1.Length;
        int[] diffs = new int[n];
        for (int i = 0; i < n; i++) {
            ans += reward2[i];
            diffs[i] = reward1[i] - reward2[i];
        }
        Array.Sort(diffs);
        for (int i = 1; i <= k; i++) {
            ans += diffs[n - i];
        }
        return ans;
    }
}
