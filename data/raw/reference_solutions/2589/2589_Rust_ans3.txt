impl Solution {
    pub fn find_minimum_time(tasks: Vec<Vec<i32>>) -> i32 {
        let mut res = 0;
        let mut i = 1;
        let mut tasks = tasks;
        loop {
            let mut finished = true;
            let mut run = false;
            for task in &tasks {
                if task[2] > 0 && task[1] - i + 1 == task[2] {
                    run = true;
                }
                if i <= task[1] {
                    finished = false;
                }
            }
            if finished {
                break;
            }
            if run {
                for task in &mut tasks {
                    if i >= task[0] && i <= task[1] && task[2] > 0 {
                        task[2] -= 1;
                    }
                }
                res += 1;
            }
            i += 1;
        }
        res
    }
}
