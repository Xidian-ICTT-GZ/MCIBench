class Solution:
    def convert(self, s: str, numRows: int) -> str:
        n, r = len(s), numRows
        if r == 1 or r >= n:
            return s
        t = r * 2 - 2
        ans = []
        for i in range(r):  
            for j in range(0, n - i, t):  
                ans.append(s[j + i])  
                if 0 < i < r - 1 and j + t - i < n:
                    ans.append(s[j + t - i])  
        return ''.join(ans)
