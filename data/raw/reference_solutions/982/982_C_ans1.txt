int countTriplets(int* nums, int numsSize) {
    int *cnt = (int *)calloc(sizeof(int), 1 << 16);
    for (int i = 0; i < numsSize; i++) {
        int x = nums[i];
        for (int j = 0; j < numsSize; j++) {
            int y = nums[j];
            ++cnt[x & y];
        }
    }
    int ans = 0;
    for (int i = 0; i < numsSize; i++) {
        int x = nums[i];
        for (int mask = 0; mask < (1 << 16); ++mask) {
            if ((x & mask) == 0) {
                ans += cnt[mask];
            }
        }
    }
    free(cnt);
    return ans;
}
