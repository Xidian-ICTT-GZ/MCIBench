func diagonalPrime(nums [][]int) int {
	n, res := len(nums), 0
	for i := 0; i < n; i++ {
		if isPrime(nums[i][i]) {
			res = max(res, nums[i][i])
		}
		if isPrime(nums[i][n - i - 1]) {
			res = max(res, nums[i][n - i - 1])
		}
	}
	return res
}

func isPrime(num int) bool {
	if num == 1 {
		return false
	}
	factor := 2
	for ; factor * factor <= num; factor++ {
		if num % factor == 0 {
            return false
        }
	}
	return true
}
