class Solution:
    def numRollsToTarget(self, n: int, k: int, target: int) -> int:
        mod = 10**9 + 7
        f = [1] + [0] * target
        for i in range(1, n + 1):
            g = [0] * (target + 1)
            for j in range(target + 1):
                for x in range(1, k + 1):
                    if j - x >= 0:
                        g[j] = (g[j] + f[j - x]) % mod
            f = g
        return f[target]
