func numRollsToTarget(n int, k int, target int) int {
    mod := int(1e9 + 7)
    f := make([]int, target + 1)
    f[0] = 1;
    for i := 1; i <= n; i++ {
        for j := target; j >= 0; j-- {
            f[j] = 0
            for x := 1; x <= k; x++ {
                if j - x >= 0 {
                    f[j] = (f[j] + f[j - x]) % mod
                }
            }
        }
    }
    return f[target]
}
