int nearestValidPoint(int x, int y, int** points, int pointsSize, int* pointsColSize) {
    int best = INT_MAX, bestid = -1;
    for (int i = 0; i < pointsSize; ++i) {
        int px = points[i][0], py = points[i][1];
        if (x == px) {
            int dist = abs(y - py);
            if (dist < best) {
                best = dist;
                bestid = i;
            }
        }
        else if (y == py) {
            int dist = abs(x - px);
            if (dist < best) {
                best = dist;
                bestid = i;
            }
        }
    }
    return bestid;
}
