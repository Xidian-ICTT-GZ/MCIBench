var getTargetCopy = function(original, cloned, target) {
    const q1 = [original];
    const q2 = [cloned];
    while (q1.length > 0) {
        const node1 = q1[0], node2 = q2[0];
        q1.shift();
        q2.shift();
        if (node1 == target) {
            return node2;
        }
        if (node1.left != null) {
            q1.push(node1.left);
            q2.push(node2.left);
        }
        if (node1.right != null) {
            q1.push(node1.right);
            q2.push(node2.right);
        }
    }
    return null; 
};
