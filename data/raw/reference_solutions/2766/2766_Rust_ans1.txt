use std::collections::HashMap;

impl Solution {
    pub fn relocate_marbles(nums: Vec<i32>, move_from: Vec<i32>, move_to: Vec<i32>) -> Vec<i32> {
        let mut mp = HashMap::new();
        let mut ans = Vec::new();
        for &num in &nums {
            mp.insert(num, true);
        }
        for i in 0..move_from.len() {
            mp.remove(&move_from[i]);
            mp.insert(move_to[i], true);
        }
        for (&key, _) in &mp {
            ans.push(key);
        }
        ans.sort();
        ans
    }
}
