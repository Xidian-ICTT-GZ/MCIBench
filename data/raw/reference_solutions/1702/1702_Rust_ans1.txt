impl Solution {
    pub fn maximum_binary_string(binary: String) -> String {
        let n = binary.len();
        let mut s: Vec<char> = binary.chars().collect();
        let mut j = 0;
        for i in 0..n {
            if s[i] == '0' {
                while j <= i || (j < n && s[j] == '1') {
                    j += 1;
                }
                if j < n {
                    s[j] = '1';
                    s[i] = '1';
                    s[i + 1] = '0';
                }
            }
        }
        s.iter().collect()
    }
}
