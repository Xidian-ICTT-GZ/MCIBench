public class Solution {
    public IList<int> LuckyNumbers (int[][] matrix) {
        IList<int> ret = new List<int>();
        int m = matrix.Length, n = matrix[0].Length;
        int rowMax = 0, k = 0;
        for (int i = 0; i < m; i++) {
            int curMin = matrix[i][0];
            int c = 0;
            for (int j = 1; j < n; j++) {
                if (curMin > matrix[i][j]) {
                    curMin = matrix[i][j];
                    c = j;
                }
            }
            if (rowMax < curMin) {
                rowMax = curMin;
                k = c;
            }
        }
        for (int i = 0; i < m; i++) {
            if (rowMax < matrix[i][k]) {
                return ret;
            }
        }
        ret.Add(rowMax);
        return ret;
    }
}
