var luckyNumbers  = function(matrix) {
    const ret = [];
    const m = matrix.length, n = matrix[0].length;
    let rowMax = 0, k = 0;
    for (let i = 0; i < m; i++) {
        let curMin = matrix[i][0];
        let c = 0;
        for (let j = 1; j < n; j++) {
            if (curMin > matrix[i][j]) {
                curMin = matrix[i][j];
                c = j;
            }
        }
        if (rowMax < curMin) {
            rowMax = curMin;
            k = c;
        }
    }
    for (let i = 0; i < m; i++) {
        if (rowMax < matrix[i][k]) {
            return ret;
        }
    }
    ret.push(rowMax);
    return ret;
};
