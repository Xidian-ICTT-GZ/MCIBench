int longestOnes(int* nums, int numsSize, int k) {
    int left = 0, lsum = 0, rsum = 0;
    int ans = 0;
    for (int right = 0; right < numsSize; ++right) {
        rsum += 1 - nums[right];
        while (lsum < rsum - k) {
            lsum += 1 - nums[left];
            ++left;
        }
        ans = fmax(ans, right - left + 1);
    }
    return ans;
}
