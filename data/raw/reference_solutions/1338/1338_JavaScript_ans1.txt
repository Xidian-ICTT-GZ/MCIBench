var minSetSize = function(arr) {
    const freq = new Map();
    for (const num of arr) {
        freq.set(num, (freq.get(num) || 0) + 1);
    }
    const occ = [...freq.values()].sort((a, b) => b - a);
    let cnt = 0, ans = 0;
    for (const c of occ) {
        cnt += c;
        ans++;
        if (cnt * 2 >= arr.length) {
            break;
        }
    }
    return ans;
};
