class Solution:
    def maximumTop(self, nums: List[int], k: int) -> int:
        n = len(nums)
        res = -1   
        if n == 1:
            
            if k % 2 == 0:
                return nums[0]
        else:
            for i in range(min(n, k + 1)):
                if i != k - 1:
                    res = max(res, nums[i])
        return res
