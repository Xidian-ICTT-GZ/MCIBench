class Solution {
public:
    int minimumTime(string s) {
        int n = s.size();
        int ans = INT_MAX;
        int presum = 0, prebest = 0;
        for (int j = 0; j < n; ++j) {
            prebest = min(prebest, j - 2 * presum);
            presum += (s[j] - '0');
            ans = min(ans, prebest + 2 * presum - j);
        }
        return min(ans + n - 1, n);
    }
};
