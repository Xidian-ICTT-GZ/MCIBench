static int cmp(const void *a, const void *b) {
    return *(int *)a - *(int *)b;
}

int countPairs(int* nums, int numsSize, int target) {
    qsort(nums, numsSize, sizeof(int), cmp);
    int res = 0;
    for (int i = 0, j = numsSize - 1; i < j; i++) {
        while (i < j && nums[i] + nums[j] >= target) {
            j--;
        }
        res += j - i;
    }
    return res;
}
