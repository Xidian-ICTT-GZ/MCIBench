var differenceOfDistinctValues = function(grid) {
    const m = grid.length;
    const n = grid[0].length;
    const res = Array(m).fill(null).map(() => Array(n).fill(0));

    for (let i = 0; i < m; ++i) {
        let x = i, y = 0;
        const s = new Set();
        while (x < m && y < n) {
            res[x][y] += s.size;
            s.add(grid[x][y]);
            x += 1;
            y += 1;
        }
    }

    for (let j = 1; j < n; ++j) {
        let x = 0, y = j;
        const s = new Set();
        while (x < m && y < n) {
            res[x][y] += s.size;
            s.add(grid[x][y]);
            x += 1;
            y += 1;
        }
    }

    for (let i = 0; i < m; ++i) {
        let x = i, y = n - 1;
        const s = new Set();
        while (x >= 0 && y >= 0) {
            res[x][y] -= s.size;
            res[x][y] = Math.abs(res[x][y]);
            s.add(grid[x][y]);
            x -= 1;
            y -= 1;
        }
    }

    for (let j = n - 2; j >= 0; --j) {
        let x = m - 1, y = j;
        const s = new Set();
        while (x >= 0 && y >= 0) {
            res[x][y] -= s.size;
            res[x][y] = Math.abs(res[x][y]);
            s.add(grid[x][y]);
            x -= 1;
            y -= 1;
        }
    }

    return res;
};
