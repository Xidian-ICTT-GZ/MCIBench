var differenceOfDistinctValues = function(grid) {
    const m = grid.length, n = grid[0].length;
    const res = Array(m).fill(null).map(() => Array(n).fill(0));
    for (let i = 0; i < m; ++i) {
        for (let j = 0; j < n; ++j) {
            const s1 = new Set();
            let x = i + 1, y = j + 1;
            while (x < m && y < n) {
                s1.add(grid[x][y]);
                x += 1;
                y += 1;
            }
            const s2 = new Set();
            x = i - 1, y = j - 1;
            while (x >= 0 && y >= 0) {
                s2.add(grid[x][y]);
                x -= 1;
                y -= 1;
            }
            res[i][j] = Math.abs(s1.size - s2.size);
        }
    }
    return res;
};
