int finalPositionOfSnake(int n, char** commands, int commandsSize) {
    int ans = 0;
    for (int i = 0; i < commandsSize; i++) {
        if (commands[i][0] == 'U') {
            ans -= n;
        } else if (commands[i][0] == 'D') {
            ans += n;
        } else if (commands[i][0] == 'L') {
            ans--;
        } else {
            ans++;
        }
    }
    return ans;
}
