public class Solution {
    public int FinalPositionOfSnake(int n, IList<string> commands) {
        int ans = 0;
        foreach (string c in commands) {
            if (c[0] == 'U') {
                ans -= n;
            } else if (c[0] == 'D') {
                ans += n;
            } else if (c[0] == 'L') {
                --ans;
            } else {
                ++ans;
            }
        }
        return ans;
    }
}
