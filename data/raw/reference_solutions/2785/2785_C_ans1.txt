bool isVowel(char c) {
    return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' ||
           c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U';
}

int compare(const void *a, const void *b) {
    return *(char *)a - *(char *)b;
}

char *sortVowels(char *s) {
    int len = strlen(s);
    char *tmp = (char *)malloc(len + 1);
    int count = 0;
    
    for (int i = 0; i < len; i++) {
        if (isVowel(s[i])) {
            tmp[count++] = s[i];
        }
    }
    tmp[count] = '\0';
    
    qsort(tmp, count, sizeof(char), compare);
    
    int idx = 0;
    for (int i = 0; i < len; i++) {
        if (isVowel(s[i])) {
            s[i] = tmp[idx++];
        }
    }
    
    free(tmp);
    return s;
}
