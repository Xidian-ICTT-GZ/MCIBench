func sortVowels(s string) string {
    vowels := map[rune]bool{
        'a': true, 'e': true, 'i': true, 'o': true, 'u': true,
        'A': true, 'E': true, 'I': true, 'O': true, 'U': true,
    }
    cnt := make([]int, 58)
    for i := range cnt {
        cnt[i] = -1
    }
    for ch := range vowels {
        cnt[ch-'A'] = 0
    }
    
    for _, ch := range s {
        i := ch - 'A'
        if cnt[i] != -1 {
            cnt[i]++
        }
    }
    
    res := []rune(s)
    idx := 0
    for i, ch := range res {
        pos := ch - 'A'
        if cnt[pos] != -1 {
            for cnt[idx] <= 0 {
                idx++
            }
            res[i] = rune(idx + 'A')
            cnt[idx]--
        }
    }
    
    return string(res)
}
