class Solution:
    def unequalTriplets(self, nums: List[int]) -> int:
        nums.sort()
        res = 0
        n = len(nums)
        i = j = 0
        while i < n:
            while j < n and nums[j] == nums[i]:
                j += 1
            res += i * (j - i) * (n - j)
            i = j
        return res
