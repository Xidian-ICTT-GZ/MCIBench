var countArrangement = function(n) {
    const f = new Array(1 << n).fill(0);
    f[0] = 1;
    for (let mask = 1; mask < (1 << n); mask++) {
        const num = mask.toString(2).split('0').join('').length
        for (let i = 0; i < n; i++) {
            if ((mask & (1 << i)) !== 0 && ((num % (i + 1)) === 0 || (i + 1) % num === 0)) {
                f[mask] += f[mask ^ (1 << i)];
            }
        }
    }
    return f[(1 << n) - 1];
};
