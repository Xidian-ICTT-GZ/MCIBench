public class Solution {
    public int MinSwaps(string s) {
        int cnt = 0, mincnt = 0;
        foreach (char ch in s) {
            if (ch == '[') {
                cnt += 1;
            } else {
                cnt -= 1;
                mincnt = Math.Min(mincnt, cnt);
            }
        }
        return (-mincnt + 1) / 2;
    }
}
