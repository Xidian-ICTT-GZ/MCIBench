public class Solution {
    public int MatrixSum(int[][] nums) {
        int res = 0;
        int m = nums.Length;
        int n = nums[0].Length;
        PriorityQueue<int, int>[] pq = new PriorityQueue<int, int>[m];
        for (int i = 0; i < m; i++) {
            pq[i] = new PriorityQueue<int, int>();
            for (int j = 0; j < n; j++) {
                pq[i].Enqueue(nums[i][j], -nums[i][j]);
            }
        }
        for (int j = 0; j < n; j++) {
            int maxVal = 0;
            for (int i = 0; i < m; i++) {
                maxVal = Math.Max(maxVal, pq[i].Dequeue());
            }
            res += maxVal;
        }
        return res;
    }
}
