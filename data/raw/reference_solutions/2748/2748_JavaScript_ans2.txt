var countBeautifulPairs = function(nums) {
    let res = 0;
    let cnt = new Array(10).fill(0);

    for (let x of nums) {
        for (let y = 1; y <= 9; y++) {
            if (gcd(x % 10, y) === 1) {
                res += cnt[y];
            }
        }
        while (x >= 10) {
            x = Math.floor(x / 10);
        }
        cnt[x]++;
    }
    return res;
};

function gcd(a, b) {
    return b === 0 ? a : gcd(b, a % b);
}
