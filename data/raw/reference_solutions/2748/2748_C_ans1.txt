int gcd(int a, int b) {
    return b == 0 ? a : gcd(b, a % b);
}

int countBeautifulPairs(int* nums, int numsSize){
    int res = 0, n = numsSize;
    for (int i = 0; i < n; i++) {
        while (nums[i] >= 10) {
            nums[i] /= 10;
        }
        for (int j = i + 1; j < n; j++) {
            if (gcd(nums[i], nums[j] % 10) == 1) {
                res++;
            }
        }
    }

    return res;
}
