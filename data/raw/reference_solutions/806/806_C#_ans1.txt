public class Solution {
    public static int MAX_WIDTH = 100;

    public int[] NumberOfLines(int[] widths, string s) {
        int lines = 1;
        int width = 0;
        for (int i = 0; i < s.Length; i++) {
            int need = widths[s[i] - 'a'];
            width += need;
            if (width > MAX_WIDTH) {
                lines++;
                width = need;
            }
        }
        return new int[]{lines, width};
    }
}
