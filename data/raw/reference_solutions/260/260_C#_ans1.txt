public class Solution {
    public int[] SingleNumber(int[] nums) {
        Dictionary<int, int> freq = new Dictionary<int, int>();
        foreach (int num in nums) {
            if (freq.ContainsKey(num)) {
                ++freq[num];
            } else {
                freq.Add(num, 1);
            }
        }
        int[] ans = new int[2];
        int index = 0;
        foreach (KeyValuePair<int, int> pair in freq) {
            if (pair.Value == 1) {
                ans[index++] = pair.Key;
            }
        }
        return ans;
    }
}
