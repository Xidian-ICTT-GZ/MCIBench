class Solution {
    static final int COLOR = 61;

    public boolean isPrintable(int[][] targetGrid) {
        int m = targetGrid.length;
        int n = targetGrid[0].length;
        
        int[] up = new int[COLOR];
        Arrays.fill(up, Integer.MAX_VALUE);
        int[] down = new int[COLOR];
        int[] left = new int[COLOR];
        Arrays.fill(left, Integer.MAX_VALUE);
        int[] right = new int[COLOR];
        Set<Integer> occur = new HashSet<Integer>();
        for (int i = 0; i < m; ++i) {
            for (int j = 0; j < n; ++j) {
                int color = targetGrid[i][j];
                occur.add(color);
                up[color] = Math.min(up[color], i);
                down[color] = Math.max(down[color], i);
                left[color] = Math.min(left[color], j);
                right[color] = Math.max(right[color], j);
            }
        }

        boolean[][] g = new boolean[COLOR][COLOR];
        int[] indeg = new int[COLOR];
        int nodes = occur.size();
        for (int outside: occur) {
            for (int i = up[outside]; i <= down[outside]; ++i) {
                for (int j = left[outside]; j <= right[outside]; ++j) {
                    int color = targetGrid[i][j];
                    if (outside != color && !g[outside][color]) {
                        g[outside][color] = true;
                        ++indeg[color];
                    }
                }
            }
        }
        
        Queue<Integer> queue = new LinkedList<Integer>();
        for (int color: occur) {
            if (indeg[color] == 0) {
                queue.offer(color);
            }
        }
        while (!queue.isEmpty()) {
            int u = queue.poll();
            --nodes;
            for (int v = 1; v < COLOR; ++v) {
                if (g[u][v]) {
                    --indeg[v];
                    if (indeg[v] == 0) {
                        queue.offer(v);
                    }
                }
            }
        }

        return nodes == 0;
    }
}
