int* canSeePersonsCount(int* heights, int heightsSize, int* returnSize) {
    int n = heightsSize;
    int stackSize = 0;
    int* stack = (int*)malloc(n * sizeof(int));
    int* res = (int*)malloc(n * sizeof(int));
    *returnSize = n;

    for (int i = n - 1; i >= 0; i--) {
        int h = heights[i];
        res[i] = 0;
        while (stackSize > 0 && stack[stackSize - 1] <= h) {
            stackSize--;
            res[i] += 1;
        }
        if (stackSize > 0) {
            res[i] += 1;
        }
        stack[stackSize++] = h;
    }
    free(stack);
    return res;
}
