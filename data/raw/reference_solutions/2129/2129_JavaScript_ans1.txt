var capitalizeTitle = function(title) {
    const words = title.split(' ');
    let res = new Array();
    for (const word of words) {
        if (word.length > 2) {
            res.push(word[0].toUpperCase() + word.slice(1).toLowerCase());
        } else {
            res.push(word.toLowerCase());
        }
    }
    return res.join(" ");
};
