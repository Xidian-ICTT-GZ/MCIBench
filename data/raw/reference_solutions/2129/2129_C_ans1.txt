char* capitalizeTitle(char* title) {
    int n = strlen(title);
    int l = 0, r = 0;   
    while (r < n) {
        while (r < n && title[r] != ' ') {
            ++r;
        }
        
        if (r - l > 2) {
            title[l++] = toupper(title[l]);
        }
        while (l < r) {
            title[l++] = tolower(title[l]);
        }
        l = ++r;
    }
    return title;
}
