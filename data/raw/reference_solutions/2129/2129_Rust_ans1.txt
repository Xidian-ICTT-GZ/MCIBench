impl Solution {
    pub fn capitalize_title(title: String) -> String {
        let words: Vec<&str> = title.split(' ').collect();
        let mut res = Vec::new();
        for word in words {
            if word.len() > 2 {
                res.push(word[..1].to_uppercase() + &word[1..].to_lowercase());
            } else {
                res.push(word.to_lowercase());
            }
        }
        res.join(" ")
    }
}
