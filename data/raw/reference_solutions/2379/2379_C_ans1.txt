#define MIN(a, b) ((a) < (b) ? (a) : (b))

int minimumRecolors(char * blocks, int k) {
    int l = 0, r = 0, cnt = 0;
    while (r < k) {
        cnt += blocks[r] == 'W' ? 1 : 0;
        r++;
    }
    int res = cnt;
    int len = strlen(blocks);
    while (r < len) {
        cnt += blocks[r] == 'W' ? 1 : 0;
        cnt -= blocks[l] == 'W' ? 1 : 0;
        res = MIN(res, cnt);
        l++;
        r++;
    }
    return res;
}
