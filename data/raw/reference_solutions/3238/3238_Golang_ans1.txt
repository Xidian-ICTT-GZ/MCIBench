func winningPlayerCount(n int, pick [][]int) int {
    cnt := make([][]int, n)
	for i := range cnt {
		cnt[i] = make([]int, 11)
	}
	for _, p := range pick {
		cnt[p[0]][p[1]]++
	}

	ans := 0
	for i := 0; i < n; i++ {
		for j := 0; j <= 10; j++ {
			if cnt[i][j] > i {
				ans++
				break
			}
		}
	}
	return ans
}
