public class Solution {
    public int MaxWidthOfVerticalArea(int[][] points) {
        Array.Sort(points, (a, b) => a[0] - b[0]);
        int mx = 0;
        for (int i = 1; i < points.Length; i++) {
            mx = Math.Max(points[i][0] - points[i - 1][0], mx);
        }
        return mx;
    }
}
