class Solution {
    public String makeFancyString(String s) {
        StringBuilder res = new StringBuilder();   
        
        for (char ch : s.toCharArray()) {
            int n = res.length();
            if (n >= 2 && res.charAt(n - 1) == ch && res.charAt(n - 2) == ch) {
                
                continue;
            }
            
            res.append(ch);
        }
        return res.toString();
    }
}
