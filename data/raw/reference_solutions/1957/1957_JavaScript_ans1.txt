var makeFancyString = function(s) {
    let res = [];   
    
    for (const ch of s) {
        const n = res.length;
        if (n >= 2 && res[n - 1] === ch && res[n - 2] === ch) {
            
            continue;
        }
        
        res.push(ch);
    }
    return res.join('');
};
