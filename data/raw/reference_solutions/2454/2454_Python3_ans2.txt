class Solution:
    def secondGreaterElement(self, nums: List[int]) -> List[int]:
        n = len(nums)
        res = [-1] * n
        st1, st2 = [], []
        
        for i in range(n):
            v = nums[i]
            while len(st2) and nums[st2[-1]] < v:
                res[st2[-1]] = v
                st2.pop()
            pos = len(st1) - 1
            while pos >= 0 and nums[st1[pos]] < v:
                pos -=1
            st2 += st1[pos + 1:]
            del st1[pos + 1:]
            st1.append(i)
        return res
