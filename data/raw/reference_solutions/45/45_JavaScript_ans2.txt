var jump = function(nums) {
    let maxPos = 0, end = 0, steps = 0;
    for (let i = 0; i < nums.length - 1; ++i) {
        if (maxPos >= i) {
            maxPos = Math.max(maxPos, i + nums[i]);
            if (i == end) {
                end = maxPos;
                ++steps;
            }
        }
    }
    return steps;
};
