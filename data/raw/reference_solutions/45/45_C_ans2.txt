int jump(int* nums, int numsSize) {
    int maxPos = 0, end = 0, steps = 0;
    for (int i = 0; i < numsSize - 1; ++i) {
        if (maxPos >= i) {
            maxPos = (maxPos > i + nums[i]) ? maxPos : i + nums[i];
            if (i == end) {
                end = maxPos;
                ++steps;
            }
        }
    }
    return steps;
}
