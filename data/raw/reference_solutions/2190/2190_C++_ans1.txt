class Solution {
public:
    int mostFrequent(vector<int>& nums, int key) {
        int n = nums.size();
        unordered_map<int, int> freq;   
        for (int i = 0; i < n - 1; ++i) {
            if (nums[i] == key) {
                ++freq[nums[i+1]];
            }
        }
        
        int maxfreq = 0;
        int res = 0;
        for (const auto& [v, f]: freq) {
            if (f > maxfreq) {
                res = v; 
                maxfreq = f;
            }
        }
        return res;
    }
};
