func minNumber(nums1 []int, nums2 []int) int {
    var same = func() int {
        s := make(map[int]bool)
        x := 10
        for _, num := range nums1 {
            s[num] = true
        }
        for _, num := range nums2 {
            if _, ok := s[num]; ok {
                x = min(x, num)
            }
        }
        if x == 10 {
            return -1
        }
        return x
    }

    if x := same(); x != -1 {
        return x;
    }
    x, y := nums1[0], nums2[0]
    for _, num := range nums1 {
        x = min(x, num)
    }
    for _, num := range nums2 {
        y = min(y, num)
    }
    return min(x * 10 + y, y * 10 + x)
}

func min(x int, y int) int {
    if x > y {
        return y
    }
    return x
}
