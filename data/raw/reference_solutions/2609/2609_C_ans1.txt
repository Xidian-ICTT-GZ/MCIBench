int findTheLongestBalancedSubstring(char * s) {
    int res = 0;
    int count[2] = {0};
    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] == '1') {
            count[1]++;
            res = fmax(res, 2 * fmin(count[0], count[1]));
        } else if (i == 0 || s[i - 1] == '1') {
            count[0] = 1;
            count[1] = 0;
        } else {
            count[0]++;
        }
    }
    return res;    
}
