class Solution:
    def resultsArray(self, nums: List[int], k: int) -> List[int]:
        n = len(nums)
        ans = [-1] * (n - k + 1)
        for i in range(n - k + 1):
            valid = True
            for j in range(i + 1, i + k):
                if nums[j] - nums[j - 1] != 1:
                    valid = False
                    break
            if valid:
                ans[i] = nums[i + k - 1]
                
        return ans
