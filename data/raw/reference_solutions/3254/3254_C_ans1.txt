int* resultsArray(int* nums, int numsSize, int k, int* returnSize) {
    int n = numsSize;
    *returnSize = n - k + 1;
    int* ans = (int*)malloc(*returnSize * sizeof(int));
    for (int i = 0; i < *returnSize; i++) {
        ans[i] = -1;
    }
    for (int i = 0; i <= n - k; i++) {
        bool valid = true;
        for (int j = i + 1; j < i + k; j++) {
            if (nums[j] - nums[j - 1] != 1) {
                valid = false;
                break;
            }
        }
        if (valid) {
            ans[i] = nums[i + k - 1];
        }
    }
    return ans;
}
