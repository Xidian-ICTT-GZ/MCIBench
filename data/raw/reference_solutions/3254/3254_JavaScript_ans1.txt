var resultsArray = function(nums, k) {
    const n = nums.length;
    const ans = new Array(n - k + 1).fill(-1);
    for (let i = 0; i <= n - k; i++) {
        let valid = true;
        for (let j = i + 1; j < i + k; j++) {
            if (nums[j] - nums[j - 1] !== 1) {
                valid = false;
                break;
            }
        }
        if (valid) {
            ans[i] = nums[i + k - 1];
        }
    }
    return ans;
};
