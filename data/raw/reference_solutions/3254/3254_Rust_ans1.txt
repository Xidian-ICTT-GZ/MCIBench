impl Solution {
    pub fn results_array(nums: Vec<i32>, k: i32) -> Vec<i32> {
        let n = nums.len();
        let mut ans = vec![-1; n - k as usize + 1];
        for i in 0..= n - k as usize {
            let mut valid = true;
            for j in i + 1..i + k as usize {
                if nums[j] - nums[j - 1] != 1 {
                    valid = false;
                    break;
                }
            }
            if valid {
                ans[i] = nums[i + k as usize - 1];
            }
        }
        ans
    }
}
