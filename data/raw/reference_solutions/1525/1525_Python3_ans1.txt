class Solution:
    def numSplits(self, s: str) -> int:
        n = len(s)
        left, right = [0] * (n + 2), [0] * (n + 2)

        rec_left = [False] * 26
        rec_right = [False] * 26
        for i in range(1, n + 1):
            c = ord(s[i - 1]) - ord("a")
            if rec_left[c]:
                left[i] = left[i - 1]
            else:
                rec_left[c] = True
                left[i] = left[i - 1] + 1
        
        for i in range(n, 0, -1):
            c = ord(s[i - 1]) - ord("a")
            if (rec_right[c]):
                right[i] = right[i + 1]
            else:
                rec_right[c] = True
                right[i] = right[i + 1] + 1
        
        ret = sum(1 for i in range(1, n) if left[i] == right[i + 1])
        return ret
