public class Solution {
    public string BreakPalindrome(string palindrome) {
        int n = palindrome.Length;
        if (n == 1) {
            return "";
        }
        char[] data = palindrome.ToCharArray();
        for (int i = 0; i * 2 + 1 < n; i++) {
            if (data[i] != 'a') {
                data[i] = 'a';
                return new string(data);
            }
        }
        data[n - 1] = 'b';
        return new string(data);
    }
}
