func breakPalindrome(palindrome string) string {
    n := len(palindrome)
    if n == 1 {
        return ""
    }
    data := []byte(palindrome)
    for i := 0; i * 2 + 1 < n; i++ {
        if data[i] != 'a' {
            data[i] = 'a'
            return string(data)
        }
    }
    data[n-1]++
    return string(data)
}
