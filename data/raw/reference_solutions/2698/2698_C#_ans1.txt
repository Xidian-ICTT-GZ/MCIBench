public class Solution {
    public int PunishmentNumber(int n) {
        int res = 0;
        for (int i = 1; i <= n; i++) {
            string s = (i * i).ToString();
            if (DFS(s, 0, 0, i)) {
                res += i * i;
            }
        }
        return res;
    }

    public bool DFS(string s, int pos, int tot, int target) {
        if (pos == s.Length) {
            return tot == target;
        } 
        int sum = 0;
        for (int i = pos; i < s.Length; i++) {
            sum = sum * 10 + s[i] - '0';
            if (sum + tot > target) {
                break;
            }
            if (DFS(s, i + 1, sum + tot, target)) {
                return true;
            }
        }
        return false;
    }
}
