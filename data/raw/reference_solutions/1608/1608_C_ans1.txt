static inline int cmp(const void *pa, const void *pb) {
    return *(int *)pb - *(int *)pa;
}

int specialArray(int* nums, int numsSize) {
    qsort(nums, numsSize, sizeof(int), cmp);
    for (int i = 1; i <= numsSize; ++i) {
        if (nums[i - 1] >= i && (i == numsSize || nums[i] < i)) {
            return i;
        }
    }
    return -1;
}
