var maxScore = function(nums, x) {
    let res = nums[0];
    let dp = [-Infinity, -Infinity];
    dp[nums[0] % 2] = nums[0];
    for (let i = 1; i < nums.length; i++) {
        let parity = nums[i] % 2;
        let cur = Math.max(dp[parity] + nums[i], dp[1 - parity] - x + nums[i]);
        res = Math.max(res, cur);
        dp[parity] = Math.max(dp[parity], cur);
    }
    return res;
};
