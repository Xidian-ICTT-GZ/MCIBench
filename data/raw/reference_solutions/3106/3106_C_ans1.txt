char* getSmallestString(char* s, int k) {
    int len = strlen(s);
    for (int i = 0; i < len; ++i) {
        int dis = fmin(s[i] - 'a', 'z' - s[i] + 1);
        if (dis <= k) {
            s[i] = 'a';
            k -= dis;
        }
        else {
            s[i] -= k;
            break;
        }
    }
    return s;
}
