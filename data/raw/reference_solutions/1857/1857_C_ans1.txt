struct ListNode *createListNode(int val) {
    struct ListNode *obj = (struct ListNode *)malloc(sizeof(struct ListNode));
    obj->val = val;
    obj->next = NULL;
    return obj;
}

void freeList(struct ListNode *list) {
    while (list) {
        struct ListNode *p = list;
        list = list->next;
        free(p);
    }
}

int largestPathValue(char* colors, int** edges, int edgesSize, int* edgesColSize) {
    int n = strlen(colors);
    
    struct ListNode *g[n];
    
    int indeg[n];
    for (int i = 0; i < n; i++) {
        g[i] = NULL;
        indeg[i] = 0;
    }
    for (int i = 0; i < edgesSize; ++i) {
        int x = edges[i][0], y = edges[i][1];
        ++indeg[y];
        struct ListNode *node = createListNode(y);
        node->next = g[x];
        g[x] = node;
    }

    
    
    int found = 0;
    int f[n][26];
    int queue[n];
    int head = 0, tail = 0;
    memset(f, 0, sizeof(f));
    for (int i = 0; i < n; ++i) {
        if (indeg[i] == 0) {
            queue[tail++] = i;
        }
    }

    while (head != tail) {
        ++found;
        int u = queue[head];
        head++;
        
        ++f[u][colors[u] - 'a'];
        
        for (struct ListNode *p = g[u]; p != NULL; p = p->next) {
            int v = p->val;
            --indeg[v];
            
            for (int c = 0; c < 26; ++c) {
                f[v][c] = (f[v][c] > f[u][c]) ? f[v][c] : f[u][c];
            }
            if (indeg[v] == 0) {
                queue[tail++] = v;
            }
        }
    }

    if (found != n) {
        return -1;
    }
    int ans = 0;
    for (int i = 0; i < n; ++i) {
        for (int c = 0; c < 26; ++c) {
            if (f[i][c] > ans) {
                ans = f[i][c];
            }
        }
        freeList(g[i]);
    }
    
    return ans;
}
