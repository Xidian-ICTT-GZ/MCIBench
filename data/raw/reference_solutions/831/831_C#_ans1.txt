public class Solution {
    string[] country = {"", "+*-", "+**-", "+***-"};

    public string MaskPII(string s) {
        int at = s.IndexOf("@");
        if (at > 0) {
            s = s.ToLower();
            return (s[0] + "*****" + s.Substring(at - 1)).ToLower();
        }
        StringBuilder sb = new StringBuilder();
        foreach (char c in s) {
            if (char.IsDigit(c)) {
                sb.Append(c);
            }
        }
        s = sb.ToString();
        return country[s.Length - 10] + "***-***-" + s.Substring(s.Length - 4);
    }
}
