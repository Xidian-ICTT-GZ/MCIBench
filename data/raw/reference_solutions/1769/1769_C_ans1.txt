int* minOperations(char * boxes, int* returnSize) {
    int n = strlen(boxes);
    int *res = (int *)malloc(sizeof(int) * n);
    for (int i = 0; i < n; i++) {
        int sm = 0;
        for (int j = 0; j < n; j++) {
            if (boxes[j] == '1') {
                sm += abs(j - i);
            }
        }
        res[i] = sm;
    }
    *returnSize = n;
    return res;
}
