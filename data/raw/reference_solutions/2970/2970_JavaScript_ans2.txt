var incremovableSubarrayCount = function(nums) {
    const n = nums.length;
    let res = 0;
    let l = 1;
    while (l < n && nums[l - 1] < nums[l]) {
        l++;
    }
    res += l + (l < n ? 1 : 0);
    for (let r = n - 2; r >= 0; r--) {
        while (l > 0 && nums[l - 1] >= nums[r + 1]) {
            l--;
        }
        res += l + (l <= r ? 1 : 0);
        if (nums[r] >= nums[r + 1]) {
            break;
        }
    }
    return res;
};
