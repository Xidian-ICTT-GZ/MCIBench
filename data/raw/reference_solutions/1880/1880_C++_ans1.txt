class Solution {
public:
    bool isSumEqual(string firstWord, string secondWord, string targetWord) {
        auto decode = [](const string& s) -> int {
            int res = 0;
            for (char ch: s){
                res *= 10;
                res += ch - 'a';
            }
            return res;
        };
        
        return decode(firstWord) + decode(secondWord) == decode(targetWord);
    }
};
