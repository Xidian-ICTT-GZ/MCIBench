class Solution:
    def maximumANDSum(self, nums: List[int], numSlots: int) -> int:
        n = len(nums)
        mask_max = 3 ** numSlots

        f = [0] * mask_max        
        for mask in range(1, mask_max):
            cnt, dummy = 0, mask

            for i in range(numSlots):
                cnt += dummy % 3
                dummy //= 3
            
            if cnt > n:
                continue
            
            dummy, w = mask, 1
            for i in range(numSlots):
                has = dummy % 3
                if has > 0:
                    f[mask] = max(f[mask], f[mask - w] + (nums[cnt - 1] & (i + 1)))
                dummy //= 3
                w *= 3
        
        return max(f)
