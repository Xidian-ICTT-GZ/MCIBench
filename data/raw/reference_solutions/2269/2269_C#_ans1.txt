public class Solution {
    public int DivisorSubstrings(int num, int k) {
        string s = num.ToString();   
        int n = s.Length;
        int res = 0;
        for (int i = 0; i <= n - k; ++i) {
            
            int tmp = int.Parse(s.Substring(i, k));
            if (tmp != 0 && num % tmp == 0) {
                ++res;
            }
        }
        return res;
    }
}
