use std::collections::VecDeque;
use std::cmp::{max, min};

impl Solution {
    pub fn max_task_assign(tasks: Vec<i32>, workers: Vec<i32>, pills: i32, strength: i32) -> i32 {
        let n = tasks.len();
        let m = workers.len();
        let mut tasks = tasks;
        let mut workers = workers;
        tasks.sort();
        workers.sort();
        
        let check = |mid: usize| -> bool {
            let mut p = pills;
            let mut ws = VecDeque::new();
            let mut ptr = m - 1;
            
            for i in (0..mid).rev() {
                while ptr as i32 >= (m - mid) as i32 && workers[ptr] + strength >= tasks[i] {
                    ws.push_front(workers[ptr]);
                    ptr -= 1;
                }
                if ws.is_empty() {
                    return false;
                }
                
                if *ws.back().unwrap() >= tasks[i] {
                    ws.pop_back();
                } else {
                    if p == 0 {
                        return false;
                    }
                    p -= 1;
                    ws.pop_front();
                }
            }
            true
        };
        
        let mut left = 1;
        let mut right = min(m, n);
        let mut ans = 0;
        while left <= right {
            let mid = (left + right) / 2;
            if check(mid) {
                ans = mid as i32;
                left = mid + 1;
            } else {
                right = mid - 1;
            }
        }
        ans
    }
}
