func minimumSubarrayLength(nums []int, k int) int {
    n := len(nums)
    res := n + 1
    for i := 0; i < n; i++ {
        value := 0
        for j := i; j < n; j++ {
            value |= nums[j]
            if value >= k {
                res = min(res, j-i+1)
                break
            }
        }
    }
    if res == n + 1 {
        return -1
    }
    return res
}
