int minimumSubarrayLength(int* nums, int numsSize, int k) {
    int res = INT_MAX;
    for (int i = 0; i < numsSize; i++) {
        int value = 0;
        for (int j = i; j < numsSize; j++) {
            value |= nums[j];
            if (value >= k) {
                res = (res < j - i + 1) ? res : j - i + 1;
                break;
            }
        }
    }
    return res == INT_MAX ? -1 : res;
}
