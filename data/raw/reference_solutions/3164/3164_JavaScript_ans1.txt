var numberOfPairs = function(nums1, nums2, k) {
    const count = {};
    const count2 = {};
    let res = 0, max1 = 0;
    for (let num of nums1) {
        count[num] = (count[num] || 0) + 1;
        max1 = Math.max(max1, num);
    }
    for (let num of nums2) {
        count2[num] = (count2[num] || 0) + 1;
    }
    for (let a in count2) {
        let cnt = count2[a];
        for (let b = a * k; b <= max1; b += a * k) {
            if (b in count) {
                res += count[b] * cnt;
            }
        }
    }
    return res;
};
