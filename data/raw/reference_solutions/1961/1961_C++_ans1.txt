class Solution {
public:
    bool isPrefixString(string s, vector<string>& words) {
        int i = 0;
        int n = s.size();
        for (const string& word : words){
            for (char ch : word){
                if (i < n && ch == s[i]){
                    ++i;
                }
                else{
                    
                    return false;
                }
            }
            if (i == n){
                
                return true;
            }
        }
        
        return false;
    }
};
