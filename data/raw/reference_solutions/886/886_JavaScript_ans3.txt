var possibleBipartition = function(n, dislikes) {
    const fa = new Array(n + 1).fill(-1);
    const g = new Array(n + 1).fill(0);
    for (let i = 0; i <= n; ++i) {
        g[i] = [];
    }
    for (const p of dislikes) {
        g[p[0]].push(p[1]);
        g[p[1]].push(p[0]);
    }
    for (let i = 1; i <= n; ++i) {
        for (let j = 0; j < g[i].length; ++j) {
            unit(g[i][0], g[i][j], fa);
            if (isconnect(i, g[i][j], fa)) {
                return false;
            }
        }
    }
    return true;
}

const unit = (x, y, fa) => {
    x = findFa(x, fa);
    y = findFa(y, fa);
    if (x === y) {
        return ;
    }
    if (fa[x] < fa[y]) {
        const temp = x;
        x = y;
        y = temp;
    }
    fa[x] += fa[y];
    fa[y] = x;
}

const isconnect = (x, y, fa) => {
    x = findFa(x, fa);
    y = findFa(y, fa);
    return x === y;
}

const findFa = (x, fa) => {
    return fa[x] < 0 ? x : (fa[x] = findFa(fa[x], fa));
}
