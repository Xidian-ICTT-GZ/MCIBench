impl Solution {
    pub fn count_subarrays(nums: Vec<i32>, min_k: i32, max_k: i32) -> i64 {
        let mut res = 0i64;
        let mut border = -1;
        let mut last_min = -1;
        let mut last_max = -1;
        for i in 0..nums.len() {
            let num = nums[i];
            if num < min_k || num > max_k {
                last_max = -1;
                last_min = -1;
                border = i as i32;
            }
            if num == min_k {
                last_min = i as i32;
            }
            if num == max_k {
                last_max = i as i32;
            }
            if last_min != -1 && last_max != -1 {
                res += (last_min.min(last_max) - border) as i64;
            }
        }
        res
    }
}
