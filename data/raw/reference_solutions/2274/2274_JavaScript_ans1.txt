var maxConsecutive = function(bottom, top, special) {
    special.push(bottom - 1);
    special.push(top + 1);
    special.sort((a, b) => a - b);
    let ans = 0;
    for (let i = 0; i < special.length - 1; ++i) {
        ans = Math.max(ans, special[i + 1] - special[i] - 1);
    }
    return ans;
};
