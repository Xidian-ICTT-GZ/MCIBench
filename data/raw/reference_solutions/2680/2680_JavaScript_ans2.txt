var maximumOr = function(nums, k) {
    let orSum = 0n, multiBits = 0n;
    for (let x of nums) {
        multiBits |= BigInt(x) & orSum;
        orSum |= BigInt(x);
    }

    let res = 0n;
    for (let x of nums) {
        res = Math.max(Number(res), Number((orSum ^ BigInt(x)) | (BigInt(x) << BigInt(k)) | multiBits));
    }
    return Number(res);
};
