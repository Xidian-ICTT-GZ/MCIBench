class Solution:
    def canBeIncreasing(self, nums: List[int]) -> bool:
        n = len(nums)
        
        def check(idx: int) -> bool:
            for i in range(1, n - 1):
                prev, curr = i - 1, i
                if prev >= idx:
                    prev += 1
                if curr >= idx:
                    curr += 1
                if nums[curr] <= nums[prev]:
                    return False
            return True
        
        for i in range(1, n):
            
            if nums[i] <= nums[i-1]:
                return check(i) or check(i - 1)
        return True
