long long numberOfWeeks(int* milestones, int milestonesSize) {
    
    int longest = 0;
    long long rest = 0;
    for (int i = 0; i < milestonesSize; ++i) {
        longest = fmax(longest, milestones[i]);
        rest += milestones[i];
    }
    
    rest -= longest;
    if (longest > rest + 1) {
        
        return rest * 2 + 1; 
    } else {
        
        return longest + rest;
    }
}
