class Solution {
    public long numberOfWeeks(int[] milestones) {
        int longest = milestones[0]; 
        long rest = 0;
        for (int count : milestones) {
            longest = Math.max(longest, count);
            rest += count;
        }
        
        rest -= longest;
        if (longest > rest + 1) {
            
            return rest * 2 + 1;
        } else {
            
            return longest + rest;
        }
    }
}
