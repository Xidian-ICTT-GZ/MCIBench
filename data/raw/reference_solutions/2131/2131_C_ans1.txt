int longestPalindrome(char** words, int wordsSize) {
    
    int freq[26 * 26] = {0}; 
    for (int i = 0; i < wordsSize; i++) {
        int idx = (words[i][0] - 'a') * 26 + (words[i][1] - 'a');
        freq[idx]++;
    }
    int res = 0;
    int mid = 0;
    for (int i = 0; i < 26 * 26; i++) {
        if (freq[i] == 0) continue;
        int rev = (i % 26) * 26 + (i / 26); 
        if (i == rev) {
            if (freq[i] % 2 == 1) {
                mid = 1;
            }
            res += 2 * (freq[i] / 2 * 2);
        } else if (i > rev) {
            res += 4 * fmin(freq[i], freq[rev]);
        }
    }
    if (mid) {
        res += 2;
    }
    return res;
}
