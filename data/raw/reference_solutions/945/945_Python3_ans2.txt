class Solution:
    def minIncrementForUnique(self, nums: List[int]) -> int:
        nums.sort()
        nums.append(100000)
        ans = taken = 0

        for i in range(1, len(nums)):
            if nums[i - 1] == nums[i]:
                taken += 1
                ans -= nums[i]
            else:
                give = min(taken, nums[i] - nums[i - 1] - 1)
                ans += give * (give + 1) // 2 + give * nums[i - 1]
                taken -= give

        return ans
