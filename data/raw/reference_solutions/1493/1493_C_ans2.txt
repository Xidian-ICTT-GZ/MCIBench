int longestSubarray(int* nums, int numsSize) {
    int ans = 0;
    int p0 = 0, p1 = 0;
    for (int i = 0; i < numsSize; i++) {
        if (nums[i] == 0) {
            p1 = p0;
            p0 = 0;
        } else {
            ++p0;
            ++p1;
        }
        ans = fmax(ans, p1);
    }
    if (ans == numsSize) {
        --ans;
    }
    return ans;
}
