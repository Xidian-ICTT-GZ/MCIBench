var divideString = function(s, k, fill) {
    const res = []; 
    const n = s.length;
    let curr = 0; 
    
    while (curr < n) {
        const end = Math.min(curr + k, n);
        res.push(s.slice(curr, end));
        curr += k;
    }
    
    const last = res[res.length - 1];
    if (last.length < k) {
        res[res.length - 1] = last + fill.repeat(k - last.length);
    }
    return res;
};
