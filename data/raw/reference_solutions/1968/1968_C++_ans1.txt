class Solution {
public:
    vector<int> rearrangeArray(vector<int>& nums) {
        
        sort(nums.begin(), nums.end());
        int n = nums.size();
        int m = (n + 1) / 2;
        vector<int> res;
        for (int i = 0; i < m; ++i){
            
            res.push_back(nums[i]);
            if (i + m < n){
                
                res.push_back(nums[i + m]);
            }
        }
        return res;
    }
};
