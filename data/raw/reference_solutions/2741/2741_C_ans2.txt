#define MOD 1000000007

int specialPerm(int* nums, int numsSize) {
    int n = numsSize;
    int f[1 << n][n];
    for (int i = 0; i < (1 << n); i++) {
        for (int j = 0; j < n; j++) {
            f[i][j] = 0;
        }
    }

    for (int i = 0; i < n; i++) {
        f[1 << i][i] = 1;
    }

    for (int state = 1; state < (1 << n); state++) {
        for (int i = 0; i < n; i++) {
            if (!(state >> i & 1)) {
                continue;
            }
            for (int j = 0; j < n; j++) {
                if (i == j || !(state >> j & 1)) {
                    continue;
                }
                int x = nums[i];
                int y = nums[j];
                if (x % y != 0 && y % x != 0) {
                    continue;
                }
                f[state][i] = (f[state][i] + f[state ^ (1 << i)][j]) % MOD;
            }
        }
    }

    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum = (sum + f[(1 << n) - 1][i]) % MOD;
    }

    return sum;
}
