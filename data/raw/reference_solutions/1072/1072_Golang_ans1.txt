func maxEqualRowsAfterFlips(matrix [][]int) int {
    m, n := len(matrix), len(matrix[0])
    mp := make(map[string]int)
    for i := 0; i < m; i++ {
        arr := make([]byte, n)
        for j := 0; j < n; j++ {
            
            if matrix[i][j] ^ matrix[i][0] == 0 {
                arr[j] = '0'
            } else {
                arr[j] = '1'
            }
        }
        s := string(arr)
        mp[s]++
    }
    res := 0
    for _, value := range mp {
        if value > res {
            res = value
        }
    }
    return res
}
