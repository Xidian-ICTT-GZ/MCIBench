int minimumSum(int* nums, int numsSize) {
    int res = 1000;
    for (int i = 0; i < numsSize; i++) {
        for (int j = i + 1; j < numsSize; j++) {
            for (int k = j + 1; k < numsSize; k++) {
                if (nums[i] < nums[j] && nums[k] < nums[j]) {
                    int sum = nums[i] + nums[j] + nums[k];
                    res = res < sum ? res : sum;
                }
            }
        }
    }
    return res < 1000 ? res : -1;
}
