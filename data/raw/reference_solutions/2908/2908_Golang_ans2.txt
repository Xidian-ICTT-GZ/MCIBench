func min(a, b int) int {
    if (a < b) {
        return a
    }
    return b
}

func minimumSum(nums []int) int {
    n := len(nums)
    res := 1000
    mn := 1000
    left := make([]int, n)
    left[0] = 0
    for i := 1; i < n; i++ {
        left[i] = int(math.Min(float64(nums[i - 1]), float64(mn)))
        mn = left[i]
    }

    right := nums[n - 1]
    for i := n - 2; i > 0; i-- {
        if left[i] < nums[i] && nums[i] > right {
            res = min(res, left[i] + nums[i] + right)
        }
        right = min(right, nums[i])
    }

    if res < 1000 {
        return res
    }
    return -1
}
