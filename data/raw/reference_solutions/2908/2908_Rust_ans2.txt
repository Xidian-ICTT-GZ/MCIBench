impl Solution {
    pub fn minimum_sum(nums: Vec<i32>) -> i32 {
        let n = nums.len();
        let mut res = 1000;
        let mut mn = 1000;
        let mut left = vec![0; n];
        for i in 1..n {
            mn = nums[i - 1].min(mn);
            left[i]  = mn;
        }

        let mut right = nums[n - 1];
        for i in (1..n - 1).rev() {
            if left[i] < nums[i] && nums[i] > right {
                res = res.min(left[i] + nums[i] + right);
            }
            right = right.min(nums[i]);
        }

        if res < 1000 {
            return res
        }
        return -1
    }
}
