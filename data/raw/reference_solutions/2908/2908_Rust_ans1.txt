impl Solution {
    pub fn minimum_sum(nums: Vec<i32>) -> i32 {
        let n = nums.len();
        let mut res = 1000;
        for i in 0..n {
            for j in i + 1..n {
                for k in j + 1..n {
                    if nums[i] < nums[j] && nums[k] < nums[j] {
                        res = res.min(nums[i] + nums[j] + nums[k]);
                    }
                }
            }
        }
        if res < 1000 {
            res
        } else {
            -1
        }
    }
}
