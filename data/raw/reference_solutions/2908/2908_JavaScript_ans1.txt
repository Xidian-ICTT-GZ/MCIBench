var minimumSum = function(nums) {
    const n = nums.length;
    let res = 1000;
    for (let i = 0; i < n; i++) {
        for (let j = i + 1; j < n; j++) {
            for (let k = j + 1; k < n; k++) {
                if (nums[i] < nums[j] && nums[k] < nums[j]) {
                    res = Math.min(res, nums[i] + nums[j] + nums[k]);
                }
            }
        }
    }
    return res < 1000 ? res : -1;
};
